    """Return a case-insensitive absolute identity for a Windows executable."""

def _show_bootstrap_error(detail: str) -> None:
    """Show an error even when the current interpreter has no console."""

    if sys.stderr is not None:
        sys.stderr.write(detail + "\n")
    try:
        import ctypes

        ctypes.windll.user32.MessageBoxW(0, detail, "DOE 分析工具", 0x10)
    except (AttributeError, OSError):
        pass


def _ensure_configured_python() -> None:
    """Relaunch this file with the configured portable Python when necessary."""

    configured_pythonw = PYTHONW_EXE.expanduser()
    configured_python = configured_pythonw.with_name("python.exe")
    current_key = _executable_path_key(Path(sys.executable))
    configured_keys = {
        _executable_path_key(configured_pythonw),
        _executable_path_key(configured_python),
    }
    if current_key in configured_keys:
        return

    if not configured_pythonw.is_file():
        _show_bootstrap_error(
            "找不到指定的 Python 執行檔：\n{}\n\n"
            "請用文字編輯器開啟 doe_gui.pyw，修改檔案最上方的 PYTHONW_EXE。".format(configured_pythonw)
        )
        raise SystemExit(1)

    current_is_console_python = Path(sys.executable).name.lower() == "python.exe"
    target_executable = (
        configured_python if current_is_console_python and configured_python.is_file() else configured_pythonw
    )
    script_path = Path(__file__).resolve()
    command = [str(target_executable), str(script_path)] + sys.argv[1:]
    try:
        if current_is_console_python or "--smoke-test" in sys.argv[1:]:
            return_code = subprocess.call(command, cwd=str(script_path.parent))
            raise SystemExit(return_code)
        subprocess.Popen(command, cwd=str(script_path.parent))
    except OSError as exc:
        _show_bootstrap_error("無法使用指定的 Python 啟動程式：\n{}\n\n{}".format(target_executable, exc))
        raise SystemExit(1)
    raise SystemExit(0)

@dataclass(frozen=True)
class OrthogonalArray:
    """兩水準正交表，以 table column 索引 1..N-1 表示（column j 是長度 N 的 ±1 向量）。"""

    p: int  # 基底欄數
    N: int  # run 數 (2^p)
    base_columns: Tuple[Tuple[int, ...], ...]  # G_1..G_p，各長度 N
    columns: Dict[int, Tuple[int, ...]]  # table column j (1..N-1) -> 長度 N 的 ±1 向量
    generators: Dict[int, Tuple[int, ...]]  # table column j -> 組成它的基底欄編號 (1-indexed)

    def interaction_column(self, col_a: int, col_b: int) -> int:
        """兩個 table column 交互作用對應到的 table column（正交表的 XOR 性質）。"""

        if col_a == col_b:
            raise ValueError("交互作用需要兩個不同的欄位")
        return col_a ^ col_b


def generate_orthogonal_array(k: int, min_runs: Optional[int] = None) -> OrthogonalArray:
    """對 k 個兩水準因子，產生標準正交表（L4/L8/L16/L32...）。

    演算法：p = ceil(log2(k+1))，N = 2^p；建立 p 個基底欄（標準 Yates order，
    G_m 以區塊大小 N/2^m 交替 ±1）；table column j（1..N-1）= 依 j 的二進位分解，
    把對應的基底欄逐一相乘。factor_i 依序指派給 table column i（嚴格遞增，
    不是先填滿基底欄、再填交互欄——已用實際 L8 資料逐欄驗證此指派順序）。
    (k+1)..(N-1) 為 spare column，供 ANOVA 誤差估計使用。

    `min_runs`：實際資料的 run 數下限（例如使用者做的是比最小需求更寬裕的全因子
    設計，run 數比 k 理論上的最小需求多）。給了就取「k 的最小需求」跟「min_runs」
    兩者較大的 N（都要滿足 N=2^p），不會因為 k 較小就把 N 縮小、截斷實際資料。
    """

    if k < 1:
        raise ValueError("因子數必須 >= 1")

    p = 1
    while (1 << p) - 1 < k:
        p += 1
    if min_runs is not None:
        while (1 << p) < min_runs:
            p += 1
    N = 1 << p

    base_columns: List[Tuple[int, ...]] = []
    for m in range(1, p + 1):
        block = N // (1 << m)
        col = tuple(-1 if (r // block) % 2 == 0 else 1 for r in range(N))
        base_columns.append(col)

    columns: Dict[int, Tuple[int, ...]] = {}
    generators: Dict[int, Tuple[int, ...]] = {}
    for j in range(1, N):
        bits = [b for b in range(p) if (j >> b) & 1]
        col = [1] * N
        for b in bits:
            base = base_columns[b]
            col = [a * x for a, x in zip(col, base)]
        columns[j] = tuple(col)
        generators[j] = tuple(b + 1 for b in bits)

    return OrthogonalArray(
        p=p,
        N=N,
        base_columns=tuple(base_columns),
        columns=columns,
        generators=generators,
    )


@dataclass(frozen=True)
class AliasInfo:
    """兩個已指派因子交互作用的別名（confounding）狀態。"""

    factor_a: int  # table column
    factor_b: int  # table column
    interaction_column: int
    confounded_with: Optional[int]  # 若交互作用欄位剛好是另一已指派因子的 table column


def detect_alias_structure(k: int, oa: OrthogonalArray) -> List[AliasInfo]:
    """對所有已指派因子兩兩配對，判斷其交互作用是否與第三個因子混淆。"""

    infos: List[AliasInfo] = []
    for a in range(1, k + 1):
        for b in range(a + 1, k + 1):
            col = oa.interaction_column(a, b)
            confounded = col if col <= k else None
            infos.append(AliasInfo(factor_a=a, factor_b=b, interaction_column=col, confounded_with=confounded))
    return infos


def spare_columns(k: int, oa: OrthogonalArray) -> List[int]:
    """未指派給任何因子的 table column（供 ANOVA 誤差估計使用）。"""

    return list(range(k + 1, oa.N))

class SNDirection(Enum):
    SMALLER = "smaller"  # Smaller-the-better
    LARGER = "larger"  # Larger-the-better
    NOMINAL = "nominal"  # Nominal-is-best


@dataclass
class Factor:
    factor_id: str  # "factor1", "factor2", ...
    name: str
    unit: str = ""
    factor_type: str = "numeric"  # "numeric" or "categorical"
    level_low_label: str = ""
    level_low_value: Optional[float] = None
    level_high_label: str = ""
    level_high_value: Optional[float] = None
    baseline: Optional[object] = None
    note: str = ""
    group: Optional[str] = None  # 同一個 group 的因子列共用同一個 table column（例如同一個材料的多個性質）
    table_column: Optional[int] = None  # 明確指定 table column（見 DesignConfig._column_assignments 說明）


@dataclass
class DesignConfig:
    response_name: str
    response_unit: str
    temperatures: List[float]
    sn_direction: SNDirection
    nominal_target: Optional[float]
    factors: List[Factor]  # 順序＋分組決定 table column 指派，見 _column_assignments()

    @property
    def k(self) -> int:
        """因子『列』數（不是欄位數）。同一 group 的多列只算一個欄位，欄位數請用
        `num_columns`；`k` 只用來描述「列出幾列因子」，不能拿來算正交表大小/自由度。
        """

        return len(self.factors)

    def _column_assignments(self) -> Dict[str, int]:
        """因子 -> table column。

        `Factor.table_column` 若有明確指定就直接用（`doe_excel.read_design_and_runs`
        一律會設定：因為使用者可能用「select」欄排除掉某些因子列，剩下的因子在檔案
        裡原本對應的 table column 不能因為前面少了誰就往前遞補——遞補後剩下因子的
        欄位編號會跟它們實際 leg 資料的正交表 pattern 對不上，變成假的「不符」警告
        跟錯誤的 SS。手刻的 DesignConfig（例如測試 fixture）沒有設定 `table_column`
        時，才退回依列出順序＋分組自動依序指派（1,2,3,...）。
        """

        assignments: Dict[str, int] = {}
        group_columns: Dict[str, int] = {}
        next_col = 1
        for f in self.factors:
            if f.table_column is not None:
                assignments[f.factor_id] = f.table_column
                continue
            key = f.group if f.group else f"__solo__{f.factor_id}"
            if key not in group_columns:
                group_columns[key] = next_col
                next_col += 1
            assignments[f.factor_id] = group_columns[key]
        return assignments

    @property
    def num_columns(self) -> int:
        """實際用掉的正交表欄位數：無 group 的因子各佔一欄，同一 group 的因子列
        （例如同一個材料的 Shrinkage/Ex/PRXY）全部共用一欄。這才是決定正交表大小
        （`generate_orthogonal_array` 的 k）與自由度計算該用的數字，不是 `k`。

        用「最大 table column 編號」而不是「不重複欄位數」：用 select 欄排除掉某些
        因子列時，剩下因子的 table column 是固定的（見 `_column_assignments` 說明），
        中間可能有缺口（被排除的因子原本佔的欄位）。用最大編號能確保
        `spare_columns()` 不會把「還在用的較大編號欄位」誤判成備用/誤差欄位（那樣
        會讓那個因子的訊號被重複算進誤差項）；代價是缺口那欄本身既不算因子、也不
        算備用欄位，總變異數分解會少算那一小塊，這是可以接受的已知限制。
        """

        assignments = self._column_assignments()
        return max(assignments.values()) if assignments else 0

    def factor_table_column(self, factor_id: str) -> int:
        assignments = self._column_assignments()
        if factor_id not in assignments:
            raise KeyError(f"未知的因子代號: {factor_id}")
        return assignments[factor_id]

    def factors_by_column(self) -> Dict[int, List[Factor]]:
        """table column -> 指派到這個欄位的因子列（依原本列出順序）。無 group 的
        欄位固定只有 1 個因子；同一 group 的欄位可能有多個（例如材料組的
        Shrinkage/Ex/PRXY 全部指到同一欄）。呼叫端需要把「欄位」換算回「因子」
        時（例如交互作用/別名結構的標籤、混淆因子顯示），一律透過這個函式，不要
        自己用 `design.factors[col - 1]` 這種假設「欄位索引＝列索引＋1」的寫法
        （分組之後這個假設不成立）。
        """

        assignments = self._column_assignments()
        result: Dict[int, List[Factor]] = {}
        for f in self.factors:
            result.setdefault(assignments[f.factor_id], []).append(f)
        return result


@dataclass(frozen=True)
class DesignColumnSpec:
    """一個真正獨立變動的 DOE 欄位。

    一般因子一列就是一欄；同一個材料 ``group`` 的多列性質會共用一欄。統計模型
    必須以欄位為單位建模，否則共用欄位會造成完全共線、重複 SS 與重複貢獻率。
    ``factor_ids``/``names`` 保留所有原始列，讓報告仍可追溯到 Excel。
    """

    column: int
    factor_ids: Tuple[str, ...]
    names: Tuple[str, ...]

    @property
    def label(self) -> str:
        return "/".join(self.factor_ids)

    @property
    def display_name(self) -> str:
        return "/".join(self.names)


def design_column_specs(design: DesignConfig) -> List[DesignColumnSpec]:
    """依 table column 排序，回傳每個實際 DOE 變數一次。"""

    specs: List[DesignColumnSpec] = []
    for col, factors in sorted(design.factors_by_column().items()):
        specs.append(
            DesignColumnSpec(
                column=col,
                factor_ids=tuple(f.factor_id for f in factors),
                names=tuple(f.name for f in factors),
            )
        )
    return specs

@dataclass
class RunMatrix:
    """從 DOE 工作表讀回、解碼成 coded ±1 的因子矩陣，以及每個溫度的回應值。"""

    run_labels: List[str]  # leg1..legN，依序，順序須與 OA 的 canonical row order 一致
    factor_coded: Dict[str, List[Optional[int]]]  # factor_id -> 長度 N 的 -1/+1（None=無法解碼）
    responses: Dict[float, List[Optional[float]]]  # 溫度 -> 長度 N 的量測值（None=未填）
    baseline_factor_values: Dict[str, object] = field(default_factory=dict)
    baseline_responses: Dict[float, Optional[float]] = field(default_factory=dict)


def decode_level(raw_value: object, factor: Factor) -> Optional[int]:
    """把 DOE 表中填的原始值（標籤或數值）解碼成 -1 (Low) / +1 (High)；無法判斷回傳 None。"""

    if raw_value is None or (isinstance(raw_value, str) and not raw_value.strip()):
        return None
    raw_str = str(raw_value).strip()
    if factor.level_low_label and raw_str == str(factor.level_low_label).strip():
        return -1
    if factor.level_high_label and raw_str == str(factor.level_high_label).strip():
        return 1
    try:
        raw_num = float(raw_value)
    except (TypeError, ValueError):
        return None
    if factor.level_low_value is not None and math.isclose(raw_num, float(factor.level_low_value), rel_tol=1e-9, abs_tol=1e-9):
        return -1
    if factor.level_high_value is not None and math.isclose(raw_num, float(factor.level_high_value), rel_tol=1e-9, abs_tol=1e-9):
        return 1
    return None


_PLACEHOLDER_LEVEL_TOKENS = {"low", "high"}


def _dedup_key(value: object):
    """供 infer_factor_levels 判斷「是不是同一個值」用：數值正規化比較，其餘轉字串比較。"""

    try:
        return round(float(value), 12)
    except (TypeError, ValueError):
        return str(value).strip()


def infer_factor_levels(raw_leg_values: List[object]) -> Tuple[object, object, bool]:
    """從一列因子實際填入的 leg 值，直接反推 Low/High 兩個水準是什麼。

    依「第一次出現的不重複值＝Low、第二次出現的不重複值＝High」判斷（跟
    decode_level() 比對時的方向定義一致，只是水準值的來源從『另外宣告』改成
    『直接看這一列填了什麼』）。回傳 (low_value, high_value, is_numeric)；
    is_numeric 為 True 時代表兩個值都能轉成數字（用來決定要當數值因子還是
    類別因子）。若不重複值不是剛好 2 個，丟出 ValueError 說明目前偵測到幾個值。
    """

    seen: Dict[object, object] = {}
    order: List[object] = []
    for v in raw_leg_values:
        if v is None or (isinstance(v, str) and not v.strip()):
            continue
        key = _dedup_key(v)
        if key not in seen:
            seen[key] = v
            order.append(key)

    if len(order) != 2:
        distinct_display = ", ".join(repr(seen[k]) for k in order)
        raise ValueError(
            f"這個因子目前偵測到 {len(order)} 個不同的值（{distinct_display}），"
            "需要剛好 2 個才能判斷 Low/High，請確認每個 leg 都已填入正確的水準值。"
        )

    low, high = seen[order[0]], seen[order[1]]

    def _is_num(x: object) -> bool:
        try:
            float(x)
            return True
        except (TypeError, ValueError):
            return False

    return low, high, _is_num(low) and _is_num(high)


def validate_filled_doe(design: DesignConfig, oa: OrthogonalArray, run_matrix: RunMatrix) -> List[str]:
    """檢查讀回的因子矩陣是否與預期正交表 pattern 完全吻合。"""

    warnings: List[str] = []
    for f in design.factors:
        actual = run_matrix.factor_coded.get(f.factor_id)
        if actual is None:
            warnings.append(f"因子 {f.factor_id}（{f.name}）在 DOE 表中找不到資料列。")
            continue
        if any(v is None for v in actual):
            missing = [run_matrix.run_labels[i] for i, v in enumerate(actual) if v is None]
            warnings.append(
                f"因子 {f.factor_id}（{f.name}）有值無法解碼為 Low/High（{', '.join(missing)}），"
                "請確認每個 leg 是否都已填入這個因子的兩個水準值之一。"
            )
            continue
        expected = oa.columns[design.factor_table_column(f.factor_id)]
        # Low/High 哪個對應 -1、哪個對應 +1 是任意的（依 leg 值第一次出現的順序決定，
        # 不是像舊架構那樣由 Factors 表宣告、寫入時強制對齊 oa.columns 的正負號），
        # 所以整欄正負號整體反過來（expected 的負版本）一樣是合法的正交表 pattern，
        # 只有兩種都不符時才是真的排列錯誤。
        expected_negated = tuple(-v for v in expected)
        if tuple(actual) not in (expected, expected_negated):
            warnings.append(
                f"因子 {f.factor_id}（{f.name}）的水準排列與預期正交表 pattern 不符，"
                "leg 值可能填錯位置或被手動修改，結果可能不可靠，建議對照建立範本時的 "
                "Low/High 佔位樣板重新檢查。"
            )
        low_str = str(f.level_low_label or f.level_low_value)
        high_str = str(f.level_high_label or f.level_high_value)
        if low_str.strip().lower() in _PLACEHOLDER_LEVEL_TOKENS or high_str.strip().lower() in _PLACEHOLDER_LEVEL_TOKENS:
            warnings.append(
                f"因子 {f.factor_id}（{f.name}）的水準看起來還是範本的 Low/High 佔位文字，"
                "尚未換成實際數值，分析結果不具意義。"
            )
    return warnings


def _norm_temp(t) -> float:
    return float(t)


def _response_vector(run_matrix: RunMatrix, temperature) -> List[float]:
    key = _norm_temp(temperature)
    vals = run_matrix.responses.get(key)
    if vals is None:
        available = ", ".join(str(t) for t in sorted(run_matrix.responses))
        raise ValueError(f"找不到溫度 {temperature} 的量測資料。可用溫度：{available}")
    missing = [run_matrix.run_labels[i] for i, v in enumerate(vals) if v is None]
    if missing:
        raise ValueError(f"溫度 {temperature} 的量測資料尚未填完整，缺少：{', '.join(missing)}")
    return [float(v) for v in vals]

@dataclass
class MainEffectRow:
    factor_id: str
    name: str
    mean_low: float
    mean_high: float
    effect: float


def main_effects(design: DesignConfig, run_matrix: RunMatrix, temperature) -> List[MainEffectRow]:
    y = _response_vector(run_matrix, temperature)
    rows: List[MainEffectRow] = []
    for f in design.factors:
        coded = run_matrix.factor_coded[f.factor_id]
        low = [yv for yv, c in zip(y, coded) if c == -1]
        high = [yv for yv, c in zip(y, coded) if c == 1]
        mean_low = sum(low) / len(low)
        mean_high = sum(high) / len(high)
        rows.append(MainEffectRow(f.factor_id, f.name, mean_low, mean_high, mean_high - mean_low))
    return rows


def main_effects_by_design_column(
    design: DesignConfig, rows: List[MainEffectRow]
) -> List[MainEffectRow]:
    """把相容舊 API 的逐因子列結果，收斂成每個獨立設計欄位一列。

    同一材料 group 的性質共用 coded vector，因此 response 的 Low/High 平均與 effect
    必然相同；統計摘要若重複列出會讓使用者誤以為 Shrinkage、Ex 等性質是各自獨立
    操控的因子。原始 ``main_effects()`` 仍保留逐列結果供追溯，報表／排名使用本函式。
    """

    by_id = {row.factor_id: row for row in rows}
    grouped: List[MainEffectRow] = []
    for spec in design_column_specs(design):
        source = by_id[spec.factor_ids[0]]
        grouped.append(
            MainEffectRow(
                factor_id=spec.label,
                name=spec.display_name,
                mean_low=source.mean_low,
                mean_high=source.mean_high,
                effect=source.effect,
            )
        )
    return grouped


# ---------------------------------------------------------------------------
# 2. ANOVA（變異數分析）
# ---------------------------------------------------------------------------


@dataclass
class AnovaRow:
    factor_id: str
    name: str
    ss: float
    df: int
    ms: Optional[float]
    f_value: Optional[float]
    p_value: Optional[float]
    pct_contribution: float
    table_column: Optional[int] = None
    member_factor_ids: Tuple[str, ...] = ()


@dataclass
class AnovaResult:
    rows: List[AnovaRow]
    ss_error: float
    df_error: int
    ms_error: Optional[float]
    ss_total: float
    df_total: int
    warnings: List[str] = field(default_factory=list)
    design_rows: List[AnovaRow] = field(default_factory=list)
    ss_model: float = 0.0


def anova(design: DesignConfig, oa: OrthogonalArray, run_matrix: RunMatrix, temperature) -> AnovaResult:
    y = _response_vector(run_matrix, temperature)
    N = oa.N
    y_bar = sum(y) / N
    ss_total = sum((v - y_bar) ** 2 for v in y)
    df_total = N - 1

    # 以真正獨立的 table column 計算一次。材料 group 的多個性質列共用同一 coded
    # 欄位，若逐列相加會讓 SS/%Contribution 被重複計入。
    raw_by_column: Dict[int, float] = {}
    specs = design_column_specs(design)
    for spec in specs:
        representative_id = spec.factor_ids[0]
        coded = run_matrix.factor_coded[representative_id]
        s = sum(v * c for v, c in zip(y, coded))
        ss = (s ** 2) / N
        raw_by_column[spec.column] = ss

    spare = spare_columns(design.num_columns, oa)
    ss_error = 0.0
    for col in spare:
        colvec = oa.columns[col]
        s = sum(v * c for v, c in zip(y, colvec))
        ss_error += (s ** 2) / N
    df_error = len(spare)
    ms_error = ss_error / df_error if df_error > 0 else None

    warnings: List[str] = []
    if df_error == 0:
        warnings.append("此設計為飽和設計（誤差自由度=0），無法做 F 檢定，%貢獻度僅供排序參考，不代表統計顯著性。")
    elif df_error <= 2:
        warnings.append(f"誤差自由度僅 {df_error}，檢定力低，F/p 值僅供參考。")
    if df_error > 0 and ms_error == 0:
        warnings.append("誤差均方為 0（spare columns 對應之離均差平方和為 0），F 檢定無法計算。")

    design_rows: List[AnovaRow] = []
    for spec in specs:
        ss = raw_by_column[spec.column]
        ms = ss / 1
        if ms_error and ms_error > 0:
            fval = ms / ms_error
            pval = float(scipy_stats.f.sf(fval, 1, df_error))
        else:
            fval = None
            pval = None
        pct = ss / ss_total * 100 if ss_total > 0 else 0.0
        design_rows.append(
            AnovaRow(
                spec.label, spec.display_name, ss, 1, ms, fval, pval, pct,
                table_column=spec.column, member_factor_ids=spec.factor_ids,
            )
        )

    # 相容既有核心 API：rows 仍可依原始 factor_id 查詢；同組列共享同一份結果。
    # 報告與總變異彙整則使用 design_rows，確保同一 group 只顯示/加總一次。
    design_row_by_col = {row.table_column: row for row in design_rows}
    rows: List[AnovaRow] = []
    for f in design.factors:
        source = design_row_by_col[design.factor_table_column(f.factor_id)]
        rows.append(
            AnovaRow(
                f.factor_id, f.name, source.ss, source.df, source.ms, source.f_value,
                source.p_value, source.pct_contribution, table_column=source.table_column,
                member_factor_ids=source.member_factor_ids,
            )
        )

    return AnovaResult(
        rows=rows,
        ss_error=ss_error,
        df_error=df_error,
        ms_error=ms_error,
        ss_total=ss_total,
        df_total=df_total,
        warnings=warnings,
        design_rows=design_rows,
        ss_model=sum(raw_by_column.values()),
    )


# ---------------------------------------------------------------------------
# 3. Regression（迴歸分析）
# ---------------------------------------------------------------------------


@dataclass
class RegressionTerm:
    label: str
    coefficient: float
    kind: str  # "intercept" | "main" | "interaction"
    source_columns: Tuple[int, ...] = ()
    std_error: Optional[float] = None
    t_value: Optional[float] = None
    p_value: Optional[float] = None
    ci_low: Optional[float] = None
    ci_high: Optional[float] = None


@dataclass
class RegressionResult:
    terms: List[RegressionTerm]
    r_squared: float
    residual_df: int
    excluded_terms: List[Tuple[str, str]]  # (label, reason)
    adjusted_r_squared: Optional[float] = None
    rmse: Optional[float] = None
    press: Optional[float] = None
    predicted_r_squared: Optional[float] = None
    observed_values: List[float] = field(default_factory=list)
    fitted_values: List[float] = field(default_factory=list)
    residuals: List[float] = field(default_factory=list)
    leverage: List[Optional[float]] = field(default_factory=list)
    cooks_distance: List[Optional[float]] = field(default_factory=list)
    covariance_matrix: Optional[List[List[float]]] = None
    model_rank: int = 0
    warnings: List[str] = field(default_factory=list)


def regression(
    design: DesignConfig,
    oa: OrthogonalArray,
    run_matrix: RunMatrix,
    temperature,
    include_interactions: bool = False,
) -> RegressionResult:
    y = _response_vector(run_matrix, temperature)
    N = oa.N
    num_cols = design.num_columns

    X_cols: List[List[float]] = [[1.0] * N]
    labels: List[str] = ["Intercept"]
    kinds: List[str] = ["intercept"]
    source_columns: List[Tuple[int, ...]] = [()]
    specs = design_column_specs(design)
    factors_by_col = design.factors_by_column()
    for spec in specs:
        X_cols.append([float(v) for v in run_matrix.factor_coded[spec.factor_ids[0]]])
        labels.append(spec.label)
        kinds.append("main")
        source_columns.append((spec.column,))

    excluded: List[Tuple[str, str]] = []
    if include_interactions:
        # detect_alias_structure 是對「欄位」兩兩配對（同一 group 共用的欄位在這裡
        # 只會出現一次，所以同一 group 內的因子彼此不會被配成交互作用——這正是需要
        # 的「跳過同組配對」效果，不用再另外判斷）。
        aliases = detect_alias_structure(num_cols, oa)
        candidates: List[Tuple[str, int, int, int]] = []
        seen_columns: Dict[int, str] = {}
        for info in aliases:
            # 用 select 欄排除因子後，該因子原本佔的欄位會變成沒有因子的「洞」，
            # `factors_by_col` 不會有這個欄位的紀錄——info.factor_a/factor_b 落在
            # 洞裡代表這組配對本身就不存在（其中一邊沒有因子），直接跳過。
            fa_list = factors_by_col.get(info.factor_a)
            fb_list = factors_by_col.get(info.factor_b)
            if fa_list is None or fb_list is None:
                continue
            fa = fa_list[0]  # 代表因子（該欄位第一個因子列）
            fb = fb_list[0]
            label = f"{fa.factor_id}*{fb.factor_id}"
            if info.confounded_with is not None:
                conf_factors = factors_by_col.get(info.confounded_with)
                if conf_factors is not None:
                    conf_label = "/".join(f.factor_id for f in conf_factors)
                    excluded.append((label, f"與 {conf_label}（{conf_factors[0].name}）主效應混淆，結構性不可估"))
                    continue
                # confounded_with 指向的欄位其實是洞（原本佔用的因子被排除），這組
                # 交互作用對目前納入分析的因子來說其實可以估計，繼續往下走，不排除。
            col = info.interaction_column
            if col in seen_columns:
                # 多個交互作用別名到同一個備用欄位（同一個自由度），彼此無法同時估計，
                # 只保留依 aliases 列舉順序第一個遇到的候選項。
                excluded.append((label, f"與 {seen_columns[col]} 別名相同（同一備用欄位 column{col}），無法同時估計"))
                continue
            seen_columns[col] = label
            candidates.append((label, col, info.factor_a, info.factor_b))
        candidates.sort(key=lambda t: t[1])  # table column 索引小的優先保留

        max_addable = N - 1 - len(specs)
        kept = candidates[:max_addable] if max_addable >= 0 else []
        dropped = candidates[len(kept):]
        for label, _col, _a, _b in dropped:
            excluded.append((label, "自由度不足，已依優先權排除"))

        for label, _col, col_a, col_b in kept:
            fa_id = factors_by_col[col_a][0].factor_id
            fb_id = factors_by_col[col_b][0].factor_id
            # 使用實際讀回的 coded 欄相乘，不直接拿 canonical OA 欄；如此即使 Excel
            # 的 Low/High 極性與產生器相反，交互作用係數方向仍正確。
            inter_vec = [
                float(a * b)
                for a, b in zip(run_matrix.factor_coded[fa_id], run_matrix.factor_coded[fb_id])
            ]
            X_cols.append(inter_vec)
            labels.append(label)
            kinds.append("interaction")
            source_columns.append((col_a, col_b))

    X = np.array(X_cols, dtype=float).T  # N x num_terms
    y_arr = np.array(y, dtype=float)
    coef, _residuals, rank, _sv = np.linalg.lstsq(X, y_arr, rcond=None)
    y_hat = X @ coef
    ss_res = float(np.sum((y_arr - y_hat) ** 2))
    y_mean = float(np.mean(y_arr))
    ss_tot = float(np.sum((y_arr - y_mean) ** 2))
    r_squared = 1.0 if ss_tot < 1e-12 else 1 - ss_res / ss_tot
    residual_df = N - int(rank)

    warnings: List[str] = []
    if residual_df == 0:
        warnings.append("無殘差自由度，R²/模型品質無法評估，係數仍可視為效應估計。")
    if int(rank) < X.shape[1]:
        warnings.append(
            f"設計矩陣秩不足（rank={int(rank)}，terms={X.shape[1]}），部分係數不具唯一解；"
            "請檢查因子分組、別名或手動修改的水準排列。"
        )
    if any("別名相同" in reason for _label, reason in excluded):
        warnings.append(
            "迴歸中保留的交互作用是該 interaction column 的代表項；若同欄另有被排除的"
            "交互作用，其係數代表整個 alias class，不能歸因於單獨一組因子。"
        )

    mse = ss_res / residual_df if residual_df > 0 else None
    rmse = math.sqrt(mse) if mse is not None else None
    adjusted_r_squared = None
    if residual_df > 0 and N > 1:
        adjusted_r_squared = 1.0 - (1.0 - r_squared) * (N - 1) / residual_df

    xtx_inv = np.linalg.pinv(X.T @ X)
    full_column_rank = int(rank) == X.shape[1]
    # rank deficient 時係數不是唯一解；即使 pinv 能產出一個數值矩陣，也不能把它
    # 當成有效的係數 covariance 進一步算 t/p/CI。
    covariance = xtx_inv * mse if mse is not None and full_column_rank else None
    std_errors: List[Optional[float]] = [None] * len(coef)
    t_values: List[Optional[float]] = [None] * len(coef)
    p_values: List[Optional[float]] = [None] * len(coef)
    ci_lows: List[Optional[float]] = [None] * len(coef)
    ci_highs: List[Optional[float]] = [None] * len(coef)
    if covariance is not None and mse is not None and mse > 0:
        t_critical = float(scipy_stats.t.ppf(0.975, residual_df))
        for i, c in enumerate(coef):
            se = math.sqrt(max(0.0, float(covariance[i, i])))
            std_errors[i] = se
            if se > 0:
                t_value = float(c / se)
                t_values[i] = t_value
                p_values[i] = float(2.0 * scipy_stats.t.sf(abs(t_value), residual_df))
                ci_lows[i] = float(c - t_critical * se)
                ci_highs[i] = float(c + t_critical * se)
    elif residual_df > 0 and mse == 0:
        warnings.append("殘差均方為 0，無法用 t/p/信賴區間判斷係數不確定性。")

    hat = X @ xtx_inv @ X.T
    leverage = [float(v) for v in np.diag(hat)]
    press = None
    predicted_r_squared = None
    cooks: List[Optional[float]] = [None] * N
    if all(abs(1.0 - h) > 1e-12 for h in leverage):
        loo_residuals = [(float(e) / (1.0 - h)) for e, h in zip(y_arr - y_hat, leverage)]
        press = float(sum(v * v for v in loo_residuals))
        if ss_tot > 1e-12:
            predicted_r_squared = 1.0 - press / ss_tot
    if mse is not None and mse > 0 and int(rank) > 0:
        for i, (e, h) in enumerate(zip(y_arr - y_hat, leverage)):
            if abs(1.0 - h) > 1e-12:
                cooks[i] = float((e * e / (int(rank) * mse)) * h / ((1.0 - h) ** 2))

    terms = [
        RegressionTerm(
            label=lab,
            coefficient=float(c),
            kind=kind,
            source_columns=src,
            std_error=std_errors[i],
            t_value=t_values[i],
            p_value=p_values[i],
            ci_low=ci_lows[i],
            ci_high=ci_highs[i],
        )
        for i, (lab, c, kind, src) in enumerate(zip(labels, coef, kinds, source_columns))
    ]

    return RegressionResult(
        terms=terms,
        r_squared=r_squared,
        residual_df=residual_df,
        excluded_terms=excluded,
        adjusted_r_squared=adjusted_r_squared,
        rmse=rmse,
        press=press,
        predicted_r_squared=predicted_r_squared,
        observed_values=[float(v) for v in y_arr],
        fitted_values=[float(v) for v in y_hat],
        residuals=[float(v) for v in y_arr - y_hat],
        leverage=leverage,
        cooks_distance=cooks,
        covariance_matrix=covariance.tolist() if covariance is not None else None,
        model_rank=int(rank),
        warnings=warnings,
    )


def regression_physical_coefficients(design: DesignConfig, result: RegressionResult) -> Dict[str, float]:
    """把 coded ±1 迴歸係數換算回物理量尺度。

    只處理單獨佔一個設計欄位的 numeric 因子。材料 group 的多個性質一起變動，無法
    從同一個係數分離出各性質的 per-unit 斜率，因此刻意不輸出，避免製造假精度。
    """

    physical: Dict[str, float] = {}
    main_by_column = {t.source_columns[0]: t for t in result.terms if t.kind == "main" and t.source_columns}
    for spec in design_column_specs(design):
        if len(spec.factor_ids) != 1:
            continue
        f = next(item for item in design.factors if item.factor_id == spec.factor_ids[0])
        t = main_by_column.get(spec.column)
        if t and f.level_low_value is not None and f.level_high_value is not None:
            half_range = (float(f.level_high_value) - float(f.level_low_value)) / 2.0
            if half_range != 0:
                physical[f.factor_id] = t.coefficient / half_range
    return physical


# ---------------------------------------------------------------------------
# 3b. Effect Screening（標準化效應／Lenth PSE）
# ---------------------------------------------------------------------------


@dataclass
class EffectScreeningRow:
    label: str
    name: str
    kind: str
    effect: float
    standard_error: Optional[float]
    statistic: Optional[float]
    p_value: Optional[float]
    ci_low: Optional[float]
    ci_high: Optional[float]
    active: Optional[bool]
    method: str


@dataclass
class EffectScreeningResult:
    rows: List[EffectScreeningRow]
    method: str
    lenth_pse: Optional[float] = None
    critical_value: Optional[float] = None
    warnings: List[str] = field(default_factory=list)


def effect_screening(design: DesignConfig, regression_result: RegressionResult) -> EffectScreeningResult:
    """把 coded regression coefficient 轉成 effect，並提供顯著性篩選。

    有可用殘差誤差時沿用迴歸 t 檢定；無殘差自由度／無可用 MSE 時採 Lenth PSE，
    這是無重複二水準設計常用的 pseudo standard error。Lenth 判定是篩選工具，
    不是實驗重複量測的替代品，因此報告會明確標示方法。
    """

    non_intercept = [t for t in regression_result.terms if t.kind != "intercept"]
    if not non_intercept:
        return EffectScreeningResult(rows=[], method="N/A", warnings=["模型沒有可供篩選的效應項。"])

    spec_by_col = {s.column: s for s in design_column_specs(design)}

    def _term_name(term: RegressionTerm) -> str:
        if term.kind == "main" and term.source_columns:
            spec = spec_by_col.get(term.source_columns[0])
            return spec.display_name if spec else term.label
        if term.kind == "interaction" and len(term.source_columns) == 2:
            a = spec_by_col.get(term.source_columns[0])
            b = spec_by_col.get(term.source_columns[1])
            return f"{a.display_name if a else term.source_columns[0]} × {b.display_name if b else term.source_columns[1]}"
        return term.label

    if regression_result.model_rank < len(regression_result.terms):
        rows = [
            EffectScreeningRow(
                label=term.label,
                name=_term_name(term),
                kind=term.kind,
                effect=2.0 * term.coefficient,
                standard_error=None,
                statistic=None,
                p_value=None,
                ci_low=None,
                ci_high=None,
                active=None,
                method="N/A — rank deficient",
            )
            for term in non_intercept
        ]
        rows.sort(key=lambda row: -abs(row.effect))
        return EffectScreeningResult(
            rows=rows,
            method="N/A — rank deficient",
            warnings=["設計矩陣秩不足，係數不具唯一解，因此不執行 t 或 Lenth PSE 效應判定。"],
        )

    has_classical_error = (
        regression_result.residual_df > 0
        and any(t.std_error is not None for t in non_intercept)
    )
    rows: List[EffectScreeningRow] = []
    if has_classical_error:
        for term in non_intercept:
            effect = 2.0 * term.coefficient
            rows.append(
                EffectScreeningRow(
                    label=term.label,
                    name=_term_name(term),
                    kind=term.kind,
                    effect=effect,
                    standard_error=2.0 * term.std_error if term.std_error is not None else None,
                    statistic=term.t_value,
                    p_value=term.p_value,
                    ci_low=2.0 * term.ci_low if term.ci_low is not None else None,
                    ci_high=2.0 * term.ci_high if term.ci_high is not None else None,
                    active=(term.p_value < 0.05) if term.p_value is not None else None,
                    method="Regression t",
                )
            )
        rows.sort(key=lambda r: -abs(r.effect))
        critical = float(scipy_stats.t.ppf(0.975, regression_result.residual_df))
        return EffectScreeningResult(rows=rows, method="Regression t", critical_value=critical)

    effects = [2.0 * t.coefficient for t in non_intercept]
    absolute = [abs(v) for v in effects]
    s0 = 1.5 * float(np.median(absolute))
    candidates = [v for v in absolute if s0 > 0 and v <= 2.5 * s0]
    if not candidates:
        candidates = [v for v in absolute if v > 1e-15]
    pse = 1.5 * float(np.median(candidates)) if candidates else None
    pseudo_df = max(1, len(regression_result.observed_values) // 3)
    critical = float(scipy_stats.t.ppf(0.975, pseudo_df)) if pse and pse > 0 else None
    for term, effect in zip(non_intercept, effects):
        statistic = effect / pse if pse and pse > 0 else None
        p_value = (
            float(2.0 * scipy_stats.t.sf(abs(statistic), pseudo_df))
            if statistic is not None else None
        )
        margin = critical * pse if critical is not None and pse is not None else None
        rows.append(
            EffectScreeningRow(
                label=term.label,
                name=_term_name(term),
                kind=term.kind,
                effect=effect,
                standard_error=pse,
                statistic=statistic,
                p_value=p_value,
                ci_low=effect - margin if margin is not None else None,
                ci_high=effect + margin if margin is not None else None,
                active=(abs(effect) > margin) if margin is not None else None,
                method="Lenth PSE",
            )
        )
    rows.sort(key=lambda r: -abs(r.effect))
    warnings = [
        "沒有可用的殘差誤差，效應顯著性改用 Lenth PSE 篩選；pseudo p 值僅供排序，"
        "不能取代重複實驗估計的純誤差。"
    ]
    if pse is None:
        warnings.append("所有效應皆接近 0，Lenth PSE 無法計算。")
    return EffectScreeningResult(
        rows=rows, method="Lenth PSE", lenth_pse=pse, critical_value=critical, warnings=warnings
    )


# ---------------------------------------------------------------------------
# 3c. Alias／交互作用量化
# ---------------------------------------------------------------------------


@dataclass
class AliasInteractionRow:
    interaction: str
    factor_a: str
    factor_b: str
    interaction_column: int
    status: str
    aliased_with: str
    effect: float
    ss: float
    error_df: int
    error_ms: Optional[float]
    f_value: Optional[float]
    p_value: Optional[float]


@dataclass
class AliasInteractionResult:
    rows: List[AliasInteractionRow]
    warnings: List[str] = field(default_factory=list)


def alias_interaction_analysis(
    design: DesignConfig,
    oa: OrthogonalArray,
    run_matrix: RunMatrix,
    temperature,
    anova_result: AnovaResult,
) -> AliasInteractionResult:
    """列出所有實際設計欄位配對的 alias 狀態與 interaction contrast。"""

    y = _response_vector(run_matrix, temperature)
    factors_by_col = design.factors_by_column()
    available_error_columns = spare_columns(design.num_columns, oa)
    pair_specs: List[Tuple[int, int, str, int]] = []
    labels_by_interaction_column: Dict[int, List[str]] = {}
    for col_a, col_b in itertools.combinations(sorted(factors_by_col), 2):
        fa = factors_by_col[col_a][0]
        fb = factors_by_col[col_b][0]
        label = f"{fa.factor_id}*{fb.factor_id}"
        interaction_col = oa.interaction_column(col_a, col_b)
        pair_specs.append((col_a, col_b, label, interaction_col))
        labels_by_interaction_column.setdefault(interaction_col, []).append(label)

    rows: List[AliasInteractionRow] = []
    for col_a, col_b, label, interaction_col in pair_specs:
        fa = factors_by_col[col_a][0]
        fb = factors_by_col[col_b][0]
        target_factors = factors_by_col.get(interaction_col)
        peer_interactions = [
            item for item in labels_by_interaction_column[interaction_col] if item != label
        ]
        status = "可單獨估計"
        aliased_with = ""
        if target_factors is not None:
            status = "與主效應混淆"
            aliased_with = "/".join(f.factor_id for f in target_factors)
            if peer_interactions:
                aliased_with += "；另與 " + ", ".join(peer_interactions) + " 共用欄位"
        elif peer_interactions:
            # 同一個 interaction column 只提供一個 contrast。不能因為列舉時先遇到
            # 某一對，就把第一對誤稱為「可單獨估計」、只把後面的列成 alias。
            status = "交互作用別名組"
            aliased_with = ", ".join(peer_interactions)

        coded = [
            int(a * b)
            for a, b in zip(run_matrix.factor_coded[fa.factor_id], run_matrix.factor_coded[fb.factor_id])
        ]
        contrast = sum(v * c for v, c in zip(y, coded))
        coefficient = contrast / len(y)
        effect = 2.0 * coefficient
        ss = (contrast ** 2) / len(y)
        # 被檢定的 interaction column 原本若在 ANOVA 的 spare/error pool 中，必須先
        # 從分母移除，否則會拿同一個 SS 同時當分子與誤差，機械式得到 F=1。
        error_columns = [col for col in available_error_columns if col != interaction_col]
        error_ss = 0.0
        for error_col in error_columns:
            error_contrast = sum(v * c for v, c in zip(y, oa.columns[error_col]))
            error_ss += (error_contrast ** 2) / len(y)
        error_df = len(error_columns)
        error_ms = error_ss / error_df if error_df > 0 else None
        if status == "可單獨估計" and error_ms is not None and error_ms > 0:
            f_value = ss / error_ms
            p_value = float(scipy_stats.f.sf(f_value, 1, error_df))
        else:
            f_value = None
            p_value = None
        rows.append(
            AliasInteractionRow(
                interaction=label,
                factor_a=fa.factor_id,
                factor_b=fb.factor_id,
                interaction_column=interaction_col,
                status=status,
                aliased_with=aliased_with,
                effect=effect,
                ss=ss,
                error_df=error_df,
                error_ms=error_ms,
                f_value=f_value,
                p_value=p_value,
            )
        )

    warnings: List[str] = []
    if any(r.status != "可單獨估計" for r in rows):
        warnings.append(
            "標示為主效應混淆或交互作用別名組的項目只有同一欄位的合併 contrast，"
            "不能解讀為單獨交互作用效應。"
        )
    return AliasInteractionResult(rows=rows, warnings=warnings)


# ---------------------------------------------------------------------------
# 3d. 模型預測最佳組合／確認 run
# ---------------------------------------------------------------------------


@dataclass
class CombinationPredictionRow:
    rank: int
    coded_levels: Dict[int, int]
    predicted: float
    observed_run_labels: List[str]
    mean_ci_low: Optional[float] = None
    mean_ci_high: Optional[float] = None
    prediction_low: Optional[float] = None
    prediction_high: Optional[float] = None


@dataclass
class CombinationPredictionResult:
    rows: List[CombinationPredictionRow]
    column_specs: List[DesignColumnSpec]
    confirmation_row: Optional[CombinationPredictionRow]
    skipped: bool = False
    skip_reason: str = ""
    warnings: List[str] = field(default_factory=list)


def _model_vector_for_levels(terms: List[RegressionTerm], levels: Dict[int, int]) -> np.ndarray:
    values: List[float] = []
    for term in terms:
        if term.kind == "intercept":
            values.append(1.0)
        elif term.kind == "main":
            values.append(float(levels[term.source_columns[0]]))
        else:
            product = 1.0
            for col in term.source_columns:
                product *= levels[col]
            values.append(product)
    return np.array(values, dtype=float)


def predict_combinations(
    design: DesignConfig,
    run_matrix: RunMatrix,
    regression_result: RegressionResult,
    max_columns: int = 12,
) -> CombinationPredictionResult:
    """枚舉模型涵蓋的 Low/High 組合，排序並指出第一個未實跑的 confirmation run。"""

    specs = design_column_specs(design)
    if len(specs) > max_columns:
        return CombinationPredictionResult(
            rows=[], column_specs=specs, confirmation_row=None, skipped=True,
            skip_reason=f"獨立設計變數有 {len(specs)} 個，完整枚舉超過安全上限 {max_columns} 個。",
        )
    if regression_result.model_rank < len(regression_result.terms):
        return CombinationPredictionResult(
            rows=[], column_specs=specs, confirmation_row=None, skipped=True,
            skip_reason="迴歸模型秩不足，係數不具唯一解，不適合列出最佳預測組合。",
        )

    observed: Dict[Tuple[int, ...], List[str]] = {}
    for run_idx, run_label in enumerate(run_matrix.run_labels):
        key = tuple(int(run_matrix.factor_coded[s.factor_ids[0]][run_idx]) for s in specs)
        observed.setdefault(key, []).append(run_label)

    coefficients = np.array([t.coefficient for t in regression_result.terms], dtype=float)
    covariance = (
        np.array(regression_result.covariance_matrix, dtype=float)
        if regression_result.covariance_matrix is not None else None
    )
    t_critical = (
        float(scipy_stats.t.ppf(0.975, regression_result.residual_df))
        if regression_result.residual_df > 0 else None
    )
    rows: List[CombinationPredictionRow] = []
    columns = [s.column for s in specs]
    for combo in itertools.product((-1, 1), repeat=len(columns)):
        levels = dict(zip(columns, combo))
        model_vector = _model_vector_for_levels(regression_result.terms, levels)
        predicted = float(model_vector @ coefficients)
        mean_low = mean_high = pred_low = pred_high = None
        if covariance is not None and t_critical is not None:
            mean_var = max(0.0, float(model_vector @ covariance @ model_vector.T))
            mean_se = math.sqrt(mean_var)
            mean_low = predicted - t_critical * mean_se
            mean_high = predicted + t_critical * mean_se
            if regression_result.rmse is not None:
                pred_se = math.sqrt(mean_var + regression_result.rmse ** 2)
                pred_low = predicted - t_critical * pred_se
                pred_high = predicted + t_critical * pred_se
        key = tuple(combo)
        rows.append(
            CombinationPredictionRow(
                rank=0,
                coded_levels=levels,
                predicted=predicted,
                observed_run_labels=list(observed.get(key, [])),
                mean_ci_low=mean_low,
                mean_ci_high=mean_high,
                prediction_low=pred_low,
                prediction_high=pred_high,
            )
        )

    if design.sn_direction == SNDirection.LARGER:
        rows.sort(key=lambda r: -r.predicted)
    elif design.sn_direction == SNDirection.NOMINAL and design.nominal_target is not None:
        rows.sort(key=lambda r: abs(r.predicted - design.nominal_target))
    else:
        rows.sort(key=lambda r: r.predicted)
    for i, row in enumerate(rows, start=1):
        row.rank = i
    confirmation = next((row for row in rows if not row.observed_run_labels), None)
    warnings: List[str] = []
    if regression_result.residual_df == 0:
        warnings.append("模型沒有殘差自由度；組合排名可供篩選，但無法提供可信的預測區間。")
    if confirmation is None:
        warnings.append("所有可列舉組合都已存在於目前 runs，沒有額外的未實跑確認組合。")
    return CombinationPredictionResult(
        rows=rows, column_specs=specs, confirmation_row=confirmation, warnings=warnings
    )


# ---------------------------------------------------------------------------
# 4. Correlation（相關性）
# ---------------------------------------------------------------------------


@dataclass
class CorrelationResult:
    labels: List[str]  # factor_ids + response label
    matrix: List[List[float]]
    flagged_pairs: List[Tuple[str, str, float]]
    warnings: List[str] = field(default_factory=list)


def correlation_matrix(
    design: DesignConfig, run_matrix: RunMatrix, temperature, threshold: float = 0.3
) -> CorrelationResult:
    y = _response_vector(run_matrix, temperature)
    N = len(y)
    labels = [f.factor_id for f in design.factors] + [design.response_name or "Response"]
    data = [run_matrix.factor_coded[f.factor_id] for f in design.factors] + [y]
    arr = np.array(data, dtype=float)
    with np.errstate(invalid="ignore"):
        corr = np.corrcoef(arr)

    warnings: List[str] = []
    if N < 10:
        warnings.append(f"樣本數 N={N} 偏低，相關係數僅供排序參考，未做顯著性檢定。")

    flagged: List[Tuple[str, str, float]] = []
    k = design.k
    for i in range(k):
        for j in range(i + 1, k):
            # 同一 group（例如同一個材料的 Shrinkage/Ex/PRXY）本來就共用同一欄位，
            # coded 完全相同、相關係數理論上必為 1，這是設計上的預期結果，不是資料
            # 被改壞，不列入「可能被手動修改」的警告。
            if design.factor_table_column(design.factors[i].factor_id) == design.factor_table_column(
                design.factors[j].factor_id
            ):
                continue
            r = corr[i][j]
            if not math.isnan(r) and abs(r) > threshold:
                flagged.append((labels[i], labels[j], float(r)))
    if flagged:
        warnings.append(
            f"偵測到 {len(flagged)} 組因子間相關係數超過門檻 |r|>{threshold}，"
            "DOE 表的水準排列可能被手動修改而破壞正交性，建議重新產生 DOE 表確認。"
        )

    return CorrelationResult(labels=labels, matrix=corr.tolist(), flagged_pairs=flagged, warnings=warnings)


# ---------------------------------------------------------------------------
# 5. Taguchi Method（S/N 比）
# ---------------------------------------------------------------------------


@dataclass
class TaguchiRow:
    factor_id: str
    name: str
    sn_low: float
    sn_high: float
    delta: float
    rank: int = 0


@dataclass
class TaguchiResult:
    direction: SNDirection
    rows: List[TaguchiRow]
    per_run_sn: List[float]
    warnings: List[str] = field(default_factory=list)
    skipped: bool = False
    skip_reason: Optional[str] = None


def taguchi_sn(design: DesignConfig, run_matrix: RunMatrix, temperature) -> TaguchiResult:
    direction = design.sn_direction
    if direction == SNDirection.NOMINAL:
        return TaguchiResult(
            direction=direction,
            rows=[],
            per_run_sn=[],
            skipped=True,
            skip_reason=(
                "Nominal-is-best 需要每個 run 至少 2 筆重複量測才能算變異數，"
                "目前範本結構是每個 run 僅 1 筆量測，請改選 Smaller-the-better 或 "
                "Larger-the-better，或提供重複量測資料。"
            ),
        )

    y = _response_vector(run_matrix, temperature)
    warnings: List[str] = []
    sn_vals: List[float] = []
    for v in y:
        if v == 0:
            if direction == SNDirection.SMALLER:
                sn_vals.append(float("inf"))
                warnings.append("量測值出現 0，S/N（smaller-the-better）在該點趨近無限大，請確認資料正確性。")
            else:
                sn_vals.append(float("-inf"))
                warnings.append("量測值出現 0，S/N（larger-the-better）在該點趨近負無限大，請確認資料正確性。")
            continue
        if direction == SNDirection.SMALLER:
            sn_vals.append(-20 * math.log10(abs(v)))
        else:
            sn_vals.append(-20 * math.log10(1.0 / abs(v)))

    rows: List[TaguchiRow] = []
    for spec in design_column_specs(design):
        coded = run_matrix.factor_coded[spec.factor_ids[0]]
        low = [s for s, c in zip(sn_vals, coded) if c == -1]
        high = [s for s, c in zip(sn_vals, coded) if c == 1]
        sn_low = sum(low) / len(low)
        sn_high = sum(high) / len(high)
        rows.append(TaguchiRow(spec.label, spec.display_name, sn_low, sn_high, abs(sn_high - sn_low)))
    rows.sort(key=lambda r: -r.delta)
    for i, r in enumerate(rows):
        r.rank = i + 1

    return TaguchiResult(direction=direction, rows=rows, per_run_sn=sn_vals, warnings=warnings)

@dataclass
class ParetoBar:
    factor_id: str
    name: str
    magnitude: float


def pareto_data(main_effect_rows: List[MainEffectRow]) -> List[ParetoBar]:
    bars = [ParetoBar(r.factor_id, r.name, abs(r.effect)) for r in main_effect_rows]
    bars.sort(key=lambda b: -b.magnitude)
    return bars


# ---------------------------------------------------------------------------
# 7. Interaction Plot
# ---------------------------------------------------------------------------


@dataclass
class InteractionPoint:
    level_a: int
    level_b: int
    mean_response: float
    n: int


@dataclass
class InteractionResult:
    factor_a: str
    factor_b: str
    points: List[InteractionPoint]
    confounded_with: Optional[str]
    aliased_with: List[str] = field(default_factory=list)
    warnings: List[str] = field(default_factory=list)


def interaction_data(
    design: DesignConfig,
    oa: OrthogonalArray,
    run_matrix: RunMatrix,
    temperature,
    factor_id_a: str,
    factor_id_b: str,
) -> InteractionResult:
    y = _response_vector(run_matrix, temperature)
    coded_a = run_matrix.factor_coded[factor_id_a]
    coded_b = run_matrix.factor_coded[factor_id_b]
    col_a = design.factor_table_column(factor_id_a)
    col_b = design.factor_table_column(factor_id_b)
    if col_a == col_b:
        raise ValueError(
            f"{factor_id_a} 跟 {factor_id_b} 屬於同一個 group（共用同一個正交表欄位，"
            "例如同一個材料的不同性質），兩者之間沒有獨立的交互作用可以畫，請換一組因子。"
        )
    interaction_col = oa.interaction_column(col_a, col_b)
    factors_by_col = design.factors_by_column()

    confounded_with: Optional[str] = None
    aliased_with: List[str] = []
    warnings: List[str] = []
    # 用 select 欄排除因子後，該因子原本佔的欄位會變成沒有因子的「洞」，
    # `factors_by_col` 不會有這個欄位的紀錄——interaction_col 落在洞裡代表這組
    # 交互作用其實沒有跟任何「目前納入分析」的主效應混淆，不能直接用
    # `interaction_col <= design.num_columns` 判斷（那只代表欄位號碼在使用範圍
    # 內，不代表這個欄位真的有因子）。
    if interaction_col in factors_by_col:
        conf_factors = factors_by_col[interaction_col]
        conf_label = "/".join(f.factor_id for f in conf_factors)
        confounded_with = conf_label
        warnings.append(
            f"{factor_id_a}×{factor_id_b} 交互作用與 {conf_label}（{conf_factors[0].name}）"
            "主效應混淆，效應量僅供趨勢參考，無法單獨量化。"
        )

    target_pair = frozenset((col_a, col_b))
    for peer_a, peer_b in itertools.combinations(sorted(factors_by_col), 2):
        if frozenset((peer_a, peer_b)) == target_pair:
            continue
        if oa.interaction_column(peer_a, peer_b) == interaction_col:
            peer_fa = factors_by_col[peer_a][0].factor_id
            peer_fb = factors_by_col[peer_b][0].factor_id
            aliased_with.append(f"{peer_fa}*{peer_fb}")
    if aliased_with:
        warnings.append(
            f"{factor_id_a}×{factor_id_b} 與 {', '.join(aliased_with)} 共用 interaction column"
            "，圖上的趨勢代表 alias class，不能單獨歸因。"
        )

    points: List[InteractionPoint] = []
    for la in (-1, 1):
        for lb in (-1, 1):
            vals = [v for v, ca, cb in zip(y, coded_a, coded_b) if ca == la and cb == lb]
            if vals:
                points.append(InteractionPoint(la, lb, sum(vals) / len(vals), len(vals)))
            else:
                warnings.append(f"水準組合 ({factor_id_a}={la}, {factor_id_b}={lb}) 無資料，交互作用圖該點缺失。")

    return InteractionResult(
        factor_a=factor_id_a,
        factor_b=factor_id_b,
        points=points,
        confounded_with=confounded_with,
        aliased_with=aliased_with,
        warnings=warnings,
    )


# ---------------------------------------------------------------------------
# 8. Sensitivity Analysis（敏感度綜合排名）
# ---------------------------------------------------------------------------


@dataclass
class SensitivityRow:
    factor_id: str
    name: str
    score: float
    components: Dict[str, float]
    rank: int = 0
    table_column: Optional[int] = None
    member_factor_ids: Tuple[str, ...] = ()


@dataclass
class SensitivityResult:
    rows: List[SensitivityRow]
    warnings: List[str] = field(default_factory=list)


def sensitivity_ranking(
    design: DesignConfig,
    main_effect_rows: List[MainEffectRow],
    anova_result: AnovaResult,
    regression_result: RegressionResult,
    taguchi_result: TaguchiResult,
) -> SensitivityResult:
    specs = design_column_specs(design)
    components: Dict[int, Dict[str, float]] = {spec.column: {} for spec in specs}
    warnings: List[str] = []

    grouped_main_effects = main_effects_by_design_column(design, main_effect_rows)
    main_by_label = {row.factor_id: row for row in grouped_main_effects}
    max_effect = max((abs(r.effect) for r in grouped_main_effects), default=0.0) or 1.0
    for spec in specs:
        components[spec.column]["main_effect"] = abs(main_by_label[spec.label].effect) / max_effect

    for row in anova_result.design_rows:
        if row.table_column in components:
            components[row.table_column]["anova_pct"] = row.pct_contribution / 100.0

    main_coefs_by_col = {
        t.source_columns[0]: t.coefficient
        for t in regression_result.terms
        if t.kind == "main" and t.source_columns
    }
    if main_coefs_by_col:
        max_coef = max((abs(v) for v in main_coefs_by_col.values()), default=0.0) or 1.0
        for spec in specs:
            if spec.column in main_coefs_by_col:
                components[spec.column]["regression_coef"] = abs(main_coefs_by_col[spec.column]) / max_coef
    else:
        warnings.append("迴歸係數無法用於敏感度綜合排名（模型未產生主效應係數）。")

    if not taguchi_result.skipped:
        max_delta = max((r.delta for r in taguchi_result.rows), default=0.0) or 1.0
        taguchi_by_label = {row.factor_id: row for row in taguchi_result.rows}
        for spec in specs:
            row = taguchi_by_label.get(spec.label)
            if row is not None:
                components[spec.column]["taguchi_delta"] = row.delta / max_delta
    else:
        warnings.append(f"Taguchi S/N 未納入敏感度綜合排名：{taguchi_result.skip_reason}")

    rows: List[SensitivityRow] = []
    for spec in specs:
        comp = components[spec.column]
        score = sum(comp.values()) / len(comp) if comp else 0.0
        rows.append(
            SensitivityRow(
                spec.label, spec.display_name, score, dict(comp),
                table_column=spec.column, member_factor_ids=spec.factor_ids,
            )
        )
    rows.sort(key=lambda r: -r.score)
    for i, r in enumerate(rows):
        r.rank = i + 1

    return SensitivityResult(rows=rows, warnings=warnings)


# ---------------------------------------------------------------------------
# 8b. Leg Ranking（依 leg/run 排序）——敏感度綜合排名是「因子」排名，這裡補一份
# 「run」排名：哪個 leg 的量測值最好，方便直接照那個 leg 的水準組合去試產。
# ---------------------------------------------------------------------------


@dataclass
class LegRankRow:
    run_label: str
    response_value: float
    rank: int = 0


def rank_legs_by_response(
    run_matrix: RunMatrix, temperature, direction: SNDirection, nominal_target: Optional[float] = None
) -> List[LegRankRow]:
    """把各 leg（run）在指定溫度的量測值依 S/N 最佳化方向排名，rank 1 是最好的
    那個 leg：Smaller-the-better 排小到大、Larger-the-better 排大到小、
    Nominal-is-best 依離目標值的距離排（沒有目標值時退化成 smaller-the-better，
    跟 `generate_recommendations`/`_preferred_level_is_high` 的退化規則一致）。
    """

    y = _response_vector(run_matrix, temperature)

    if direction == SNDirection.LARGER:
        sort_key = lambda v: -v  # noqa: E731
    elif direction == SNDirection.NOMINAL and nominal_target is not None:
        sort_key = lambda v: abs(v - nominal_target)  # noqa: E731
    else:
        sort_key = lambda v: v  # noqa: E731

    rows = [LegRankRow(run_label=lbl, response_value=v) for lbl, v in zip(run_matrix.run_labels, y)]
    rows.sort(key=lambda r: sort_key(r.response_value))
    for i, r in enumerate(rows):
        r.rank = i + 1
    return rows


# ---------------------------------------------------------------------------
# 整合分析入口
# ---------------------------------------------------------------------------


@dataclass
class FullAnalysisResult:
    temperature: float
    main_effects: List[MainEffectRow]
    anova: AnovaResult
    regression: RegressionResult
    correlation: CorrelationResult
    taguchi: TaguchiResult
    sensitivity: SensitivityResult
    leg_ranking: List[LegRankRow]
    pareto: List[ParetoBar]
    effect_screening: EffectScreeningResult
    alias_interactions: AliasInteractionResult
    combinations: CombinationPredictionResult
    warnings: List[str]


def run_full_analysis(
    design: DesignConfig,
    oa: OrthogonalArray,
    run_matrix: RunMatrix,
    temperature,
    include_interactions: bool = False,
    correlation_threshold: float = 0.3,
) -> FullAnalysisResult:
    """跑完整套兩水準 DOE 分析，並把各方法產生的警告彙整、去重。

    彙整時每則警告會加上來源標籤（例如 `[ANOVA]`），方便對照 Summary 裡的
    「樣本數／自由度／完整性警告」是哪個方法/檢查產生的，不用逐字比對內容猜來源。
    """

    warnings: List[str] = []
    for w in validate_filled_doe(design, oa, run_matrix):
        warnings.append(f"[資料完整性] {w}")

    if oa.N < 4:
        warnings.append(f"[整體設計] run 數過少（N={oa.N}），DOE 分析參考價值有限，建議至少 3 個以上因子（N>=4）。")

    me = main_effects(design, run_matrix, temperature)
    an = anova(design, oa, run_matrix, temperature)
    reg = regression(design, oa, run_matrix, temperature, include_interactions=include_interactions)
    corr = correlation_matrix(design, run_matrix, temperature, threshold=correlation_threshold)
    tag = taguchi_sn(design, run_matrix, temperature)
    sens = sensitivity_ranking(design, me, an, reg, tag)
    leg_rank = rank_legs_by_response(run_matrix, temperature, design.sn_direction, design.nominal_target)
    pareto = pareto_data(main_effects_by_design_column(design, me))
    effect_result = effect_screening(design, reg)
    alias_result = alias_interaction_analysis(design, oa, run_matrix, temperature, an)
    combination_result = predict_combinations(design, run_matrix, reg)

    for label, src in (
        ("ANOVA", an.warnings),
        ("Regression", reg.warnings),
        ("Correlation", corr.warnings),
        ("Taguchi", tag.warnings),
        ("Sensitivity Analysis", sens.warnings),
        ("Effect Screening", effect_result.warnings),
        ("Alias/Interaction", alias_result.warnings),
        ("Optimization", combination_result.warnings),
    ):
        for w in src:
            tagged = f"[{label}] {w}"
            if tagged not in warnings:
                warnings.append(tagged)

    return FullAnalysisResult(
        temperature=_norm_temp(temperature),
        main_effects=me,
        anova=an,
        regression=reg,
        correlation=corr,
        taguchi=tag,
        sensitivity=sens,
        leg_ranking=leg_rank,
        pareto=pareto,
        effect_screening=effect_result,
        alias_interactions=alias_result,
        combinations=combination_result,
        warnings=warnings,
    )


def adequacy_report(result: FullAnalysisResult) -> List[str]:
    """回傳整合分析中彙整好的樣本數／自由度不足警告清單。"""

    return list(result.warnings)


# ---------------------------------------------------------------------------
# 建議（依敏感度排名 + S/N 方向，推薦各顯著因子該選哪個水準）
# ---------------------------------------------------------------------------


@dataclass
class Recommendation:
    factor_id: str
    name: str
    recommended_level: str  # "Low" or "High"
    recommended_label: str
    rationale: str


def _preferred_level_is_high(
    direction: SNDirection, mean_low: float, mean_high: float, nominal_target: Optional[float]
) -> bool:
    if direction == SNDirection.LARGER:
        return mean_high >= mean_low
    if direction == SNDirection.SMALLER:
        return mean_high < mean_low
    # NOMINAL：選平均值離目標值較近的水準；沒有目標值時退化成 smaller-the-better 判斷。
    if nominal_target is None:
        return mean_high < mean_low
    return abs(mean_high - nominal_target) < abs(mean_low - nominal_target)


def generate_recommendations(design: DesignConfig, result: FullAnalysisResult, top_n: int = 3) -> List[Recommendation]:
    """依敏感度排名前 top_n 個獨立設計變數，推薦該選哪個水準。"""

    me_by_id = {r.factor_id: r for r in result.main_effects}
    specs_by_column = {spec.column: spec for spec in design_column_specs(design)}
    factors_by_id = {f.factor_id: f for f in design.factors}
    recs: List[Recommendation] = []
    for srow in result.sensitivity.rows[:top_n]:
        spec = specs_by_column[srow.table_column]
        factors = [factors_by_id[factor_id] for factor_id in spec.factor_ids]
        me = me_by_id[spec.factor_ids[0]]
        prefer_high = _preferred_level_is_high(design.sn_direction, me.mean_low, me.mean_high, design.nominal_target)
        level_code = 1 if prefer_high else -1
        level_name = "High" if prefer_high else "Low"
        observed = me.mean_high if prefer_high else me.mean_low
        if len(factors) == 1:
            f = factors[0]
            if prefer_high:
                level_label = f.level_high_label or (
                    str(f.level_high_value) if f.level_high_value is not None else "High"
                )
            else:
                level_label = f.level_low_label or (
                    str(f.level_low_value) if f.level_low_value is not None else "Low"
                )
        else:
            level_label = _spec_level_display(design, spec, level_code)
        rationale = (
            f"敏感度排名第 {srow.rank}（score={srow.score:.2f}），"
            f"{level_name} 水準（{level_label}）平均 {design.response_name}={observed:.4g}"
            f"{design.response_unit}，優於另一水準。"
        )
        recs.append(
            Recommendation(spec.label, spec.display_name, level_name, str(level_label), rationale)
        )
    return recs


# ---------------------------------------------------------------------------
# Baseline leg 比較（選用功能：拿某個 leg 當基準，跟其他候選 leg 比較改善幅度、
# 並說明因子該往哪個方向調整）
# ---------------------------------------------------------------------------


@dataclass
class BaselineFactorAdjustment:
    factor_id: str
    name: str
    baseline_value: object  # 基準 leg 該因子的實際物理值（numeric）或標籤（categorical）
    leg_value: object  # 候選 leg 該因子的實際物理值/標籤
    pct_change: Optional[float]  # 僅 numeric 因子且 baseline!=0 時有值
    direction: str  # "建議調升" / "建議調降" / "維持不變" / "從 A 改成 B" / "無資料，無法比較"


@dataclass
class BaselineCompareRow:
    run_label: str
    rank: int  # 1＝在候選 leg 之中改善幅度最大
    response_value: float
    baseline_response: float
    response_delta: float
    improvement_pct: Optional[float]  # 依 sn_direction 正規化，正值＝比基準好；baseline=0 時為 None
    factor_adjustments: List[BaselineFactorAdjustment]


@dataclass
class BaselineCompareResult:
    temperature: float
    baseline_run_label: str
    baseline_response: float
    rows: List[BaselineCompareRow]  # 依 rank 排序，rows[0] 改善幅度最大
    warnings: List[str] = field(default_factory=list)
    enabled: bool = True  # False＝完全沒有標記 1/2，這次不啟用比較
    skipped: bool = False  # True＝有標記但資料不完整/矛盾，優雅跳過
    skip_reason: Optional[str] = None


BASELINE_COLUMN_RUN_LABEL = "baseline"  # 代表 DOE 表 D 欄（既有的 baseline 值欄）
# 這個標籤本身，讓它可以跟 leg1..legN 共用同一套 1/2 標記機制（見
# `compare_to_baseline_leg` 開頭的說明）；doe_excel.read_baseline_compare_marks()
# 讀到 row1 的 D 欄標記時，用的就是這個字串當 label，跟這裡的判斷對應。


def _physical_value_for_code(factor: Factor, coded: Optional[int]) -> object:
    """coded ±1 -> 該因子實際用的物理值（numeric）或標籤（categorical）；coded 為
    None（無法解碼）時回傳 None。"""

    if coded is None:
        return None
    if factor.factor_type == "numeric":
        return factor.level_low_value if coded == -1 else factor.level_high_value
    return factor.level_low_label if coded == -1 else factor.level_high_label


def _physical_value_for_label(
    factor: Factor, run_matrix: RunMatrix, label: str, label_to_index: Dict[str, int]
) -> object:
    """把「基準/候選」的 label（leg 標籤或 `BASELINE_COLUMN_RUN_LABEL`）換算成該
    因子的實際物理值：label 是 leg 時走 coded ±1 換算（`_physical_value_for_code`）；
    label 是 `BASELINE_COLUMN_RUN_LABEL` 時改讀 `run_matrix.baseline_factor_values`
    （DOE 表 D 欄使用者自己填的基準值），numeric 因子會嘗試轉成 float，轉不了或
    留空一律回傳 None（呼叫端會視為「無資料，無法比較」，不會中斷其他因子）。
    """

    if label == BASELINE_COLUMN_RUN_LABEL:
        raw = (run_matrix.baseline_factor_values or {}).get(factor.factor_id)
        if raw is None or (isinstance(raw, str) and not raw.strip()):
            return None
        if factor.factor_type == "numeric":
            try:
                return float(raw)
            except (TypeError, ValueError):
                return None
        return str(raw)

    idx = label_to_index.get(label)
    if idx is None:
        return None
    return _physical_value_for_code(factor, run_matrix.factor_coded[factor.factor_id][idx])


def _response_value_for_label(
    run_matrix: RunMatrix, y: List[float], temp_key: float, label: str, label_to_index: Dict[str, int]
) -> Optional[float]:
    """把「基準/候選」的 label 換算成指定溫度下的 Response 數值：label 是 leg 時
    直接從已算好的 `y`（`_response_vector` 的結果，跟 `run_matrix.run_labels` 同順
    序）取值；label 是 `BASELINE_COLUMN_RUN_LABEL` 時改讀
    `run_matrix.baseline_responses[temp_key]`（DOE 表該溫度量測列的 D 欄），轉不成
    數字或沒填一律回傳 None。
    """

    if label == BASELINE_COLUMN_RUN_LABEL:
        raw = (run_matrix.baseline_responses or {}).get(temp_key)
        try:
            return float(raw)
        except (TypeError, ValueError):
            return None
    idx = label_to_index.get(label)
    return y[idx] if idx is not None else None


def _factor_adjustment(factor: Factor, baseline_value: object, leg_value: object) -> BaselineFactorAdjustment:
    if baseline_value is None or leg_value is None:
        return BaselineFactorAdjustment(
            factor.factor_id, factor.name, baseline_value, leg_value, None, "無資料，無法比較"
        )

    if factor.factor_type == "numeric":
        if leg_value == baseline_value:
            direction = "維持不變"
        elif leg_value > baseline_value:
            direction = "建議調升"
        else:
            direction = "建議調降"
        pct_change = (leg_value - baseline_value) / abs(baseline_value) * 100 if baseline_value != 0 else None
    else:
        direction = "維持不變" if leg_value == baseline_value else f"從 {baseline_value} 改成 {leg_value}"
        pct_change = None

    return BaselineFactorAdjustment(factor.factor_id, factor.name, baseline_value, leg_value, pct_change, direction)


def compare_to_baseline_leg(
    design: DesignConfig,
    run_matrix: RunMatrix,
    temperature,
    baseline_leg_label: Optional[str],
    compare_leg_labels: List[str],
) -> BaselineCompareResult:
    """拿 `baseline_leg_label` 當基準，跟 `compare_leg_labels` 裡每個候選比較指定
    溫度下的 response，找出改善幅度最大的候選、並逐因子說明調整方向。

    `baseline_leg_label`／`compare_leg_labels` 裡的每個 label 可以是 leg 標籤
    （"leg1".."legN"）或 `BASELINE_COLUMN_RUN_LABEL`（"baseline"，對應 DOE 表既有
    的 D 欄／baseline 值欄，跟 leg 共用同一套 1/2 標記機制）——leg 的實際數值來自
    正交表 coded ±1 換算，"baseline" 的實際數值則來自 `RunMatrix.baseline_responses`
    /`baseline_factor_values`（使用者自己在 D 欄填的基準值），兩種來源在這個函式
    裡是對等的，呼叫端不用分別處理。

    這是選用功能：兩個參數都是從 `doe_excel.read_baseline_compare_marks()` 讀回的
    原始標記（未過濾 leg 勾選列），呼叫端不用自己先過濾——這個函式會自行判斷每個
    label 是否「目前有資料可用」（leg 要在 `run_matrix.run_labels` 內；"baseline"
    要在指定溫度/因子上有填且能轉成數字），沒有的一律排除並在 warnings 說明，不
    會拋例外中斷。完全沒有標記（`baseline_leg_label is None` 且
    `compare_leg_labels` 為空）回傳 `enabled=False`；標記不完整或基準/候選都沒有
    可用資料則回傳 `skipped=True` 並附上原因，兩者都不會擋住其他分析。
    """

    if not baseline_leg_label and not compare_leg_labels:
        return BaselineCompareResult(
            temperature=_norm_temp(temperature), baseline_run_label="", baseline_response=0.0, rows=[],
            enabled=False,
        )

    if baseline_leg_label is None:
        return BaselineCompareResult(
            temperature=_norm_temp(temperature), baseline_run_label="", baseline_response=0.0, rows=[],
            skipped=True,
            skip_reason="尚未標記基準（最上面那一列填 1 的欄位），已略過 baseline 比較。",
        )

    label_to_index = {label: i for i, label in enumerate(run_matrix.run_labels)}
    known_labels = set(run_matrix.run_labels) | {BASELINE_COLUMN_RUN_LABEL}
    temp_key = _norm_temp(temperature)

    if baseline_leg_label not in known_labels:
        return BaselineCompareResult(
            temperature=temp_key, baseline_run_label=baseline_leg_label, baseline_response=0.0,
            rows=[], skipped=True,
            skip_reason=(
                f"基準（{baseline_leg_label}）目前沒有被 leg 勾選列勾選，不在本次分析範圍內，"
                "已略過 baseline 比較。"
            ),
        )

    warnings: List[str] = []
    valid_compare_labels = [lbl for lbl in compare_leg_labels if lbl in known_labels]
    excluded = [lbl for lbl in compare_leg_labels if lbl not in known_labels]
    if excluded:
        warnings.append(
            f"候選比較對象中有 {len(excluded)} 個（{'、'.join(excluded)}）目前沒有被 leg 勾選列勾選，"
            "不在本次分析範圍內，已從比較清單排除。"
        )

    y = _response_vector(run_matrix, temperature)
    baseline_response = _response_value_for_label(run_matrix, y, temp_key, baseline_leg_label, label_to_index)

    if baseline_response is None:
        return BaselineCompareResult(
            temperature=temp_key, baseline_run_label=baseline_leg_label, baseline_response=0.0,
            rows=[], warnings=warnings, skipped=True,
            skip_reason=(
                f"基準（{baseline_leg_label}）在這個溫度沒有可用的 Response 數值"
                "（baseline 欄未填或無法解析為數字），已略過 baseline 比較。"
            ),
        )

    direction = design.sn_direction
    target = design.nominal_target

    rows: List[BaselineCompareRow] = []
    for label in valid_compare_labels:
        leg_response = _response_value_for_label(run_matrix, y, temp_key, label, label_to_index)
        if leg_response is None:
            warnings.append(f"候選 {label} 在這個溫度沒有可用的 Response 數值，已從比較清單排除。")
            continue
        response_delta = leg_response - baseline_response

        if direction == SNDirection.LARGER:
            improvement_pct = (
                (leg_response - baseline_response) / abs(baseline_response) * 100 if baseline_response != 0 else None
            )
        elif direction == SNDirection.NOMINAL and target is not None:
            base_dist = abs(baseline_response - target)
            leg_dist = abs(leg_response - target)
            improvement_pct = (base_dist - leg_dist) / base_dist * 100 if base_dist != 0 else None
        else:  # SMALLER，或 NOMINAL 沒有 target 時退化成 smaller-the-better
            improvement_pct = (
                (baseline_response - leg_response) / abs(baseline_response) * 100 if baseline_response != 0 else None
            )

        if improvement_pct is None:
            warnings.append(f"baseline Response 為 0，{label} 無法計算改善百分比，僅顯示差值。")

        factor_adjustments = [
            _factor_adjustment(
                f,
                _physical_value_for_label(f, run_matrix, baseline_leg_label, label_to_index),
                _physical_value_for_label(f, run_matrix, label, label_to_index),
            )
            for f in design.factors
        ]

        rows.append(
            BaselineCompareRow(
                run_label=label, rank=0, response_value=leg_response, baseline_response=baseline_response,
                response_delta=response_delta, improvement_pct=improvement_pct, factor_adjustments=factor_adjustments,
            )
        )

    if not rows:
        return BaselineCompareResult(
            temperature=temp_key, baseline_run_label=baseline_leg_label, baseline_response=baseline_response,
            rows=[], warnings=warnings, skipped=True,
            skip_reason="沒有可比較的候選（標記 2 的欄位皆無可用資料，或完全沒有標記 2）。",
        )

    def _sort_key(row: BaselineCompareRow):
        if row.improvement_pct is not None:
            return -row.improvement_pct
        # baseline_response==0 邊界：退化用差值方向排序，跟 improvement_pct 的方向定義一致。
        return -row.response_delta if direction == SNDirection.LARGER else row.response_delta

    rows.sort(key=_sort_key)
    for i, row in enumerate(rows):
        row.rank = i + 1

    return BaselineCompareResult(
        temperature=_norm_temp(temperature), baseline_run_label=baseline_leg_label,
        baseline_response=baseline_response, rows=rows, warnings=warnings,
    )

    ("bottom cu(cu)", 14, 13),

# 這份預設量測值是在 leg 數＝8 的前提下記錄的（baseline, [leg1..leg8]）；leg 數
# 跟這個不同時，位置對不上，一律改用隨機值（見 `_write_doe_sheet_placeholders`）。
DEFAULT_WARPAGE_LEG_COUNT = 8
DEFAULT_WARPAGE_BY_TEMP: Dict[float, Tuple[float, List[float]]] = {
    25.0: (60, [23, -14, 36, 17, 72, 27, 89, 54]),
    50.0: (54, [46, 32, 55, 40, 56, 42, 65, 49]),
    75.0: (35, [33, 30, 35, 32, 32, 29, 34, 32]),
    260.0: (33, [34, 34, 34, 33, 31, 30, 30, 30]),
}


def _random_factor_levels() -> Tuple[float, float]:
    """因子個數超出 `DEFAULT_MAT_FACTOR_VALUES`/`DEFAULT_OTHER_FACTOR_VALUES` 筆數
    時，隨機產生一組不同的 Low/High 數值頂替，避免又退回通用的 Low/High 佔位
    文字；實際數值仍需使用者自行確認調整。
    """

    low = round(random.uniform(1, 200), 2)
    high = round(random.uniform(1, 200), 2)
    while high == low:
        high = round(random.uniform(1, 200), 2)
    return low, high


def _random_warpage_value() -> float:
    """溫度階段不在 `DEFAULT_WARPAGE_BY_TEMP`、或 leg 數跟 `DEFAULT_WARPAGE_LEG_COUNT`
    不同時，隨機產生一個落在既有量測資料量級範圍內的 Warpage 數值頂替，不留空。
    """

    return round(random.uniform(-20, 90), 1)


def _default_temperatures_for_count(count: int) -> List[float]:
    """依使用者指定的 response（量測列）數量，回傳建立新範本時要用的溫度階段
    清單：數量在 `DEFAULT_TEMPERATURES` 筆數以內就取前幾個；超出時在最後一個
    預設溫度之後，依 25 的間距往上疊加（僅是佔位遞增溫度，超出的溫度本來就沒有
    `DEFAULT_WARPAGE_BY_TEMP` 對應資料，量測值會改用隨機值，見
    `_write_doe_sheet_placeholders`），實際數值仍需使用者自行確認調整。
    """

    if count < 1:
        raise ValueError("response 數量必須至少為 1。")
    if count <= len(DEFAULT_TEMPERATURES):
        return DEFAULT_TEMPERATURES[:count]
    extra_needed = count - len(DEFAULT_TEMPERATURES)
    extra = [DEFAULT_TEMPERATURES[-1] + 25.0 * (i + 1) for i in range(extra_needed)]
    return DEFAULT_TEMPERATURES + extra


_NOT_FACTOR_ROW = object()  # _classify_factor_row 的哨兵值：這一列不是因子列（例如空白分隔列）


def _is_selected(value: object) -> bool:
    return isinstance(value, str) and value.strip().lower() == SELECT_MARK


def _is_response_item(item_val: object) -> bool:
    """C 欄（識別碼欄，因子名稱或溫度共用）的值能解析成數字，就是量測（temperature）列；否則是因子列。"""

    if item_val is None:
        return False
    try:
        float(item_val)
        return True
    except (TypeError, ValueError):
        return False


def _build_merge_map(ws: Worksheet) -> Dict[Tuple[int, int], object]:
    """回傳 (row, col) -> 該儲存格所屬合併範圍左上角的值。

    openpyxl 合併儲存格只有左上角真的存值，其餘儲存格讀回來是 None——讀取 mat
    群組跨列合併的 B 欄（同組除了第一列以外都會是 None）時，一律要透過這個表反查
    左上角的值，不能直接 `.value`（已實際踩過這個錯，此註解記錄原因）。
    """

    merge_map: Dict[Tuple[int, int], object] = {}
    for rng in ws.merged_cells.ranges:
        top_value = ws.cell(row=rng.min_row, column=rng.min_col).value
        for r in range(rng.min_row, rng.max_row + 1):
            for c in range(rng.min_col, rng.max_col + 1):
                merge_map[(r, c)] = top_value
    return merge_map


def _classify_factor_row(ws: Worksheet, row_idx: int, merge_map: Dict[Tuple[int, int], object]):
    """判斷因子列區塊（`FACTOR_START_ROW` 到重複標題列之間）裡的某一列是因子列
    還是純空白分隔列。是因子列時回傳 group（`GROUP_MAT` 或 None）；不是因子列（例如
    因子列與量測列之間那一列空白分隔列）時回傳 `_NOT_FACTOR_ROW`。

    判斷依據是 B 欄（分類標籤欄）的「有效值」是否非空——B 欄一律是整段合併的
    分類標籤（材料因子區塊只有一列時退化成沒有合併，直接讀 `.value` 也一樣有
    效），不看 C 欄（因子名稱）是否已填寫，才能讓「調整因子數量」後的空白範本
    也能被正確辨識出因子列數量。有效值等於 `GROUP_MAT`（"mats"）就是材料因子，
    `Factor.group` 設為 `GROUP_MAT`；其他任何非空值（例如 `GROUP_OTHER_LABEL`
    "factors"）都只是視覺分類，`group` 一律是 `None`，才能保持「各自獨立正交表
    欄位」的分析語意。
    """

    group_resolved = merge_map.get((row_idx, DOE_COL_GROUP), ws.cell(row=row_idx, column=DOE_COL_GROUP).value)
    if group_resolved is None or str(group_resolved).strip() == "":
        return _NOT_FACTOR_ROW
    if str(group_resolved).strip() == GROUP_MAT:
        return GROUP_MAT
    return None  # 因子列，但不是材料群組（group=None，各自獨立欄位）

def _detect_run_labels(ws: Worksheet) -> List[str]:
    """從 DOE 工作表 header 列偵測 leg1..legN 欄位，回傳 run label 清單。"""

    leg_count = 0
    for col in range(DOE_COL_LEG_START, ws.max_column + 1):
        val = ws.cell(row=ROW_HEADER, column=col).value
        if isinstance(val, str) and val.startswith("leg"):
            leg_count += 1
        else:
            break
    return [f"leg{i}" for i in range(1, leg_count + 1)]


def _is_valid_doe_sheet(ws: Worksheet) -> bool:
    return (
        ws.cell(row=ROW_HEADER, column=DOE_COL_BASELINE).value == "baseline"
        and ws.cell(row=ROW_HEADER, column=DOE_COL_LEG_START).value == "leg1"
        and ws.cell(row=ROW_LEG_SELECT, column=DOE_COL_SELECT).value == SELECT_LABEL
        and ws.cell(row=ROW_BASELINE_COMPARE, column=DOE_COL_SELECT).value == BASELINE_COMPARE_LABEL
    )


def _find_header_repeat_row(ws: Worksheet) -> Optional[int]:
    """找出因子列結束、量測列開始前的「重複標題列」（D 欄＝"baseline"，跟
    `ROW_HEADER` 那一列的標題列格式相同，範圍依 leg 數決定）。範本產生時動態算
    出這一列的位置，讀檔時同樣用掃描找，不假設固定在第幾列。
    """

    for row_idx in range(FACTOR_START_ROW, ws.max_row + 1):
        if ws.cell(row=row_idx, column=DOE_COL_BASELINE).value == "baseline":
            return row_idx
    return None


@dataclass
class CurrentStructure:
    mat_count: int
    other_count: int
    leg_count: int


def _current_structure(wb) -> Optional[CurrentStructure]:
    """回傳現有 DOE 工作表目前的（材料因子列數、非材料因子列數、leg 數），
    檔案還沒有 DOE 表（或不是本工具的格式）時回傳 None。
    """

    if DOE_SHEET not in wb.sheetnames:
        return None
    ws = wb[DOE_SHEET]
    if not _is_valid_doe_sheet(ws):
        return None
    header_repeat_row = _find_header_repeat_row(ws)
    if header_repeat_row is None:
        return None
    run_labels = _detect_run_labels(ws)
    merge_map = _build_merge_map(ws)
    mat_count = 0
    other_count = 0
    for row_idx in range(FACTOR_START_ROW, header_repeat_row):
        group = _classify_factor_row(ws, row_idx, merge_map)
        if group is _NOT_FACTOR_ROW:
            continue
        if group == GROUP_MAT:
            mat_count += 1
        else:
            other_count += 1
    return CurrentStructure(mat_count=mat_count, other_count=other_count, leg_count=len(run_labels))


def _read_existing_temperatures(wb) -> List[float]:
    """重新產生因子佔位樣板前，先取出目前的溫度階段，維持不變。"""

    if DOE_SHEET not in wb.sheetnames:
        return list(DEFAULT_TEMPERATURES)
    ws = wb[DOE_SHEET]
    header_repeat_row = _find_header_repeat_row(ws)
    if header_repeat_row is None:
        return list(DEFAULT_TEMPERATURES)
    temps: List[float] = []
    for row_idx in range(header_repeat_row + 1, ws.max_row + 1):
        item_val = ws.cell(row=row_idx, column=DOE_COL_NAME).value
        if _is_response_item(item_val):
            temps.append(float(item_val))
    if not temps:
        return list(DEFAULT_TEMPERATURES)
    return sorted(set(temps))


def _extract_preserved_responses(wb) -> Dict[Tuple[float, str], object]:
    """在覆寫 DOE 工作表前，先取出既有的量測值，供重新產生時儘量保留。"""

    preserved: Dict[Tuple[float, str], object] = {}
    if DOE_SHEET not in wb.sheetnames:
        return preserved
    ws = wb[DOE_SHEET]
    if not _is_valid_doe_sheet(ws):
        return preserved
    header_repeat_row = _find_header_repeat_row(ws)
    if header_repeat_row is None:
        return preserved
    run_labels = _detect_run_labels(ws)

    for row_idx in range(header_repeat_row + 1, ws.max_row + 1):
        item_val = ws.cell(row=row_idx, column=DOE_COL_NAME).value
        if not _is_response_item(item_val):
            continue
        temp = float(item_val)
        for leg_idx, leg_label in enumerate(run_labels):
            v = ws.cell(row=row_idx, column=DOE_COL_LEG_START + leg_idx).value
            if v is not None:
                preserved[(temp, leg_label)] = v
    return preserved


def _read_existing_selected_response_temperature(wb) -> Optional[float]:
    """重新產生因子佔位樣板前，先取出目前 select 欄（A欄）打 v 的量測列對應的溫度，
    重建時繼續保留同一個溫度被勾選，不會因為調整因子數量就把使用者原本選定要
    分析的溫度重設掉。剛好勾選一個才算數；沒勾、勾了超過一個、或是還沒有這個
    機制的舊檔案，一律回傳 None，交由呼叫端決定預設值（第一個溫度）。
    """

    if DOE_SHEET not in wb.sheetnames:
        return None
    ws = wb[DOE_SHEET]
    header_repeat_row = _find_header_repeat_row(ws)
    if header_repeat_row is None:
        return None
    selected: List[float] = []
    for row_idx in range(header_repeat_row + 1, ws.max_row + 1):
        item_val = ws.cell(row=row_idx, column=DOE_COL_NAME).value
        if not _is_response_item(item_val):
            continue
        if _is_selected(ws.cell(row=row_idx, column=DOE_COL_SELECT).value):
            selected.append(float(item_val))
    return selected[0] if len(selected) == 1 else None


def read_selected_response_temperature(path: str) -> float:
    """回傳 DOE 工作表量測列裡，select 欄（A欄）打 v 的那一個溫度——取代舊版 GUI
    的溫度下拉選單，要分析哪個溫度直接在 Excel 對應的量測列 select 欄打 v，跟
    因子列／leg 勾選列共用同一套「打 v 才納入」的操作方式。剛好只能勾一個溫度
    （一次分析只能對應一個溫度），沒勾或勾了超過一個都視為使用者輸入不完整，
    直接報錯並列出目前勾了哪些，不用猜測該分析哪一個。
    """

    wb = load_workbook(path, data_only=True)
    if DOE_SHEET not in wb.sheetnames:
        raise ValueError(f"{path} 找不到 '{DOE_SHEET}' 工作表，請先執行範本產生流程。")
    ws = wb[DOE_SHEET]
    header_repeat_row = _find_header_repeat_row(ws)
    if header_repeat_row is None:
        raise ValueError(f"{path} 的 '{DOE_SHEET}' 工作表找不到重複標題列（baseline/leg1..），格式可能不正確。")

    selected: List[float] = []
    for row_idx in range(header_repeat_row + 1, ws.max_row + 1):
        item_val = ws.cell(row=row_idx, column=DOE_COL_NAME).value
        if not _is_response_item(item_val):
            continue
        if _is_selected(ws.cell(row=row_idx, column=DOE_COL_SELECT).value):
            selected.append(float(item_val))

    if not selected:
        raise ValueError("量測列（溫度）目前沒有勾選任何一列，請在要分析的溫度那一列 select 欄打 v。")
    if len(selected) > 1:
        temps_str = "、".join(str(t) for t in selected)
        raise ValueError(
            f"量測列（溫度）目前勾選了 {len(selected)} 個（{temps_str}），一次只能分析一個溫度，"
            "請只保留其中一列的 select 欄打 v，其餘清空。"
        )
    return selected[0]


def read_baseline_compare_marks(path: str) -> Tuple[Optional[str], List[str]]:
    """讀 ROW_BASELINE_COMPARE（最上面那一列）的標記，回傳
    `(baseline_leg_label, compare_leg_labels)`：填 `1` 的那一欄是比較基準
    （剛好只能有一欄，超過一欄視為輸入矛盾直接報錯）；填 `2` 的欄位是要拿來跟
    基準比較的候選對象（0 個以上）；其餘（含空白）忽略。

    掃描範圍是 **D 欄（baseline 值欄）＋ E 欄開始的 leg1..legN**，不是只有 leg：
    D 欄本來就是「這一列（因子或量測）的基準值」，讓它也能被標記成這次比較用的
    基準/候選，1、2 可以隨時換位置，不用侷限在 leg 欄位。D 欄對應的 label 固定是
    `doe_core.BASELINE_COLUMN_RUN_LABEL`（"baseline"）——這跟 D3 儲存格顯示的
    "baseline" 字樣是同一件事，只是這裡用來標示「這欄的資料來源是 D 欄，不是某個
    leg」，呼叫端（`doe_core.compare_to_baseline_leg`）看到這個 label 就會改讀
    `RunMatrix.baseline_responses`/`baseline_factor_values`，不會去查 leg 資料。

    這是選用功能：完全沒有任何 1/2 標記時回傳 `(None, [])`，呼叫端要把這個視為
    「這次不啟用 baseline 比較」，優雅跳過、不當作錯誤。回傳的 leg label 是依
    `leg1..legN` 的欄位順序（未過濾 leg 勾選列），呼叫端需要自行對照
    `RunMatrix.run_labels`（已過濾的子集合）；"baseline" 這個 label 不受 leg 勾選
    列影響，永遠視為有效。
    """

    wb = load_workbook(path, data_only=True)
    if DOE_SHEET not in wb.sheetnames:
        raise ValueError(f"{path} 找不到 '{DOE_SHEET}' 工作表，請先執行範本產生流程。")
    ws = wb[DOE_SHEET]
    run_labels = _detect_run_labels(ws)

    columns_and_labels: List[Tuple[int, str]] = [(DOE_COL_BASELINE, BASELINE_COLUMN_RUN_LABEL)]
    columns_and_labels += [(DOE_COL_LEG_START + i, label) for i, label in enumerate(run_labels)]

    baseline_labels: List[str] = []
    compare_labels: List[str] = []
    for col_idx, label in columns_and_labels:
        raw = ws.cell(row=ROW_BASELINE_COMPARE, column=col_idx).value
        if raw is None or (isinstance(raw, str) and not raw.strip()):
            continue
        try:
            mark = int(float(raw))
        except (TypeError, ValueError):
            continue
        if mark == BASELINE_MARK:
            baseline_labels.append(label)
        elif mark == COMPARE_MARK:
            compare_labels.append(label)

    if not baseline_labels and not compare_labels:
        return None, []

    if len(baseline_labels) > 1:
        raise ValueError(
            f"最上面那一列（baseline 比較列）標記了 {len(baseline_labels)} 個基準"
            f"（{'、'.join(baseline_labels)}），D 欄／leg 欄位加起來只能有一欄標記 1，請只保留一欄。"
        )

    return (baseline_labels[0] if baseline_labels else None), compare_labels

def _write_doe_sheet_placeholders(
    wb: Workbook,
    num_mat_factors: int,
    num_other_factors: int,
    oa: OrthogonalArray,
    temperatures: List[float],
    preserved: Dict[Tuple[float, str], object],
    selected_response_temperature: Optional[float] = None,
) -> Tuple[int, int]:
    """(重)建立 DOE 工作表：因子列 leg 值依正交表 pattern 排好哪些格要打同一個值，
    值本身用 `DEFAULT_MAT_FACTOR_VALUES`/`DEFAULT_OTHER_FACTOR_VALUES` 的預設數值
    （超出筆數就隨機產生），不再是通用的 Low/High 佔位文字；量測列的量測值依
    (溫度,leg_label) 優先從 preserved 還原，其次用 `DEFAULT_WARPAGE_BY_TEMP`
    的預設量測值，都沒有就隨機產生。

    量測列的 select 欄（A欄）只會有一列打 v，代表「要分析哪個溫度」（取代舊版 GUI
    的溫度下拉選單）。`selected_response_temperature` 有給、且剛好是 `temperatures`
    其中一個時沿用該溫度；否則（新建檔案、或原本勾的溫度剛好被移除）預設勾第一個
    溫度，確保重建後一定有剛好一個溫度列被勾選，使用者不用另外去 Excel 補勾。

    材料因子（`num_mat_factors` 列）全部共用 table column 1，B 欄整段合併顯示
    `GROUP_MAT`（"mats"）；非材料因子（`num_other_factors` 列）依序指派到接下來
    的欄位，B 欄整段合併顯示 `GROUP_OTHER_LABEL`（"factors"，純視覺分類，不影響
    `group`）；量測列的 B 欄整段合併顯示 `RESPONSE_BLOCK_LABEL`（"warpage"）。
    以上三個區塊都是「B 欄整段合併成分類標籤、C 欄各自不合併填識別碼」的同一種
    模式，方便複製 C 欄或整列時不會卡到合併儲存格。全部文字置中，C 欄的識別碼
    （因子名稱／溫度）上黃色底色，B 欄分類標籤不上色。
    """

    if DOE_SHEET in wb.sheetnames:
        del wb[DOE_SHEET]
    ws = wb.create_sheet(DOE_SHEET)

    N = oa.N
    run_labels = [f"leg{i}" for i in range(1, N + 1)]
    last_col = DOE_COL_LEG_START + N - 1
    center = Alignment(horizontal="center", vertical="center")
    yellow = PatternFill("solid", fgColor="FFFF00")

    def _set(row_idx: int, col_idx: int, value: object = None) -> None:
        cell = ws.cell(row=row_idx, column=col_idx)
        if value is not None:
            cell.value = value
        cell.alignment = center

    # ROW_BASELINE_COMPARE（第 1 列）：A 欄放「baseline_vs」字樣，D 欄／leg 欄位一律
    # 留空（選用功能，不預設任何 1/2 標記，比照 ROW_LEG_SELECT 每次重建都重設的
    # 慣例，不做跨重建的標記保留）。
    _set(ROW_BASELINE_COMPARE, DOE_COL_SELECT, BASELINE_COMPARE_LABEL)

    # ROW_LEG_SELECT（第 2 列）：A 欄放「select」字樣，leg 欄位打 v（預設全選）。
    _set(ROW_LEG_SELECT, DOE_COL_SELECT, SELECT_LABEL)
    for i in range(N):
        _set(ROW_LEG_SELECT, DOE_COL_LEG_START + i, SELECT_MARK)

    # ROW_HEADER（第 3 列）：標題列。B:C 合併保留空白（之後再上底色，這裡先不寫
    # 「factor」字樣）。
    ws.merge_cells(start_row=ROW_HEADER, start_column=DOE_COL_GROUP, end_row=ROW_HEADER, end_column=DOE_COL_NAME)
    _set(ROW_HEADER, DOE_COL_GROUP)
    _set(ROW_HEADER, DOE_COL_BASELINE, "baseline")
    for i, label in enumerate(run_labels):
        _set(ROW_HEADER, DOE_COL_LEG_START + i, label)

    row = FACTOR_START_ROW
    factor_index = 0  # 因子名稱一律用 factor1..factorN 連號（mat／非材料因子共用同一組編號）

    if num_mat_factors > 0:
        colvec = oa.columns[1]
        mat_start_row = row
        for i in range(num_mat_factors):
            factor_index += 1
            name = f"factor{factor_index}"
            if i < len(DEFAULT_MAT_FACTOR_VALUES):
                _, low_value, high_value = DEFAULT_MAT_FACTOR_VALUES[i]
            else:
                low_value, high_value = _random_factor_levels()
            _set(row, DOE_COL_SELECT, SELECT_MARK)
            _set(row, DOE_COL_NAME, name)  # 子參數名稱
            ws.cell(row=row, column=DOE_COL_NAME).fill = yellow
            _set(row, DOE_COL_BASELINE, low_value)
            _write_leg_values(ws, row, colvec, low_value, high_value, center)
            row += 1
        mat_end_row = row - 1
        if mat_end_row > mat_start_row:
            ws.merge_cells(start_row=mat_start_row, start_column=DOE_COL_GROUP, end_row=mat_end_row, end_column=DOE_COL_GROUP)
        _set(mat_start_row, DOE_COL_GROUP, GROUP_MAT)

    other_start_col = 2 if num_mat_factors > 0 else 1
    if num_other_factors > 0:
        other_start_row = row
        for i in range(num_other_factors):
            colvec = oa.columns[other_start_col + i]
            factor_index += 1
            name = f"factor{factor_index}"
            if i < len(DEFAULT_OTHER_FACTOR_VALUES):
                _, low_value, high_value = DEFAULT_OTHER_FACTOR_VALUES[i]
            else:
                low_value, high_value = _random_factor_levels()
            _set(row, DOE_COL_SELECT, SELECT_MARK)
            _set(row, DOE_COL_NAME, name)  # 因子名稱
            ws.cell(row=row, column=DOE_COL_NAME).fill = yellow
            _set(row, DOE_COL_BASELINE, low_value)
            _write_leg_values(ws, row, colvec, low_value, high_value, center)
            row += 1
        other_end_row = row - 1
        if other_end_row > other_start_row:
            ws.merge_cells(
                start_row=other_start_row, start_column=DOE_COL_GROUP, end_row=other_end_row, end_column=DOE_COL_GROUP
            )
        _set(other_start_row, DOE_COL_GROUP, GROUP_OTHER_LABEL)

    row += 1  # 因子列與重複標題列之間留一列空白分隔

    # 重複標題列：只有 baseline/leg1..N，範圍依 leg 數決定（D~最後一欄 leg）。
    header_repeat_row = row
    _set(header_repeat_row, DOE_COL_BASELINE, "baseline")
    for i, label in enumerate(run_labels):
        _set(header_repeat_row, DOE_COL_LEG_START + i, label)
    row += 1

    preserved_used = 0
    response_start_row = row
    sorted_temps = sorted(temperatures)
    target_temp = (
        selected_response_temperature
        if selected_response_temperature is not None and selected_response_temperature in sorted_temps
        else sorted_temps[0]
    )
    for temp in sorted_temps:
        default_baseline, default_legs = DEFAULT_WARPAGE_BY_TEMP.get(float(temp), (None, None))
        if N != DEFAULT_WARPAGE_LEG_COUNT:
            default_baseline, default_legs = None, None  # leg 數跟預設資料的位置對不上，一律改隨機值

        _set(row, DOE_COL_SELECT, SELECT_MARK if temp == target_temp else None)
        _set(row, DOE_COL_NAME, temp)
        ws.cell(row=row, column=DOE_COL_NAME).fill = yellow
        _set(row, DOE_COL_BASELINE, default_baseline if default_baseline is not None else _random_warpage_value())
        for i, leg_label in enumerate(run_labels):
            key = (float(temp), leg_label)
            if key in preserved:
                value = preserved[key]
                preserved_used += 1
            elif default_legs is not None:
                value = default_legs[i]
            else:
                value = _random_warpage_value()
            _set(row, DOE_COL_LEG_START + i, value)
        row += 1
    response_end_row = row - 1
    if response_end_row >= response_start_row:
        if response_end_row > response_start_row:
            ws.merge_cells(
                start_row=response_start_row, start_column=DOE_COL_GROUP,
                end_row=response_end_row, end_column=DOE_COL_GROUP,
            )
        _set(response_start_row, DOE_COL_GROUP, RESPONSE_BLOCK_LABEL)

    _autofit_columns(ws)

    return preserved_used, len(preserved)


def generate_template(
    path: str,
    num_factors: Optional[int] = None,
    num_mat_factors: int = 0,
    num_other_factors: int = 0,
    num_legs: Optional[int] = None,
    num_responses: Optional[int] = None,
    force: bool = False,
) -> GenerateTemplateReport:
    """建立新範本，或（材料/非材料因子數量、leg 數變更時）重新產生因子列的
    Low/High 佔位樣板。

    `num_factors` 是舊版單一參數的相容寫法，等同 `num_mat_factors=0,
    num_other_factors=num_factors`（不要同時指定兩種寫法）。

    新檔案：需要指定因子數量（`num_factors`，或 `num_mat_factors`/
    `num_other_factors` 至少一個 > 0）。`num_legs` 可選，留空時依因子數自動算
    最小需求的 run 數（見 `doe_core.generate_orthogonal_array` 的 `min_runs`）。
    `num_responses` 可選，指定要建立幾個量測列（溫度階段），留空時預設
    `len(DEFAULT_TEMPERATURES)` 個（見 `_default_temperatures_for_count`）；只在
    建立新檔案時生效，既有檔案的量測列數量一律沿用檔案本身、不受這個參數影響。

    既有檔案：
      - 都不指定（`num_factors`/`num_mat_factors`/`num_other_factors`/
        `num_legs` 全部維持預設）或跟目前結構相同：不動 DOE 工作表（因子列/量測
        列都原樣保留），只確保「說明」工作表是最新版本。
      - 材料/非材料因子數量或 leg 數要求的正交表大小跟目前不同：這會讓正交表結
        構整個改變，所有因子列的 leg 值會被重新排成新的 Low/High 佔位樣板（覆寫
        現有內容，需要 `--force` 確認）；量測列的量測值儘量依 (溫度, leg 位置)
        保留。
    """

    if num_factors is not None:
        if num_mat_factors or num_other_factors:
            raise ValueError("不要同時指定 num_factors 與 num_mat_factors/num_other_factors。")
        num_other_factors = num_factors

    if not os.path.exists(path):
        if num_mat_factors < 1 and num_other_factors < 1:
            raise ValueError("建立新檔案需要指定因子數量（--mat-factors 和/或 --other-factors，至少一個 >= 1）。")
        wb = Workbook()
        default_sheet = wb.active
        num_columns = (1 if num_mat_factors > 0 else 0) + num_other_factors
        oa = generate_orthogonal_array(num_columns, min_runs=num_legs)
        temperatures = (
            _default_temperatures_for_count(num_responses) if num_responses is not None else DEFAULT_TEMPERATURES
        )
        _write_doe_sheet_placeholders(
            wb, num_mat_factors, num_other_factors, oa, temperatures, preserved={}
        )
        _write_help_sheet(wb)
        del wb[default_sheet.title]
        _align_cells_after_write(wb)
        _pin_doe_sheet_first(wb)
        wb.save(path)
        return GenerateTemplateReport(
            created=True, factor_count=num_mat_factors + num_other_factors,
            mat_factor_count=num_mat_factors, other_factor_count=num_other_factors,
            run_count=oa.N, preserved_cells=0, dropped_cells=0, response_count=len(temperatures),
        )

    wb = load_workbook(path)
    current = _current_structure(wb)

    # 沒指定的部分沿用目前檔案的結構（讓使用者只改其中一個數字時，另一個不會被
    # 意外歸零）。
    effective_mat = num_mat_factors if num_mat_factors > 0 else (current.mat_count if current else 0)
    effective_other = num_other_factors if num_other_factors > 0 else (current.other_count if current else 0)
    if effective_mat < 1 and effective_other < 1:
        raise ValueError(
            f"{path} 存在但沒有可辨識的因子結構，且沒有指定因子數量"
            "（--mat-factors 和/或 --other-factors，至少一個 >= 1）。"
        )
    num_columns = (1 if effective_mat > 0 else 0) + effective_other
    oa = generate_orthogonal_array(num_columns, min_runs=num_legs)

    structure_changed = current is not None and (
        current.mat_count != effective_mat or current.other_count != effective_other or current.leg_count != oa.N
    )

    if current is not None and not structure_changed:
        # 跟目前檔案結構相同（不管是使用者沒指定、或指定了但剛好跟現況一樣）：不動
        # DOE 工作表，只確保「說明」工作表是最新版本，避免每次呼叫都重建一次因子列
        # （那樣會把使用者已經填好的 leg 值/因子名稱洗掉）。
        _write_help_sheet(wb)
        _align_cells_after_write(wb)
        _pin_doe_sheet_first(wb)
        wb.save(path)
        return GenerateTemplateReport(
            created=False, factor_count=current.mat_count + current.other_count,
            mat_factor_count=current.mat_count, other_factor_count=current.other_count,
            run_count=current.leg_count, preserved_cells=0, dropped_cells=0,
        )

    if current is None and not force:
        # 檔案存在，但沒有偵測到本工具認得的 DOE 工作表格式（可能是空白檔案、或是
        # 舊版欄位配置的檔案——這次改版不做舊格式相容讀取）。直接覆寫有資料遺失的
        # 風險，要求 --force 確認，不要沒問過使用者就把可能存在的真實資料蓋掉。
        raise ValueError(
            f"{path} 已存在，但沒有偵測到本工具目前版本認得的 '{DOE_SHEET}' 工作表格式"
            "（可能是空白檔案，或是舊版欄位配置產生的檔案）。若確定要在這個檔案裡建立新範本"
            "（會覆寫檔案內容），請加上 --force 重新執行。"
        )

    if structure_changed and not force:
        raise ValueError(
            f"目前檔案有 {current.mat_count} 個材料因子、{current.other_count} 個非材料因子、"
            f"{current.leg_count} 個 leg，要求改成 {effective_mat} 個材料因子、{effective_other} 個"
            f"非材料因子、{oa.N} 個 leg，這會讓正交表結構整個改變：所有因子列的 leg 值會被重新排成"
            "新的 Low/High 佔位樣板（覆寫現有內容，需要重新填寫實際數值）。量測列的量測值會儘量依"
            "溫度/leg 位置保留。若確定要繼續，請加上 --force。"
        )

    temperatures = _read_existing_temperatures(wb)
    preserved = _extract_preserved_responses(wb)
    selected_response_temperature = _read_existing_selected_response_temperature(wb)
    preserved_used, preserved_total = _write_doe_sheet_placeholders(
        wb, effective_mat, effective_other, oa, temperatures, preserved, selected_response_temperature
    )
    _write_help_sheet(wb)
    _align_cells_after_write(wb)
    _pin_doe_sheet_first(wb)
    wb.save(path)

    warnings = []
    if structure_changed:
        warnings.append("因子數量或 leg 數已變更，所有因子列的 leg 值已重新排成 Low/High 佔位樣板，請重新填入實際數值。")

    return GenerateTemplateReport(
        created=False,
        factor_count=effective_mat + effective_other,
        mat_factor_count=effective_mat,
        other_factor_count=effective_other,
        run_count=oa.N,
        preserved_cells=preserved_used,
        dropped_cells=preserved_total - preserved_used,
        warnings=warnings,
    )


def read_design_and_runs(
    path: str, sn_direction: Optional[SNDirection] = None
) -> Tuple[DesignConfig, RunMatrix, List[str]]:
    """讀單一 DOE 工作表，回推出 DesignConfig（因子定義、溫度階段）與 RunMatrix
    （decode 過的因子矩陣、量測值），一次到位。

    `sn_direction` 是分析時的參數，不是檔案裡的設定；不給時預設 Smaller-the-better。
    Response 名稱固定用 `DEFAULT_RESPONSE_PREFIX`（新版欄位配置沒有獨立欄位可存
    per-row 的 response 名稱）。回傳的 `responses`/`design.temperatures` 涵蓋
    DOE 工作表裡所有溫度列（不受量測列 select 欄影響）；要分析哪一個溫度，是
    另外呼叫 `read_selected_response_temperature()` 決定（量測列 select 欄打 v
    的那一個），兩者職責分開。

    只有「select」欄打勾的因子列會納入分析；只有「leg 勾選列」（row2）打勾的 leg
    欄會納入分析（其餘 leg 整欄從資料剔除，不是留著但忽略）。mat 群組的因子列
    （B 欄整段合併＝`GROUP_MAT` "mats"）會共用同一個正交表欄位（見
    `doe_core.DesignConfig.factor_table_column`）；B 欄是其他非空值（例如
    `GROUP_OTHER_LABEL` "factors"）的因子列，只是視覺分類，各自仍是獨立欄位。
    因子列沒有獨立的因子代號欄，C 欄的名稱本身就是識別碼（不管是不是材料因子，
    一律讀 C 欄），名稱留空的已勾選因子列會被跳過並加警告，不會擋住其他因子的
    分析。
    """

    wb = load_workbook(path, data_only=True)
    if DOE_SHEET not in wb.sheetnames:
        raise ValueError(f"{path} 找不到 '{DOE_SHEET}' 工作表，請先執行範本產生流程。")
    ws = wb[DOE_SHEET]

    header_repeat_row = _find_header_repeat_row(ws)
    if header_repeat_row is None:
        raise ValueError(f"{path} 的 '{DOE_SHEET}' 工作表找不到重複標題列（baseline/leg1..），格式可能不正確。")

    warnings: List[str] = []
    all_run_labels = _detect_run_labels(ws)
    leg_count = len(all_run_labels)
    merge_map = _build_merge_map(ws)

    # 先掃一輪因子列區塊（不管有沒有勾選 select）算出每一列該對應到哪個 table
    # column：欄位指派（含分組）要以「列」為單位固定下來，不能因為某些因子列被
    # select 排除、或名稱還沒填，就讓後面的列往前遞補——遞補後剩下因子的 table
    # column 編號會跟它們實際 leg 資料的正交表 pattern 對不上，變成假的「不符」
    # 警告跟錯誤的 SS。
    row_group: Dict[int, Optional[str]] = {}
    row_table_column: Dict[int, int] = {}
    factor_rows: List[int] = []
    group_stable_columns: Dict[str, int] = {}
    next_stable_col = 1
    for row_idx in range(FACTOR_START_ROW, header_repeat_row):
        group = _classify_factor_row(ws, row_idx, merge_map)
        if group is _NOT_FACTOR_ROW:
            continue
        factor_rows.append(row_idx)
        row_group[row_idx] = group
        key = group if group else f"__solo_row__{row_idx}"
        if key not in group_stable_columns:
            group_stable_columns[key] = next_stable_col
            next_stable_col += 1
        row_table_column[row_idx] = group_stable_columns[key]

    selected_indices = [
        i for i in range(leg_count) if _is_selected(ws.cell(row=ROW_LEG_SELECT, column=DOE_COL_LEG_START + i).value)
    ]
    if len(selected_indices) < 2:
        raise ValueError(
            f"第 {ROW_LEG_SELECT} 列（leg 勾選列）目前勾選了 {len(selected_indices)} 個 leg，"
            "至少要勾選 2 個 leg 才能分析。"
        )
    run_labels = [all_run_labels[i] for i in selected_indices]

    def _filter_legs(all_values: List[object]) -> List[object]:
        return [all_values[i] for i in selected_indices]

    factors: List[Factor] = []
    factor_coded: Dict[str, List[Optional[int]]] = {}
    baseline_factor_values: Dict[str, object] = {}
    level_errors: List[str] = []
    seen_factor_ids: set = set()

    for row_idx in factor_rows:
        if not _is_selected(ws.cell(row=row_idx, column=DOE_COL_SELECT).value):
            continue  # 這個因子列沒有打勾，整列排除，不影響其他因子的欄位指派

        group = row_group[row_idx]
        name_val = ws.cell(row=row_idx, column=DOE_COL_NAME).value  # C 欄：因子名稱（一律不合併）
        if name_val is None or str(name_val).strip() == "":
            warnings.append(f"第 {row_idx} 列的因子名稱尚未填寫，本次分析未納入。")
            continue
        factor_id = str(name_val).strip()
        if factor_id in seen_factor_ids:
            level_errors.append(f"因子名稱「{factor_id}」重複出現，請確認每個因子名稱是唯一的。")
            continue
        seen_factor_ids.add(factor_id)

        baseline_val = ws.cell(row=row_idx, column=DOE_COL_BASELINE).value
        all_leg_values = [ws.cell(row=row_idx, column=DOE_COL_LEG_START + i).value for i in range(leg_count)]
        leg_values = _filter_legs(all_leg_values)

        try:
            low, high, is_numeric = infer_factor_levels(leg_values)
        except ValueError as exc:
            level_errors.append(f"因子 {factor_id}：{exc}")
            continue

        table_column = row_table_column[row_idx]
        if is_numeric:
            factor = Factor(
                factor_id=factor_id, name=factor_id, factor_type="numeric",
                level_low_value=float(low), level_high_value=float(high), baseline=baseline_val,
                group=group, table_column=table_column,
            )
        else:
            factor = Factor(
                factor_id=factor_id, name=factor_id, factor_type="categorical",
                level_low_label=str(low), level_high_label=str(high), baseline=baseline_val,
                group=group, table_column=table_column,
            )
        factors.append(factor)
        factor_coded[factor_id] = [decode_level(v, factor) for v in leg_values]
        baseline_factor_values[factor_id] = baseline_val

    if level_errors:
        raise ValueError("部分因子列無法判斷 Low/High 水準，請確認 leg 欄位填寫：\n" + "\n".join(level_errors))

    balance_errors = []
    for factor_id, coded in factor_coded.items():
        low_n = sum(1 for c in coded if c == -1)
        high_n = sum(1 for c in coded if c == 1)
        if low_n != high_n:
            balance_errors.append(f"因子 {factor_id}：目前勾選的 leg 裡 Low={low_n} 筆、High={high_n} 筆。")
    if balance_errors:
        raise ValueError(
            "目前勾選的 leg 子集合讓部分因子的 Low/High 數量不平衡（ANOVA 等公式假設每個水準各佔"
            "一半，不平衡時算出來的數字不正確，不是精確度變差而是公式本身不成立）。請調整 leg 勾選"
            "（或全選）讓每個因子的 Low/High 數量相等：\n" + "\n".join(balance_errors)
        )

    if not factors:
        raise ValueError(f"{path} 的 '{DOE_SHEET}' 工作表沒有找到任何已勾選且已填寫名稱的因子列。")

    # 量測列：header_repeat_row 之後，C 欄（識別碼欄，不合併）能解析成數字的列。
    responses: Dict[float, List[Optional[float]]] = {}
    baseline_responses: Dict[float, Optional[object]] = {}
    for row_idx in range(header_repeat_row + 1, ws.max_row + 1):
        item_val = ws.cell(row=row_idx, column=DOE_COL_NAME).value
        if not _is_response_item(item_val):
            continue
        temp = float(item_val)
        all_leg_values = [ws.cell(row=row_idx, column=DOE_COL_LEG_START + i).value for i in range(leg_count)]
        leg_values = _filter_legs(all_leg_values)
        parsed_vals: List[Optional[float]] = []
        for v in leg_values:
            if v is None or (isinstance(v, str) and not v.strip()):
                parsed_vals.append(None)
                continue
            try:
                parsed_vals.append(float(v))
            except (TypeError, ValueError):
                parsed_vals.append(None)
                warnings.append(f"溫度 {temp} 的量測值中有無法解析為數字的內容（{v!r}），已當作未填處理。")
        responses[temp] = parsed_vals
        baseline_responses[temp] = ws.cell(row=row_idx, column=DOE_COL_BASELINE).value

    design = DesignConfig(
        response_name=DEFAULT_RESPONSE_PREFIX,
        response_unit="",
        temperatures=sorted(responses.keys()),
        sn_direction=sn_direction or SNDirection.SMALLER,
        nominal_target=None,
        factors=factors,
    )
    run_matrix = RunMatrix(
        run_labels=run_labels,
        factor_coded=factor_coded,
        responses=responses,
        baseline_factor_values=baseline_factor_values,
        baseline_responses=baseline_responses,
    )
    return design, run_matrix, warnings

_COLOR_ORANGE = "ED7D31"

def _style_series(
    series, color_hex: str, *, fill: bool = False, marker: Optional[str] = None,
    dashed: bool = False, no_line: bool = False,
) -> None:
    """套用 Office 色票到單一 chart series：長條圖填色（fill=True）、線圖/散佈圖
    連線（預設，dashed 可畫虛線）、或純散佈點不連線（no_line=True，搭配 marker）。"""
    gp = GraphicalProperties()
    if fill:
        gp.solidFill = color_hex
    elif no_line:
        gp.line = LineProperties(noFill=True)
    else:
        gp.line = LineProperties(solidFill=color_hex)
        if dashed:
            gp.line.dashStyle = "dash"
    series.graphicalProperties = gp
    if marker:
        series.marker = Marker(symbol=marker, size=6)
        series.marker.graphicalProperties = GraphicalProperties(
            solidFill=color_hex, ln=LineProperties(solidFill=color_hex)
        )


def _col_width_emu(ws: Worksheet, col_idx: int) -> int:
    dim = ws.column_dimensions.get(get_column_letter(col_idx))
    chars = dim.width if dim and dim.width else 8.43
    return pixels_to_EMU(round(chars * 7 + 5))


def _row_height_emu(ws: Worksheet, row_idx: int) -> int:
    dim = ws.row_dimensions.get(row_idx)
    points = dim.height if dim and dim.height else DEFAULT_ROW_HEIGHT
    return int(points * 12700)  # 1 point = 12700 EMU


def _add_chart_fixed(ws: Worksheet, chart, anchor_cell: str) -> None:
    """加入圖表，並把錨定方式設成 `TwoCellAnchor(editAs="absolute")`——對應 Excel
    圖表『大小及位置』屬性的『大小位置不隨儲存格而改變』。openpyxl 預設的
    `ws.add_chart(chart, "A1")` 只會產生 `OneCellAnchor`（隨儲存格移動但不縮放），
    不是使用者要的完全固定，所以這裡改成自己組 `TwoCellAnchor`：從 `anchor_cell`
    往右/往下累加欄寬/列高（openpyxl 沒有原生欄寬轉像素公式，用 Excel 常見的
    `字元數*7+5` 近似值），直到超過 `chart.width`/`chart.height`（cm）換算成的 EMU，
    決定右下角落在哪個儲存格——只是拿來算初始大小，不影響「absolute」之後圖表
    大小固定不隨儲存格改變的行為，欄寬換算誤差不影響正確性。"""
    row0, col0 = coordinate_to_tuple(anchor_cell.upper())
    from_row, from_col = row0 - 1, col0 - 1
    target_w = cm_to_EMU(chart.width)
    target_h = cm_to_EMU(chart.height)

    to_col, acc_w = from_col, 0
    while True:
        w = _col_width_emu(ws, to_col + 1)
        if acc_w + w >= target_w:
            col_off = target_w - acc_w
            break
        acc_w += w
        to_col += 1

    to_row, acc_h = from_row, 0
    while True:
        h = _row_height_emu(ws, to_row + 1)
        if acc_h + h >= target_h:
            row_off = target_h - acc_h
            break
        acc_h += h
        to_row += 1

    anchor = TwoCellAnchor(editAs="absolute")
    anchor._from.col, anchor._from.row = from_col, from_row
    anchor.to.col, anchor.to.row = to_col, to_row
    anchor.to.colOff, anchor.to.rowOff = col_off, row_off
    ws.add_chart(chart, anchor)

# --- 2. Main Effect ------------------------------------------------------


def _main_effect_y_range(ws: Worksheet, header_row: int, last_data_row: int) -> Tuple[Optional[float], Optional[float]]:
    """算 Mean(Low)/Mean(High) 表格全部因子的數值範圍（含 10% padding），給每個因子
    小圖跟合併圖共用，讓兩種版面的 y 軸刻度一致（模擬原本 matplotlib `sharey=True`
    的效果）。沒有可用數值時回傳 (None, None)，呼叫端據此跳過設定 y 軸刻度。"""
    values: List[float] = []
    for r in range(header_row + 1, last_data_row + 1):
        for col in (3, 4):
            v = ws.cell(row=r, column=col).value
            if isinstance(v, (int, float)):
                values.append(v)
    if not values:
        return None, None
    y_min, y_max = min(values), max(values)
    y_pad = (y_max - y_min) * 0.1 if y_max > y_min else 1.0
    return y_min - y_pad, y_max + y_pad


def _build_main_effect_charts(ws: Worksheet, header_row: int, last_data_row: int, response_name: str) -> List[Tuple[LineChart, str]]:
    """直接參照既有的 Mean(Low)/Mean(High) 表格。比照原本 matplotlib 版本的版面：
    每個因子各自一張小圖（Low/High 兩點折線），並排顯示；y 軸刻度對齊全部因子的
    數值範圍，模擬原本 `sharey=True` 的效果。回傳 (chart, 欄位字母) 清單，呼叫端
    負責組出實際 anchor cell。"""
    ws.cell(row=header_row, column=7, value="Low")
    ws.cell(row=header_row, column=8, value="High")
    cats = Reference(ws, min_col=7, max_col=8, min_row=header_row, max_row=header_row)
    y_min, y_max = _main_effect_y_range(ws, header_row, last_data_row)

    charts: List[Tuple[LineChart, str]] = []
    for i, r in enumerate(range(header_row + 1, last_data_row + 1)):
        chart = LineChart()
        chart.title = ws.cell(row=r, column=1).value or ws.cell(row=r, column=2).value
        chart.legend = None
        if i == 0:
            chart.y_axis.title = response_name
        if y_min is not None:
            chart.y_axis.scaling.min = y_min
            chart.y_axis.scaling.max = y_max
        data = Reference(ws, min_col=3, max_col=4, min_row=r, max_row=r)
        chart.add_data(data, titles_from_data=False, from_rows=True)
        chart.set_categories(cats)
        _style_series(chart.series[0], _COLOR_BLUE, marker="circle")
        chart.width, chart.height = 7.0, 9.0
        charts.append((chart, get_column_letter(1 + i * 5)))
    return charts


def _build_main_effect_combined_chart(ws: Worksheet, header_row: int, last_data_row: int, response_name: str) -> LineChart:
    """跟 `_build_main_effect_charts`（每個因子一張小圖）並存的另一種版面：1 個組合
    圖表（X 軸=因子，2 條 series=Low/High），方便一眼比較所有因子。直接參照既有的
    Mean(Low)/Mean(High) 表格，不用另外寫 helper 欄；y 軸刻度跟每個因子小圖共用
    `_main_effect_y_range()`，數值範圍要一致，不是各自 auto-scale。"""
    chart = LineChart()
    chart.title = "Main Effects Plot（合併）"
    chart.y_axis.title = response_name
    y_min, y_max = _main_effect_y_range(ws, header_row, last_data_row)
    if y_min is not None:
        chart.y_axis.scaling.min = y_min
        chart.y_axis.scaling.max = y_max
    data = Reference(ws, min_col=3, max_col=4, min_row=header_row, max_row=last_data_row)
    cats = Reference(ws, min_col=1, min_row=header_row + 1, max_row=last_data_row)
    chart.add_data(data, titles_from_data=True)
    chart.set_categories(cats)
    for series, color in zip(chart.series, (_COLOR_BLUE, _COLOR_ORANGE)):
        _style_series(series, color, marker="circle")
    n = last_data_row - header_row
    chart.width = max(14.0, 1.2 * n)
    chart.height = 9.0  # 跟 _build_main_effect_charts 每個因子小圖同高，版面一致
    return chart


# --- 3. Interaction --------------------------------------------------------


def _write_interaction_helper_table(ws: Worksheet, start_row: int, result: InteractionResult) -> int:
    """寫一個 3x3 的 mean_response 交叉表（Low(A)/High(A) x Low(B)/High(B)），
    這裡原本完全沒有資料表，只有圖——回傳表格佔用的最後一列。"""

    def _lookup(level_a: int, level_b: int) -> Optional[float]:
        pt = next((p for p in result.points if p.level_a == level_a and p.level_b == level_b), None)
        return pt.mean_response if pt is not None else None

    ws.cell(row=start_row, column=2, value="Low(A)")
    ws.cell(row=start_row, column=3, value="High(A)")
    ws.cell(row=start_row + 1, column=1, value="Low(B)")
    ws.cell(row=start_row + 1, column=2, value=_lookup(-1, -1))
    ws.cell(row=start_row + 1, column=3, value=_lookup(1, -1))
    ws.cell(row=start_row + 2, column=1, value="High(B)")
    ws.cell(row=start_row + 2, column=2, value=_lookup(-1, 1))
    ws.cell(row=start_row + 2, column=3, value=_lookup(1, 1))
    return start_row + 2


def _build_interaction_chart(ws: Worksheet, start_row: int, result: InteractionResult, response_name: str) -> LineChart:
    """對應 `_write_interaction_helper_table` 寫出的 3x3 表格：每一列（Low(B)/High(B)）
    當一條 series，欄（Low(A)/High(A)）當類別，複製原本兩條線的交互作用圖。"""
    chart = LineChart()
    chart.title = f"{result.factor_a} x {result.factor_b}"
    chart.x_axis.title = result.factor_a
    chart.y_axis.title = response_name
    data = Reference(ws, min_col=1, max_col=3, min_row=start_row + 1, max_row=start_row + 2)
    chart.add_data(data, titles_from_data=True, from_rows=True)
    cats = Reference(ws, min_col=2, max_col=3, min_row=start_row, max_row=start_row)
    chart.set_categories(cats)
    for series, color in zip(chart.series, (_COLOR_BLUE, _COLOR_ORANGE)):
        _style_series(series, color, marker="circle")
    chart.width = 12.0
    chart.height = 9.0
    return chart

# --- 5. Half-Normal Plot -----------------------------------------------------


def _write_half_normal_helper_table(ws: Worksheet, start_row: int, start_col: int, result: EffectScreeningResult) -> Dict[str, Optional[int]]:
    """全新 helper 表格（Label/Half-normal quantile/|Effect|），算法照抄原本 matplotlib
    版本的 probability/quantile 公式；另外附一組 2 點擬合虛線範圍。逐點文字標籤
    （原本 ax.annotate）不做原生對應，helper 表格本身已按順序列出每點的 label 可對照。"""
    rows = sorted(result.rows, key=lambda r: abs(r.effect))
    n = len(rows)
    probabilities = [(i + 1 - 0.375) / (n + 0.25) for i in range(n)] if n else []
    expected = [float(scipy_stats.norm.ppf((1.0 + p) / 2.0)) for p in probabilities]
    observed = [abs(r.effect) for r in rows]

    ws.cell(row=start_row, column=start_col, value="Label")
    ws.cell(row=start_row, column=start_col + 1, value="Half-normal quantile")
    ws.cell(row=start_row, column=start_col + 2, value="|Effect|")
    for i, (row, x, y) in enumerate(zip(rows, expected, observed)):
        ws.cell(row=start_row + 1 + i, column=start_col, value=row.label)
        ws.cell(row=start_row + 1 + i, column=start_col + 1, value=x)
        ws.cell(row=start_row + 1 + i, column=start_col + 2, value=y)
    last_data_row = start_row + n

    info: Dict[str, Optional[int]] = {
        "start_col": start_col, "last_data_row": last_data_row,
        "fit_row0": None, "fit_row1": None,
    }
    if n >= 2:
        baseline_count = max(2, n // 2)
        slope, intercept = np.polyfit(expected[:baseline_count], observed[:baseline_count], 1)
        x0, x1 = 0.0, max(expected) * 1.05
        fit_row0, fit_row1 = last_data_row + 2, last_data_row + 3
        ws.cell(row=fit_row0 - 1, column=start_col + 1, value="RefLine X")
        ws.cell(row=fit_row0 - 1, column=start_col + 2, value="RefLine Y")
        ws.cell(row=fit_row0, column=start_col + 1, value=x0)
        ws.cell(row=fit_row0, column=start_col + 2, value=intercept + slope * x0)
        ws.cell(row=fit_row1, column=start_col + 1, value=x1)
        ws.cell(row=fit_row1, column=start_col + 2, value=intercept + slope * x1)
        info["fit_row0"], info["fit_row1"] = fit_row0, fit_row1
    return info


def _build_half_normal_chart(ws: Worksheet, start_row: int, info: Dict[str, Optional[int]]) -> ScatterChart:
    start_col = info["start_col"]
    chart = ScatterChart()
    chart.title = "Half-Normal Plot of Effects"
    chart.x_axis.title = "Half-normal theoretical quantile"
    chart.y_axis.title = "|Effect|"
    yref = Reference(ws, min_col=start_col + 2, min_row=start_row, max_row=info["last_data_row"])
    xref = Reference(ws, min_col=start_col + 1, min_row=start_row + 1, max_row=info["last_data_row"])
    main_series = Series(yref, xref, title_from_data=True)
    _style_series(main_series, _COLOR_BLUE, marker="circle", no_line=True)
    chart.series.append(main_series)
    if info["fit_row0"] is not None:
        yref2 = Reference(ws, min_col=start_col + 2, min_row=info["fit_row0"], max_row=info["fit_row1"])
        xref2 = Reference(ws, min_col=start_col + 1, min_row=info["fit_row0"], max_row=info["fit_row1"])
        fit_series = Series(yref2, xref2)
        _style_series(fit_series, _COLOR_GRAY, dashed=True)
        chart.series.append(fit_series)
    chart.width = 16.0
    chart.height = 12.0
    return chart


# --- 6. Regression Diagnostics ----------------------------------------------


def _write_qq_helper_columns(ws: Worksheet, header_row: int, residuals: List[float]) -> Dict[str, Optional[int]]:
    """Normal Q-Q 需要理論分位數，既有 Diagnostics 表格沒有這欄；`scipy_stats.probplot`
    回傳的是「依殘差排序後」的資料，不能對齊既有表格的 Run 原始順序，所以獨立寫成
    2 欄（quantile/sorted residual），不跟主表格共用列。另外附一組 2 點擬合虛線範圍。"""
    quantile_col, residual_col = 8, 9  # H, I（G 留空跟 Cook's Distance F 欄隔開）
    fit_col_x, fit_col_y = 11, 12  # K, L
    ws.cell(row=header_row, column=quantile_col, value="Q-Q Theoretical Quantile")
    ws.cell(row=header_row, column=residual_col, value="Q-Q Residual (sorted)")
    n = len(residuals)
    info: Dict[str, Optional[int]] = {
        "quantile_col": quantile_col, "residual_col": residual_col, "last_row": header_row,
        "fit_col_x": None, "fit_col_y": None, "fit_row0": None, "fit_row1": None,
    }
    if n >= 2:
        osm, osr = scipy_stats.probplot(residuals, dist="norm", fit=False)
        for i, (x, y) in enumerate(zip(osm, osr)):
            ws.cell(row=header_row + 1 + i, column=quantile_col, value=float(x))
            ws.cell(row=header_row + 1 + i, column=residual_col, value=float(y))
        slope, intercept, _r = scipy_stats.probplot(residuals, dist="norm", fit=True)[1]
        x0, x1 = float(min(osm)), float(max(osm))
        fit_row0, fit_row1 = header_row + 1, header_row + 2
        # 擬合虛線只需要 2 個點；為了不跟主資料欄的列重疊，另外用 K/L 兩欄放這 2 個點，
        # 列號沿用資料欄最前面 2 列即可（K/L 在資料欄範圍內原本是空的）。
        ws.cell(row=fit_row0, column=fit_col_x, value=x0)
        ws.cell(row=fit_row0, column=fit_col_y, value=intercept + slope * x0)
        ws.cell(row=fit_row1, column=fit_col_x, value=x1)
        ws.cell(row=fit_row1, column=fit_col_y, value=intercept + slope * x1)
        info.update({
            "last_row": header_row + n,
            "fit_col_x": fit_col_x, "fit_col_y": fit_col_y,
            "fit_row0": fit_row0, "fit_row1": fit_row1,
        })
    return info

# --- 7. Prediction Profiler --------------------------------------------------


def _write_prediction_profiler_helper_table(
    ws: Worksheet, start_row: int, regression_result: RegressionResult,
    combination_result: CombinationPredictionResult,
) -> Optional[int]:
    """跟 `_render_prediction_profiler` 一樣，組合被略過/無資料時回傳 None 讓呼叫端跳過。"""
    if combination_result.skipped or not combination_result.rows:
        return None
    best = combination_result.rows[0]
    coefficients = np.array([t.coefficient for t in regression_result.terms], dtype=float)
    specs = combination_result.column_specs
    ws.cell(row=start_row, column=1, value="Column Spec")
    ws.cell(row=start_row, column=2, value="Low")
    ws.cell(row=start_row, column=3, value="High")
    ws.cell(row=start_row, column=4, value="Best Level")
    for i, spec in enumerate(specs):
        r = start_row + 1 + i
        predicted_by_level: Dict[int, float] = {}
        for level in (-1, 1):
            levels = dict(best.coded_levels)
            levels[spec.column] = level
            vector = _model_vector_for_levels(regression_result.terms, levels)
            predicted_by_level[level] = float(vector @ coefficients)
        ws.cell(row=r, column=1, value=spec.display_name)
        ws.cell(row=r, column=2, value=predicted_by_level[-1])
        ws.cell(row=r, column=3, value=predicted_by_level[1])
        best_level = best.coded_levels.get(spec.column, -1)
        ws.cell(row=r, column=4, value="High" if best_level == 1 else "Low")
    return start_row + len(specs)


def _build_prediction_profiler_chart(ws: Worksheet, start_row: int, last_data_row: int, response_name: str) -> LineChart:
    """跟 Main Effect 同樣的「1 圖多 series」版面（X 軸=column spec，2 條 series=
    Low/High 預測值），取代原本 N 個並排 subplot。原本的「最佳水準」垂直虛線標記改用
    `dPt` 把贏家那個資料點標成紅色，取代自既有 helper 表格的 Best Level 欄。"""
    chart = LineChart()
    chart.title = "Prediction Profiler (others fixed at best predicted combination)"
    chart.y_axis.title = f"Predicted {response_name}"
    data = Reference(ws, min_col=2, max_col=3, min_row=start_row, max_row=last_data_row)
    cats = Reference(ws, min_col=1, min_row=start_row + 1, max_row=last_data_row)
    chart.add_data(data, titles_from_data=True)
    chart.set_categories(cats)
    for series, color in zip(chart.series, (_COLOR_BLUE, _COLOR_ORANGE)):
        _style_series(series, color, marker="circle")
    low_points, high_points = [], []
    for i in range(last_data_row - start_row):
        level = ws.cell(row=start_row + 1 + i, column=4).value
        target = low_points if level == "Low" else high_points
        target.append(DataPoint(idx=i, spPr=GraphicalProperties(solidFill=_COLOR_RED)))
    chart.series[0].data_points = low_points
    chart.series[1].data_points = high_points
    n = last_data_row - start_row
    chart.width = max(14.0, 1.3 * n)
    chart.height = 10.0
    return chart

def _pin_doe_sheet_first(wb: Workbook) -> None:
    """確保『DOE』分頁固定顯示在最前面（index 0）並設為開啟檔案時的作用中分頁。

    因子數量調整時 DOE 工作表會被整個刪除重建（`wb.create_sheet` 一律附加在
    目前分頁清單最後面），若不主動搬回最前面，DOE 分頁會被推到「說明」／報告
    分頁之後，使用者要找到它得先滑過一堆分頁；存檔前呼叫一次即可，不管 DOE
    當下在哪個位置都會被移到最前面。
    """

    if DOE_SHEET not in wb.sheetnames:
        return
    current_index = wb.sheetnames.index(DOE_SHEET)
    if current_index != 0:
        wb.move_sheet(DOE_SHEET, offset=-current_index)
    wb.active = wb.sheetnames.index(DOE_SHEET)


def _write_help_sheet(wb: Workbook) -> None:
    """新增（或重建）「說明」工作表。固定會加進範本與分析報告，讓使用者不用離開檔案
    就能查各方法的原理／判讀方式；已存在時先刪除重建，維持 idempotent、不重複累積。
    """

    if "說明" in wb.sheetnames:
        del wb["說明"]
    ws = wb.create_sheet("說明")
    ws.cell(row=1, column=1, value="DOE 分析方法說明").font = TITLE_FONT
    ws.cell(row=2, column=1, value="各分析方法的原理與判讀方式；產生分析報告後，對照報告裡同名的工作表看。")

    r = 4
    ws.cell(
        row=r, column=1,
        value="各方法最小樣本數／組數需求速查表（N＝run 數，q＝獨立設計變數數）",
    ).font = HEADER_FONT
    r += 1
    r = _write_table(ws, r, ["分析方法", "最小需求", "低於門檻時的行為"], MIN_SAMPLE_REQUIREMENTS)
    for row_idx in range(r - len(MIN_SAMPLE_REQUIREMENTS), r):
        for col_idx in (2, 3):
            ws.cell(row=row_idx, column=col_idx).alignment = Alignment(wrap_text=True, vertical="top")
    r += 1

    for title, paragraphs in HELP_SECTIONS:
        cell = ws.cell(row=r, column=1, value=title)
        cell.font = HEADER_FONT
        cell.fill = HEADER_FILL
        r += 1
        for p in paragraphs:
            cell = ws.cell(row=r, column=1, value=p)
            cell.alignment = Alignment(wrap_text=True, vertical="top")
            r += 1
        r += 1  # 段落間留一空白列
    ws.column_dimensions["A"].width = 110
    ws.column_dimensions["B"].width = 45
    ws.column_dimensions["C"].width = 55
    _autofit_row_heights(ws)


def _split_warning_tag(warning: str) -> Tuple[str, str]:
    """把 `doe_core.run_full_analysis` 產生的 "[來源] 訊息" 格式警告拆成
    (來源標籤, 訊息)，方便分開寫進 Summary 分頁的獨立欄位；不是這個格式（沒有
    `[...]` 開頭）就回傳空字串當標籤，訊息維持原樣。
    """

    if warning.startswith("[") and "]" in warning:
        end = warning.index("]")
        return warning[1:end], warning[end + 1:].strip()
    return "", warning


CENTER_ALIGN = Alignment(horizontal="center", vertical="center")


def _write_table(ws: Worksheet, start_row: int, headers: List[str], rows: List[list]) -> int:
    for c, h in enumerate(headers, start=1):
        cell = ws.cell(row=start_row, column=c, value=h)
        cell.font = HEADER_FONT
        cell.fill = HEADER_FILL
        cell.alignment = CENTER_ALIGN
    r = start_row + 1
    for row_vals in rows:
        for c, v in enumerate(row_vals, start=1):
            cell = ws.cell(row=r, column=c, value=v)
            cell.alignment = CENTER_ALIGN
        r += 1
    return r


def _display_width(text: str) -> int:
    """估算字串顯示寬度：全形字元（中日韓）近似算 2 個半形字元寬，其餘算 1 個，
    用來讓 autofit 欄寬對中文內容也大致準確（openpyxl 沒有原生 autofit，欄寬單位
    本來就是「半形字元數」的近似值）。"""

    return sum(2 if ord(ch) > 0x2E80 else 1 for ch in text)


def _autofit_columns(ws: Worksheet, min_width: float = 8.0, max_width: float = 60.0, padding: float = 2.0) -> None:
    """依目前工作表已寫入的內容，自動調整每欄欄寬（近似 Excel 的 autofit：抓每欄
    最長那行文字的顯示寬度＋padding）。刻意跳過 `wrap_text` 的儲存格——那些是設計
    成在固定欄寬內換行顯示的長文字（警告/建議/說明文字），不該讓欄寬被撐開，不然
    反而失去換行的意義；這類儲存格的欄寬維持呼叫端另外設定的值。"""

    widths: Dict[int, float] = {}
    for row in ws.iter_rows():
        for cell in row:
            if cell.value is None:
                continue
            if cell.alignment and cell.alignment.wrap_text:
                continue
            longest_line = max(str(cell.value).splitlines() or [""], key=_display_width)
            widths[cell.column] = max(widths.get(cell.column, 0), _display_width(longest_line))
    for col_idx, w in widths.items():
        ws.column_dimensions[get_column_letter(col_idx)].width = min(max(w + padding, min_width), max_width)


def _autofit_row_heights(ws: Worksheet, line_height: float = 15.0) -> None:
    """依 `wrap_text` 儲存格換算後的欄寬，估算需要幾行、設定對應的列高（openpyxl
    沒有原生 autofit，用「顯示寬度 / 欄寬」概算行數 × 單行高度近似）。沒有
    wrap_text 的列不處理，維持 Excel 預設的單行列高（本來就已經是自動的）。"""

    for row in ws.iter_rows():
        max_lines = 1
        for cell in row:
            if cell.value is None or not (cell.alignment and cell.alignment.wrap_text):
                continue
            col_width = ws.column_dimensions[cell.column_letter].width or 8.43
            usable_width = max(int(col_width) - 2, 1)
            for line in str(cell.value).splitlines() or [""]:
                lines_needed = -(-_display_width(line) // usable_width)  # 無條件進位
                max_lines = max(max_lines, lines_needed)
        if max_lines > 1:
            ws.row_dimensions[row[0].row].height = max_lines * line_height


def _center_align_all_cells(ws: Worksheet) -> None:
    """把整張工作表目前有值的儲存格全部改成水平置中對齊（含說明文字／警告訊息這
    類本來只有靠左對齊的散文字儲存格，使用者要求所有分頁全部置中，不只表格）。
    只改 `horizontal`，保留原本已經設定的 `vertical`／`wrap_text`（沒有另外設定
    過的預設回退成垂直置中），不會把長文字儲存格的換行設定洗掉。"""

    for row in ws.iter_rows():
        for cell in row:
            if cell.value is None:
                continue
            existing = cell.alignment
            cell.alignment = Alignment(
                horizontal="center",
                vertical=existing.vertical or "center",
                wrap_text=bool(existing.wrap_text),
            )


def _apply_title_only_center_alignment(ws: Worksheet) -> None:
    """「說明」分頁跟 Summary 分頁不套用其他分頁「全部置中」的通用規則：只有標題
    （大標題／各段落小標題／表格表頭，這幾種都用粗體字呈現）置中，其餘內容（說明
    文字、警告/建議散文字、段落本文、表格資料列）一律靠左對齊——這兩張分頁是給人
    閱讀的長篇文字／條列內容，本文置中反而比較難讀。用 `cell.font.bold` 判斷是不
    是標題：`_write_help_sheet()`／`write_report()` 寫 Summary 時都只有標題／表頭
    會設定粗體字型，本文／資料列都是預設字型，剛好可以拿來當判斷依據，不用另外
    傳參數標記哪些是標題。"""

    for row in ws.iter_rows():
        for cell in row:
            if cell.value is None:
                continue
            existing = cell.alignment
            is_title = bool(cell.font and cell.font.bold)
            cell.alignment = Alignment(
                horizontal="center" if is_title else "left",
                vertical=existing.vertical or "center",
                wrap_text=bool(existing.wrap_text),
            )


def _align_cells_after_write(wb: Workbook) -> None:
    """存檔前對整份活頁簿套用置中規則：大部分分頁全部置中，「說明」分頁跟
    Summary_<溫度> 分頁例外（只有標題置中，其餘靠左，見
    `_apply_title_only_center_alignment` 的說明）。四個會呼叫 `wb.save()` 的地方
    （新建範本／因子數量未變／調整因子數量／執行分析）都要呼叫這個函式，不要各
    自重複判斷要不要跳過這兩張分頁。Summary 分頁名稱帶溫度後綴（例如
    `Summary_25`），用 `split("_", 1)[0]` 取 base 名稱比對，跟
    `_delete_existing_report_sheets` 判斷報告分頁的方式一致。"""

    for ws in wb.worksheets:
        base_name = ws.title.split("_", 1)[0]
        if ws.title == "說明" or base_name == "Summary":
            _apply_title_only_center_alignment(ws)
        else:
            _center_align_all_cells(ws)


REPORT_SHEET_NAMES = (
    "Summary", "MainEffect", "ANOVA", "Regression", "Effects", "Diagnostics", "Alias",
    "Optimization", "Correlation", "Taguchi", "Pareto", "Interaction",
)


def _temp_suffix(temperature: float) -> str:
    """把分析溫度轉成可以放進 Excel 分頁名稱的字串（整數溫度不帶小數點）。"""

    t = float(temperature)
    return str(int(t)) if t.is_integer() else str(t)


def _report_sheet_name(base: str, temp_suffix: str) -> str:
    """報告分頁名稱＝`{base}_{溫度}`，讓分頁列表不用點開就能看出是哪個溫度的報告
    （例如 `Summary_25`）。"""

    return f"{base}_{temp_suffix}"


def _delete_existing_report_sheets(wb: Workbook) -> None:
    """刪除所有既有的報告分頁，不論是哪個溫度產生的（`{base}_{任何後綴}`），也相容
    改版前沒有溫度後綴的舊檔案（分頁名稱剛好等於 `base`）。每次分析都要整批清掉、
    重新產生，才不會換溫度重跑時把舊溫度的分頁留著，變成分頁列表一直累積
    （`Summary_25`／`Summary_75`都留著）。
    """

    report_bases = set(REPORT_SHEET_NAMES)
    for existing_name in list(wb.sheetnames):
        base = existing_name.split("_", 1)[0]
        if existing_name in report_bases or base in report_bases:
            del wb[existing_name]


def _factor_level_display(factor: Factor, code: int) -> str:
    if code == -1:
        value = factor.level_low_label or factor.level_low_value
        level = "Low"
    else:
        value = factor.level_high_label or factor.level_high_value
        level = "High"
    return f"{level}: {value}" if value is not None and str(value) != "" else level


def _spec_level_display(design: DesignConfig, spec: DesignColumnSpec, code: int) -> str:
    factors = [f for f in design.factors if f.factor_id in spec.factor_ids]
    if len(factors) == 1:
        return _factor_level_display(factors[0], code)
    details = "; ".join(f"{f.name}={_factor_level_display(f, code).split(': ', 1)[-1]}" for f in factors)
    return ("Low" if code == -1 else "High") + f": {details}"

def _write_explanation_block(ws: Worksheet, start_row: int, title: str, paragraphs: List[str]) -> int:
    """在分頁最前面寫一段白話解讀：粗體大標題＋「這組數字怎麼讀」小標題＋逐段
    wrap_text 文字，比照 `_write_help_sheet()` 段落寫法，不寫死列高（呼叫端存檔前
    會統一呼叫 `_autofit_row_heights()` 依實際內容算高度）。回傳解讀區塊後面第一個
    可用列號，供呼叫端接著放空白列／資料表。"""
    ws.cell(row=start_row, column=1, value=title).font = TITLE_FONT
    r = start_row + 1
    cell = ws.cell(row=r, column=1, value="這組數字怎麼讀")
    cell.font = HEADER_FONT
    cell.fill = HEADER_FILL
    r += 1
    for p in paragraphs:
        cell = ws.cell(row=r, column=1, value=p)
        cell.alignment = Alignment(wrap_text=True, vertical="top")
        r += 1
    r += 1  # 空白列分隔解讀文字跟資料表
    return r


def _explain_main_effect(design: DesignConfig, rows: List[MainEffectRow]) -> List[str]:
    if not rows:
        return ["目前沒有因子資料可以解讀。"]
    lines = [
        "每一列是一個因子：Mean(Low)/Mean(High) 是這個因子分別維持在 Low／High 水準時，"
        f"其餘因子平均掉之後的平均 {design.response_name}；Effect＝Mean(High)−Mean(Low)，"
        "數字絕對值越大代表這個因子影響越大，正值代表 High 水準時數值比較大、負值代表 "
        "Low 水準時比較大。",
    ]
    ranked = sorted(rows, key=lambda r: abs(r.effect), reverse=True)
    top = ranked[0]
    direction = "High" if top.effect > 0 else "Low"
    better = "較大" if top.effect > 0 else "較小"
    lines.append(
        f"目前影響最大的是「{top.name or top.factor_id}」（Effect={top.effect:.4g}）："
        f"{direction} 水準時 {design.response_name} 平均{better}，是優先該調整的因子。"
    )
    if len(ranked) > 1:
        second = ranked[1]
        lines.append(f"其次是「{second.name or second.factor_id}」（Effect={second.effect:.4g}）。")
    return lines


def _explain_anova(anova_result: AnovaResult, design_rows: List[AnovaRow]) -> List[str]:
    """`design_rows` 要傳跟報告表格同一份（`result.anova.design_rows`，同一 group
    只算一次），不是 `anova_result.rows`（那份是每個原始因子列各自一份、材料 group
    會重複），避免解讀文字跟表格數字對不上。"""
    rows = design_rows
    if not rows:
        return ["目前沒有因子資料可以解讀。"]
    lines = [
        "SS 是這個因子解釋掉的變異量，DF 是自由度，MS=SS/DF；F＝這個因子的 MS 除以 Error MS"
        "（雜訊基準線），數字越大代表訊號相對雜訊越明顯；p 是「假設這個因子其實沒有效應、"
        "純粹靠運氣湊出這麼大 SS」的機率，習慣上 p<0.05 視為顯著；%Contribution 是這個因子"
        "佔全部變異的比例，排序更直覺、不受顯著性門檻影響。",
    ]
    if anova_result.df_error <= 0:
        lines.append(
            "這次分析 Error DF＝0（飽和設計，因子用滿了正交表、沒有剩餘欄位可以估計雜訊），"
            "所以 F/p 沒有算、無法做顯著性檢定，只能看 SS／%Contribution 排序參考。"
        )
        for r in sorted(rows, key=lambda x: -x.pct_contribution)[:3]:
            lines.append(f"「{r.name or r.factor_id}」貢獻度 {r.pct_contribution:.2f}%。")
        return lines

    has_not_sig = False
    for r in sorted(rows, key=lambda x: -x.pct_contribution):
        if r.p_value is None:
            continue
        significant = r.p_value < 0.05
        has_not_sig = has_not_sig or not significant
        status = "顯著" if significant else "不顯著"
        lines.append(f"「{r.name or r.factor_id}」：p={r.p_value:.4g}（{status}）、貢獻度 {r.pct_contribution:.2f}%。")
    if has_not_sig:
        lines.append(
            "p 不顯著不代表這個因子完全沒效果，是目前樣本數／Error DF 太小，雜訊的波動範圍"
            "蓋過了訊號，沒辦法有信心把它跟純粹運氣分開——建議之後用重複量測或確認實驗加大"
            "樣本再確認。"
        )
    return lines


def _explain_regression(reg: RegressionResult) -> List[str]:
    lines = [
        f"R²={reg.r_squared:.4f} 代表這個模型解釋了 {reg.r_squared * 100:.1f}% 的變異，數字越接近 1 "
        "代表模型解釋力越好；係數是每個項目（主效應／交互作用）在 coded ±1 尺度下對響應值的"
        "影響，t／p 用來判斷這個係數是不是顯著跟雜訊不同。",
    ]
    if reg.residual_df <= 0:
        lines.append(
            "這次殘差自由度（residual_df）＝0，因子用滿了模型自由度、沒有剩餘資料可以估計"
            "不確定性，所以 Std. Error／t／p／95% CI 這些欄位都是 N/A，不是漏算——只能看係數"
            "本身的大小跟方向，無法做正式的顯著性檢定。"
        )
        return lines
    sig_terms = [t for t in reg.terms if t.kind != "intercept" and t.p_value is not None and t.p_value < 0.05]
    if sig_terms:
        names = "、".join(f"{t.label}（p={t.p_value:.4g}）" for t in sig_terms)
        lines.append(f"目前判定為顯著的項目：{names}。")
    else:
        lines.append("目前沒有項目的 p 值小於 0.05——樣本數偏少時常見，數字仍可參考係數大小排序。")
    if reg.predicted_r_squared is not None:
        lines.append(
            f"Predicted R²={reg.predicted_r_squared:.4f} 是用 LOOCV（輪流拿掉一筆資料重新配適）"
            "估計出來對「新資料」的預測力，如果跟 R² 差很多，代表模型可能過度配適這批資料。"
        )
    return lines


def _explain_effects(result: EffectScreeningResult) -> List[str]:
    lines = [
        f"這裡用「{result.method}」篩選哪些效應是 active（顯著／值得注意），圖表把效應依大小"
        "排序，紅色長條就是被判定 active 的項目。",
    ]
    if result.method != "Regression t":
        lines.append(
            "因為飽和設計沒有殘差自由度可以做正式檢定，這裡改用 Lenth PSE（一種從效應本身"
            "估計雜訊水準的方法）當篩選工具——結果可以拿來排優先順序，但不是正式的統計顯著性"
            "檢定，仍建議用確認實驗驗證。"
        )
    active = [r for r in result.rows if r.active]
    if active:
        lines.append(f"目前判定為 active 的效應：{'、'.join(r.label for r in active)}。")
    else:
        lines.append("目前沒有效應被判定為 active，可能是效應都不大，或樣本數不足以分辨。")
    return lines


def _explain_diagnostics(regression_result: RegressionResult) -> List[str]:
    lines = [
        "六張圖都是在檢查迴歸模型有沒有問題：Observed vs Fitted 越貼近對角線代表模型預測越"
        "準；Residuals vs Fitted／Run Order 應該隨機散布在 0 上下、沒有明顯規律，如果看到"
        "喇叭狀（變異隨數值變大）或有趨勢，代表模型型式可能不對；Normal Q-Q 的點應該大致"
        "貼近參考線，偏離太多代表殘差不太像常態分布；Leverage 高的 run 代表它的因子組合在"
        "設計空間裡比較極端、對模型影響力較大；Cook's Distance 特別大的 run 值得回頭檢查"
        "那筆量測是不是量錯了或製程當天有異常。",
    ]
    if regression_result.residual_df <= 0:
        lines.append(
            "這次是飽和設計（沒有殘差自由度），六張圖只能當作「資料點之間怎麼比較」的參考，"
            "沒有統計上的驗證能力（模型完美貼合了每一筆資料，沒辦法判斷模型好不好）。"
        )
    return lines


def _explain_alias(alias_result: AliasInteractionResult) -> List[str]:
    lines = [
        "「混淆／別名（confounded/aliased）」是說這個交互作用跟另一個效應共用同一個正交表"
        "欄位，這批資料沒辦法把兩者的貢獻分開——只有 Status 顯示「可單獨估計」的交互作用才能"
        "當成獨立的效應解讀，其餘的都只是同一個合併 contrast 的其中一種可能來源。",
    ]
    confounded = [r for r in alias_result.rows if r.status != "可單獨估計"]
    if confounded:
        names = "、".join(r.interaction for r in confounded[:5])
        lines.append(f"這次有 {len(confounded)} 組交互作用是混淆／別名狀態（例如 {names}），不能單獨解讀。")
    else:
        lines.append("這次所有交互作用都可以單獨估計，沒有混淆的問題。")
    return lines


def _explain_optimization(design: DesignConfig, combo_result: CombinationPredictionResult) -> List[str]:
    if combo_result.skipped:
        return [f"這個分頁被略過：{combo_result.skip_reason}"]
    if not combo_result.rows:
        return ["目前沒有可列舉的組合。"]
    best = combo_result.rows[0]
    lines = [
        f"第一列（Rank=1）是目前模型預測最好的因子組合，Predicted {design.response_name}="
        f"{best.predicted:.4g}；「Existing Run」欄標示這組組合有沒有在這批資料裡實際跑過，"
        "「Confirmation Candidate」標示模型建議下一組去實際驗證的組合。",
        "這是模型預測出來的最佳解，不是實際量到的數字，模型可能有沒考慮到的因素（例如"
        "Low/High 中間的區域可能更好）——建議實際做一次確認實驗驗證模型預測準不準，再決定"
        "要不要照這組設定調整製程。",
    ]
    if combo_result.confirmation_row is not None:
        lines.append("目前有標示一組尚未實跑過的確認組合，可以優先安排這組實驗。")
    else:
        lines.append("目前所有可列舉的組合都已經在既有資料裡跑過，沒有額外需要新增的確認實驗。")
    return lines


def _explain_correlation(result: CorrelationResult) -> List[str]:
    lines = [
        "這是 factor×factor×response 的 Pearson 相關係數矩陣。因為是正交設計，理論上任兩個"
        "因子之間的相關係數應該接近 0（同一個材料 group 內的因子例外，理論上必為 1，不算"
        "異常）；如果看到不該相關的兩個因子出現偏高的相關係數，通常代表 leg 資料填錯位置或"
        "被覆寫過，不是真的因子之間有關聯。",
    ]
    if result.flagged_pairs:
        names = "、".join(f"{a}×{b}（r={v:.2f}）" for a, b, v in result.flagged_pairs[:5])
        lines.append(f"這次超過門檻被標示的組合：{names}，建議回頭檢查 DOE 工作表對應欄位。")
    else:
        lines.append("這次沒有超過門檻的異常相關，資料排列跟正交表預期一致。")
    return lines


def _explain_taguchi(result: TaguchiResult) -> List[str]:
    if result.skipped:
        return [f"這個分頁被略過：{result.skip_reason}"]
    if not result.rows:
        return ["目前沒有因子資料可以解讀。"]
    direction_label = SN_DIRECTION_LABELS.get(result.direction, str(result.direction))
    lines = [
        f"目前的 S/N 最佳化方向是「{direction_label}」。SN(Low)/SN(High) 是這個因子在 Low/High"
        "水準下的訊噪比（同時考慮平均值跟穩定度的指標），Delta=|SN(High)−SN(Low)|，Delta 越大"
        "代表這個因子對訊噪比的影響越大，Rank 就是照 Delta 大小排序。",
    ]
    top = min(result.rows, key=lambda r: r.rank)
    lines.append(f"排名第一的是「{top.name or top.factor_id}」（Delta={top.delta:.4g}），對訊噪比影響最大。")
    return lines


def _explain_pareto(pareto_bars: List[ParetoBar]) -> List[str]:
    if not pareto_bars:
        return ["目前沒有因子資料可以解讀。"]
    lines = [
        "長條圖把每個因子的 |Effect|（效應大小，不分正負）由大到小排序，長條越長代表這個"
        "因子對響應值的影響越大，是優先該調整的目標——但這裡看不出方向（該選 Low 還是 "
        "High），方向要對照 MainEffect 分頁。",
    ]
    top = pareto_bars[0]
    lines.append(f"目前排第一的是「{top.name or top.factor_id}」（|Effect|={top.magnitude:.4g}）。")
    return lines


def _explain_interaction() -> List[str]:
    return [
        "每張圖是一對因子的交互作用圖：兩條線（分別代表另一個因子 Low/High 水準）越平行，"
        "代表這兩個因子交互作用越小，可以個別獨立調整；如果兩條線明顯交叉或斜率差很多，"
        "代表其中一個因子的最佳水準會因為另一個因子而改變，不能只看各自的主效應決定設定值。",
        "標題如果註明「confounded with」或「alias class」，代表這組交互作用跟其他效應混淆，"
        "圖形趨勢可以參考，但不能斷定這條趨勢完全來自這兩個因子的交互作用，細節見 Alias 分頁。",
    ]


def write_report(
    path: str,
    design: DesignConfig,
    oa: OrthogonalArray,
    run_matrix: RunMatrix,
    result: FullAnalysisResult,
    recommendations: List[Recommendation],
    top_interaction_pairs: int = 3,
    baseline_compare: Optional[BaselineCompareResult] = None,
) -> None:
    """把分析報告（`REPORT_SHEET_NAMES` 工作表＋「說明」）寫進同一份 DOE 來源
    檔案（`path`），不再另外產生一份 report .xlsx——分析結果跟填寫的原始資料放在
    一起，開一個檔案就能對照。分頁名稱帶有分析溫度（例如 `Summary_25`），不用點開
    就能從分頁列表看出是哪個溫度的報告。每次執行分析都會先刪除舊的報告工作表
    （不論是哪個溫度產生的）再重建（idempotent，重跑不會累積重複分頁；換一個溫度
    重新分析會覆蓋掉上一次的報告內容，不會兩個溫度的分頁同時留著），DOE 工作表
    本身不會被動到。
    """

    wb = load_workbook(path)
    _delete_existing_report_sheets(wb)
    temp_suffix = _temp_suffix(result.temperature)

    ws_summary = wb.create_sheet(_report_sheet_name("Summary", temp_suffix))

    ws_summary.cell(row=1, column=1, value=f"DOE 分析報告 — {design.response_name} @ {result.temperature}").font = TITLE_FONT
    ws_summary.cell(
        row=2, column=1,
        value=(
            f"因子列數={design.k}，獨立設計變數 q={len(design_column_specs(design))}，"
            f"run 數(leg數量) N={oa.N}"
        ),
    )
    ws_summary.cell(row=3, column=1, value=f"S/N 最佳化方向：{SN_DIRECTION_LABELS[design.sn_direction]}")

    r = 5
    ws_summary.cell(row=r, column=1, value="樣本數／自由度／完整性警告").font = HEADER_FONT
    r += 1
    if result.warnings:
        ws_summary.cell(row=r, column=1, value="來源").font = HEADER_FONT
        ws_summary.cell(row=r, column=2, value="警告內容").font = HEADER_FONT
        r += 1
        for w in result.warnings:
            tag, message = _split_warning_tag(w)
            ws_summary.cell(row=r, column=1, value=tag)
            ws_summary.cell(row=r, column=2, value=f"⚠ {message}")
            ws_summary.cell(row=r, column=2).alignment = Alignment(wrap_text=True)
            r += 1
    else:
        ws_summary.cell(row=r, column=1, value="（無）")
        r += 1

    r += 1
    ws_summary.cell(row=r, column=1, value="建議（依敏感度排名前幾名因子）").font = HEADER_FONT
    r += 1
    if recommendations:
        for rec in recommendations:
            ws_summary.cell(
                row=r, column=1,
                value=f"{rec.factor_id}（{rec.name}）建議選 {rec.recommended_level} 水準（{rec.recommended_label}）— {rec.rationale}",
            )
            ws_summary.cell(row=r, column=1).alignment = Alignment(wrap_text=True)
            r += 1
    else:
        ws_summary.cell(row=r, column=1, value="（無）")
        r += 1

    r += 1
    ws_summary.cell(row=r, column=1, value="敏感度綜合排名").font = HEADER_FONT
    r += 1
    sens_headers = ["Rank", "因子代號", "因子名稱", "Score", "組成分量"]
    sens_rows = [
        [row.rank, row.factor_id, row.name, round(row.score, 4),
         ", ".join(f"{k}={v:.3f}" for k, v in row.components.items())]
        for row in result.sensitivity.rows
    ]
    r = _write_table(ws_summary, r, sens_headers, sens_rows)

    r += 1
    ws_summary.cell(row=r, column=1, value="模型品質與效應篩選").font = HEADER_FONT
    r += 1
    quality_rows = [
        ["R²", result.regression.r_squared],
        ["Adjusted R²", result.regression.adjusted_r_squared if result.regression.adjusted_r_squared is not None else "N/A"],
        ["Predicted R² (LOOCV)", result.regression.predicted_r_squared if result.regression.predicted_r_squared is not None else "N/A"],
        ["RMSE", result.regression.rmse if result.regression.rmse is not None else "N/A"],
        ["殘差自由度", result.regression.residual_df],
        ["效應篩選方法", result.effect_screening.method],
    ]
    r = _write_table(ws_summary, r, ["指標", "值"], quality_rows)

    if not result.combinations.skipped and result.combinations.rows:
        best_combo = result.combinations.rows[0]
        r += 1
        ws_summary.cell(row=r, column=1, value="模型預測最佳組合").font = HEADER_FONT
        r += 1
        combo_text = "; ".join(
            f"{spec.label}={_spec_level_display(design, spec, best_combo.coded_levels[spec.column])}"
            for spec in result.combinations.column_specs
        )
        observed_text = ", ".join(best_combo.observed_run_labels) or "未實跑"
        ws_summary.cell(
            row=r, column=1,
            value=f"{combo_text}；預測 {design.response_name}={best_combo.predicted:.6g}；現有 run={observed_text}",
        )
        ws_summary.cell(row=r, column=1).alignment = Alignment(wrap_text=True)
        r += 1
        if result.combinations.confirmation_row is not None:
            confirm = result.combinations.confirmation_row
            confirm_text = "; ".join(
                f"{spec.label}={_spec_level_display(design, spec, confirm.coded_levels[spec.column])}"
                for spec in result.combinations.column_specs
            )
            ws_summary.cell(
                row=r, column=1,
                value=f"建議 confirmation run：{confirm_text}；預測值={confirm.predicted:.6g}",
            )
            ws_summary.cell(row=r, column=1).alignment = Alignment(wrap_text=True)
            r += 1
    elif result.combinations.skipped:
        r += 1
        ws_summary.cell(row=r, column=1, value=f"最佳組合預測已略過：{result.combinations.skip_reason}")
        r += 1

    r += 1
    ws_summary.cell(
        row=r, column=1,
        value=f"依 leg（run）排序（{SN_DIRECTION_LABELS[design.sn_direction]}，rank 1 最好）",
    ).font = HEADER_FONT
    r += 1
    leg_headers = ["Rank", "Leg", f"{design.response_name}"]
    leg_rows = [[row.rank, row.run_label, row.response_value] for row in result.leg_ranking]
    r = _write_table(ws_summary, r, leg_headers, leg_rows)

    if baseline_compare is not None and baseline_compare.enabled:
        r += 1
        ws_summary.cell(row=r, column=1, value="與 baseline 比較").font = HEADER_FONT
        r += 1
        if baseline_compare.skipped:
            ws_summary.cell(row=r, column=1, value=f"已略過：{baseline_compare.skip_reason}")
            ws_summary.cell(row=r, column=1).alignment = Alignment(wrap_text=True)
            r += 1
        else:
            ws_summary.cell(
                row=r, column=1,
                value=(
                    f"基準 leg：{baseline_compare.baseline_run_label}"
                    f"（{design.response_name}={baseline_compare.baseline_response:.4g}）；"
                    "改善% 已依 S/N 方向正規化，正值＝優於基準、負值＝劣於基準。"
                ),
            )
            ws_summary.cell(row=r, column=1).alignment = Alignment(wrap_text=True)
            r += 1
            for w in baseline_compare.warnings:
                ws_summary.cell(row=r, column=1, value=f"⚠ {w}")
                ws_summary.cell(row=r, column=1).alignment = Alignment(wrap_text=True)
                r += 1

            cmp_headers = ["Rank", "候選 Leg", design.response_name, "與基準差值", "改善%"]
            cmp_rows = [
                [
                    row.rank, row.run_label, row.response_value, round(row.response_delta, 4),
                    round(row.improvement_pct, 2) if row.improvement_pct is not None else "N/A",
                ]
                for row in baseline_compare.rows
            ]
            r = _write_table(ws_summary, r, cmp_headers, cmp_rows)

            best = baseline_compare.rows[0]
            r += 1
            ws_summary.cell(
                row=r, column=1, value=f"改善幅度最大的候選 leg（{best.run_label}）因子調整方向",
            ).font = HEADER_FONT
            r += 1
            fa_headers = ["因子代號", "因子名稱", "基準值", "候選值", "變化%", "調整方向"]
            fa_rows = [
                [
                    fa.factor_id, fa.name, fa.baseline_value, fa.leg_value,
                    round(fa.pct_change, 2) if fa.pct_change is not None else "N/A", fa.direction,
                ]
                for fa in best.factor_adjustments
            ]
            r = _write_table(ws_summary, r, fa_headers, fa_rows)

    _autofit_columns(ws_summary)
    # A/B 欄混雜了警告/建議等 wrap_text 長文字，autofit 只看非 wrap_text 儲存格
    # 會把欄寬抓得太窄，這裡固定給一個閱讀舒適的下限，蓋掉 autofit 的結果。
    ws_summary.column_dimensions["A"].width = 90
    ws_summary.column_dimensions["B"].width = 90
    _autofit_row_heights(ws_summary)

    _write_help_sheet(wb)

    # Main Effect
    ws_me = wb.create_sheet(_report_sheet_name("MainEffect", temp_suffix))
    report_main_effects = main_effects_by_design_column(design, result.main_effects)
    me_start = _write_explanation_block(
        ws_me, 1, "Main Effect — 各因子平均效應", _explain_main_effect(design, report_main_effects),
    )
    me_headers = ["因子代號", "因子名稱", "Mean(Low)", "Mean(High)", "Effect"]
    me_rows = [
        [row.factor_id, row.name, row.mean_low, row.mean_high, row.effect]
        for row in report_main_effects
    ]
    next_row = _write_table(ws_me, me_start, me_headers, me_rows)
    _autofit_columns(ws_me)
    if report_main_effects:
        me_charts = _build_main_effect_charts(ws_me, me_start, next_row - 1, design.response_name)
        for chart, col_letter in me_charts:
            _add_chart_fixed(ws_me, chart, f"{col_letter}{next_row + 2}")
        combined_chart = _build_main_effect_combined_chart(ws_me, me_start, next_row - 1, design.response_name)
        _add_chart_fixed(ws_me, combined_chart, f"A{next_row + 21}")

    # ANOVA
    ws_an = wb.create_sheet(_report_sheet_name("ANOVA", temp_suffix))
    an_start = _write_explanation_block(
        ws_an, 1, "ANOVA — 變異數分析", _explain_anova(result.anova, result.anova.design_rows),
    )
    an_headers = ["設計變數／因子代號", "因子名稱", "Table Column", "SS", "DF", "MS", "F", "p", "%Contribution"]
    an_rows = [
        [
            row.factor_id, row.name, row.table_column, row.ss, row.df, row.ms,
            row.f_value, row.p_value, round(row.pct_contribution, 2),
        ]
        for row in result.anova.design_rows
    ]
    next_row = _write_table(ws_an, an_start, an_headers, an_rows)
    ws_an.cell(row=next_row, column=1, value=f"Model: SS={result.anova.ss_model:.6g}, DF={len(result.anova.design_rows)}")
    ws_an.cell(row=next_row + 1, column=1, value=f"Error: SS={result.anova.ss_error:.6g}, DF={result.anova.df_error}, MS={result.anova.ms_error}")
    ws_an.cell(row=next_row + 2, column=1, value=f"Total: SS={result.anova.ss_total:.6g}, DF={result.anova.df_total}")
    _autofit_columns(ws_an)

    # Regression
    ws_reg = wb.create_sheet(_report_sheet_name("Regression", temp_suffix))
    reg_start = _write_explanation_block(
        ws_reg, 1, "Regression — 完整迴歸統計", _explain_regression(result.regression),
    )
    reg_headers = ["Term", "Kind", "Coefficient", "Std. Error", "t", "p", "95% CI Low", "95% CI High"]
    reg_rows = [
        [
            t.label, t.kind, t.coefficient, t.std_error, t.t_value, t.p_value,
            t.ci_low, t.ci_high,
        ]
        for t in result.regression.terms
    ]
    next_row = _write_table(ws_reg, reg_start, reg_headers, reg_rows)
    next_row += 1
    ws_reg.cell(
        row=next_row, column=1,
        value=(
            f"R²={result.regression.r_squared:.4f}；Adjusted R²="
            f"{result.regression.adjusted_r_squared if result.regression.adjusted_r_squared is not None else 'N/A'}；"
            f"Predicted R²={result.regression.predicted_r_squared if result.regression.predicted_r_squared is not None else 'N/A'}；"
            f"RMSE={result.regression.rmse if result.regression.rmse is not None else 'N/A'}；"
            f"PRESS={result.regression.press if result.regression.press is not None else 'N/A'}；"
            f"殘差自由度={result.regression.residual_df}"
        ),
    )
    next_row += 1
    physical = regression_physical_coefficients(design, result.regression)
    if physical:
        next_row += 1
        ws_reg.cell(
            row=next_row, column=1,
            value="物理量尺度係數（僅單獨佔欄的 numeric 因子；材料 group 無法分離 per-unit 斜率）",
        ).font = HEADER_FONT
        next_row += 1
        next_row = _write_table(ws_reg, next_row, ["因子代號", "係數 (per unit)"], [[k, v] for k, v in physical.items()])
    if result.regression.excluded_terms:
        next_row += 1
        ws_reg.cell(row=next_row, column=1, value="已排除的交互作用項").font = HEADER_FONT
        next_row += 1
        _write_table(ws_reg, next_row, ["Term", "排除原因"], [[lab, reason] for lab, reason in result.regression.excluded_terms])
    _autofit_columns(ws_reg)

    # Standardized effects / Lenth PSE
    ws_eff = wb.create_sheet(_report_sheet_name("Effects", temp_suffix))
    eff_start = _write_explanation_block(
        ws_eff, 1, "Effects — 標準化效應／Lenth PSE", _explain_effects(result.effect_screening),
    )
    ws_eff.cell(row=eff_start, column=1, value=f"效應篩選方法：{result.effect_screening.method}").font = TITLE_FONT
    ws_eff.cell(row=eff_start + 1, column=1, value=f"Lenth PSE：{result.effect_screening.lenth_pse if result.effect_screening.lenth_pse is not None else 'N/A'}")
    if result.effect_screening.critical_value is not None:
        ws_eff.cell(row=eff_start + 2, column=1, value=f"95% threshold ≈ {result.effect_screening.critical_value:.4g}")
    eff_table_row = eff_start + 3
    eff_headers = ["Term", "Name", "Kind", "Effect", "Std. Error/PSE", "t/Pseudo t", "p/Pseudo p", "95% CI Low", "95% CI High", "Active (0.05)", "Method"]
    eff_rows = [
        [
            row.label, row.name, row.kind, row.effect, row.standard_error, row.statistic,
            row.p_value, row.ci_low, row.ci_high,
            "Yes" if row.active else ("No" if row.active is not None else "N/A"), row.method,
        ]
        for row in result.effect_screening.rows
    ]
    next_row = _write_table(ws_eff, eff_table_row, eff_headers, eff_rows)
    if result.effect_screening.rows:
        _write_effects_abs_column(ws_eff, eff_table_row, next_row - 1, 6, 13)
        eff_chart = _build_standardized_effects_chart(ws_eff, eff_table_row, next_row - 1, 13, result.effect_screening)
        _add_chart_fixed(ws_eff, eff_chart, f"A{next_row + 2}")
        hn_info = _write_half_normal_helper_table(ws_eff, eff_table_row, 16, result.effect_screening)
        hn_chart = _build_half_normal_chart(ws_eff, eff_table_row, hn_info)
        _add_chart_fixed(ws_eff, hn_chart, f"J{next_row + 2}")
    _autofit_columns(ws_eff)

    # Regression diagnostics
    ws_diag = wb.create_sheet(_report_sheet_name("Diagnostics", temp_suffix))
    diag_start = _write_explanation_block(
        ws_diag, 1, "Diagnostics — 迴歸診斷", _explain_diagnostics(result.regression),
    )
    diag_headers = ["Run", "Observed", "Fitted", "Residual", "Leverage", "Cook's Distance"]
    diag_rows = [
        [
            label,
            result.regression.observed_values[i],
            result.regression.fitted_values[i],
            result.regression.residuals[i],
            result.regression.leverage[i] if i < len(result.regression.leverage) else None,
            result.regression.cooks_distance[i] if i < len(result.regression.cooks_distance) else None,
        ]
        for i, label in enumerate(run_matrix.run_labels)
    ]
    next_row = _write_table(ws_diag, diag_start, diag_headers, diag_rows)
    if diag_rows:
        qq_info = _write_qq_helper_columns(ws_diag, diag_start, result.regression.residuals)
        if len(diag_rows):
            ws_diag.cell(row=next_row, column=1, value=f"Cook's distance threshold (4/N) ≈ {4.0 / len(diag_rows):.4g}")
        diag_charts = _build_regression_diagnostics_charts(ws_diag, diag_start, next_row - 1, qq_info)
        for chart, col_letter, grid_row in diag_charts:
            _add_chart_fixed(ws_diag, chart, f"{col_letter}{next_row + 2 + grid_row * 17}")
    _autofit_columns(ws_diag)

    # Alias / interaction effects
    ws_alias = wb.create_sheet(_report_sheet_name("Alias", temp_suffix))
    alias_start = _write_explanation_block(
        ws_alias, 1, "Alias — 混淆／別名結構", _explain_alias(result.alias_interactions),
    )
    alias_headers = [
        "Interaction", "Factor A", "Factor B", "Interaction Column", "Status",
        "Aliased With", "Effect", "SS", "Error DF (excluding tested column)",
        "Error MS", "F", "p",
    ]
    alias_rows = [
        [
            row.interaction, row.factor_a, row.factor_b, row.interaction_column,
            row.status, row.aliased_with, row.effect, row.ss, row.error_df,
            row.error_ms, row.f_value, row.p_value,
        ]
        for row in result.alias_interactions.rows
    ]
    _write_table(ws_alias, alias_start, alias_headers, alias_rows)
    _autofit_columns(ws_alias)

    # Model-predicted combinations / confirmation run
    ws_opt = wb.create_sheet(_report_sheet_name("Optimization", temp_suffix))
    opt_start = _write_explanation_block(
        ws_opt, 1, "Optimization — 最佳組合預測", _explain_optimization(design, result.combinations),
    )
    if not result.combinations.skipped:
        combo_headers = ["Rank"] + [s.label for s in result.combinations.column_specs] + [
            f"Predicted {design.response_name}", "Mean 95% CI Low", "Mean 95% CI High",
            "Prediction 95% Low", "Prediction 95% High", "Existing Run", "Confirmation Candidate",
        ]
        output_rows = result.combinations.rows[:64]
        combo_rows = []
        for row in output_rows:
            combo_rows.append(
                [row.rank]
                + [
                    _spec_level_display(design, spec, row.coded_levels[spec.column])
                    for spec in result.combinations.column_specs
                ]
                + [
                    row.predicted, row.mean_ci_low, row.mean_ci_high,
                    row.prediction_low, row.prediction_high,
                    ", ".join(row.observed_run_labels),
                    "Yes" if row is result.combinations.confirmation_row else "",
                ]
            )
        next_row = _write_table(ws_opt, opt_start, combo_headers, combo_rows)
        if len(result.combinations.rows) > len(output_rows):
            ws_opt.cell(
                row=next_row + 1, column=1,
                value=f"完整組合共 {len(result.combinations.rows)} 組；工作表只顯示排名前 {len(output_rows)} 組。",
            )
        profiler_start_row = next_row + 3
        profiler_last_row = _write_prediction_profiler_helper_table(
            ws_opt, profiler_start_row, result.regression, result.combinations,
        )
        if profiler_last_row is not None:
            profiler_chart = _build_prediction_profiler_chart(
                ws_opt, profiler_start_row, profiler_last_row, design.response_name,
            )
            _add_chart_fixed(ws_opt, profiler_chart, f"A{profiler_last_row + 2}")
    _autofit_columns(ws_opt)

    # Correlation
    ws_corr = wb.create_sheet(_report_sheet_name("Correlation", temp_suffix))
    corr_start = _write_explanation_block(
        ws_corr, 1, "Correlation — 因子相關係數矩陣", _explain_correlation(result.correlation),
    )
    corr_headers = [""] + result.correlation.labels
    corr_rows = [[label] + [round(v, 4) for v in row] for label, row in zip(result.correlation.labels, result.correlation.matrix)]
    _write_table(ws_corr, corr_start, corr_headers, corr_rows)
    _autofit_columns(ws_corr)

    # Taguchi
    ws_tag = wb.create_sheet(_report_sheet_name("Taguchi", temp_suffix))
    tag_start = _write_explanation_block(ws_tag, 1, "Taguchi — S/N 訊噪比", _explain_taguchi(result.taguchi))
    if not result.taguchi.skipped:
        tag_headers = ["Rank", "因子代號", "因子名稱", "SN(Low)", "SN(High)", "Delta"]
        tag_rows = [
            [row.rank, row.factor_id, row.name, row.sn_low, row.sn_high, row.delta] for row in result.taguchi.rows
        ]
        _write_table(ws_tag, tag_start, tag_headers, tag_rows)
    _autofit_columns(ws_tag)

    # Pareto
    ws_pareto = wb.create_sheet(_report_sheet_name("Pareto", temp_suffix))
    pareto_start = _write_explanation_block(ws_pareto, 1, "Pareto — 效應大小排序", _explain_pareto(result.pareto))
    pareto_headers = ["因子代號", "因子名稱", "|Effect|"]
    pareto_rows = [[b.factor_id, b.name, b.magnitude] for b in result.pareto]
    next_row = _write_table(ws_pareto, pareto_start, pareto_headers, pareto_rows)
    _autofit_columns(ws_pareto)
    if pareto_rows:
        pareto_chart = _build_pareto_chart(ws_pareto, pareto_start, next_row - 1, design.response_name)
        _add_chart_fixed(ws_pareto, pareto_chart, f"A{next_row + 2}")

    # Interaction plots：取敏感度排名前 top_interaction_pairs 個因子兩兩配對
    # （同一個 group 的因子彼此共用同一欄位，沒有獨立的交互作用可畫，跳過）。
    ws_inter = wb.create_sheet(_report_sheet_name("Interaction", temp_suffix))
    inter_start = _write_explanation_block(ws_inter, 1, "Interaction — 交互作用圖", _explain_interaction())
    top_ids = [
        row.member_factor_ids[0]
        for row in result.sensitivity.rows[
            : min(design.num_columns, max(2, top_interaction_pairs))
        ]
        if row.member_factor_ids
    ]
    row_cursor = inter_start
    for i in range(len(top_ids)):
        for j in range(i + 1, len(top_ids)):
            if design.factor_table_column(top_ids[i]) == design.factor_table_column(top_ids[j]):
                continue
            inter = interaction_data(design, oa, run_matrix, result.temperature, top_ids[i], top_ids[j])
            ws_inter.cell(row=row_cursor, column=1, value=f"{top_ids[i]} x {top_ids[j]}").font = HEADER_FONT
            row_cursor += 1
            if inter.warnings:
                for w in inter.warnings:
                    ws_inter.cell(row=row_cursor, column=1, value=f"⚠ {w}")
                    row_cursor += 1
            table_last_row = _write_interaction_helper_table(ws_inter, row_cursor, inter)
            chart_anchor_row = table_last_row + 1
            inter_chart = _build_interaction_chart(ws_inter, row_cursor, inter, design.response_name)
            _add_chart_fixed(ws_inter, inter_chart, f"A{chart_anchor_row}")
            row_cursor = chart_anchor_row + 17 + 1  # 圖表高度（約17列）+ 間距，避免疊到下一組
    _autofit_columns(ws_inter)

    # DOE 分頁本身在這裡不會被改動內容，但欄寬/列高可能是很久以前產生範本時
    # 設定的，使用者可能之後在 Excel 手動改過因子名稱/數值（變長或變短）；
    # 每次分析都重新 autofit 一次，確保欄寬/列高跟目前實際內容一致。
    ws_doe = wb[DOE_SHEET]
    _autofit_columns(ws_doe)
    _autofit_row_heights(ws_doe)

    # 使用者要求「所有分頁」的儲存格都水平置中，不只前面已經用 _write_table()
    # 產生的表格——連 DOE 分頁、Summary 的說明文字／警告訊息這類本來只靠左對
    # 齊的散文字儲存格也要置中；「說明」分頁是例外（只有標題置中），見
    # `_align_cells_after_write`。
    _align_cells_after_write(wb)

    _pin_doe_sheet_first(wb)
    wb.save(path)

CONFIG_PATH = Path(__file__).resolve().parent / "doe_gui_config.json"

SN_OVERRIDE_LABELS = list(SN_DIRECTION_LABELS.values())
DEFAULT_SN_LABEL = SN_DIRECTION_LABELS[SNDirection.SMALLER]


class DoeApp:
    def __init__(self, root: tk.Tk):
        self.root = root
        self.path_var = tk.StringVar()
        self.status_var = tk.StringVar(value="請選擇既有的 DOE 檔案，或建立新範本。")
        self.sn_override_var = tk.StringVar(value=DEFAULT_SN_LABEL)
        self.interactions_var = tk.BooleanVar(value=False)

        # 「建立新範本」／「調整因子數量」共用的因子數量設定，直接呈現在主視窗
        # 「2. 因子數量設定」區塊裡（不彈出另一個對話視窗）；「重新整理」載入既有
        # 檔案後會用該檔案目前的結構覆寫這幾個欄位，方便「調整因子數量」是從目前
        # 結構改起，不是從上一次殘留的值改起。
        self.mat_factor_var = tk.StringVar(value="0")
        self.other_factor_var = tk.StringVar(value="6")
        self.legs_var = tk.StringVar(value="")
        self.responses_var = tk.StringVar(value="")

        self.design: Optional[DesignConfig] = None
        self.last_report_path: Optional[str] = None

        self._load_config()
        self._build_widgets()
        root.protocol("WM_DELETE_WINDOW", self._on_close)

        if self.path_var.get() and os.path.exists(self.path_var.get()):
            self.load_design()

    # ------------------------------------------------------------------
    # 參數記憶（存/讀 CONFIG_PATH）
    # ------------------------------------------------------------------

    def _load_config(self) -> None:
        try:
            data = json.loads(CONFIG_PATH.read_text(encoding="utf-8"))
        except (OSError, ValueError):
            return  # 檔案不存在／格式壞掉，維持預設值，不中斷啟動

        self.path_var.set(str(data.get("path", "")))
        self.mat_factor_var.set(str(data.get("mat_factors", self.mat_factor_var.get())))
        self.other_factor_var.set(str(data.get("other_factors", self.other_factor_var.get())))
        self.legs_var.set(str(data.get("legs", "")))
        self.responses_var.set(str(data.get("responses", "")))
        sn_label = data.get("sn_direction")
        if sn_label in SN_OVERRIDE_LABELS:
            self.sn_override_var.set(sn_label)
        self.interactions_var.set(bool(data.get("interactions", False)))

    def _save_config(self) -> None:
        data = {
            "path": self.path_var.get(),
            "mat_factors": self.mat_factor_var.get(),
            "other_factors": self.other_factor_var.get(),
            "legs": self.legs_var.get(),
            "responses": self.responses_var.get(),
            "sn_direction": self.sn_override_var.get(),
            "interactions": self.interactions_var.get(),
        }
        try:
            CONFIG_PATH.write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8")
        except OSError:
            pass  # 記憶功能是錦上添花，存檔失敗不該擋住使用者關閉視窗

    def _on_close(self) -> None:
        self._save_config()
        self.root.destroy()

        # 狀態列要在「4. 結果」（唯一 expand=True 的區塊）之前 pack，讓它先在最下方
        # 卡一個固定高度的位置；pack 的版面配置是依呼叫順序分配空間，expand=True
        # 的區塊會把當下剩餘空間整個吃掉，晚 pack 的狀態列在視窗被縮小/搬動導致重
        # 排版時會被擠到看不見。狀態列固定高度、不用 expand，順序放前面就能保證
        # 視窗再怎麼縮小都優先保留它的空間，「4. 結果」再用剩餘空間自動收縮。
        status_bar = ttk.Label(self.root, textvariable=self.status_var, anchor="w", relief="sunken")
        status_bar.pack(fill="x", side="bottom")

        result_frame = ttk.LabelFrame(self.root, text="4. 結果")
        result_frame.pack(fill="both", expand=True, padx=PAD, pady=(0, PAD))
        self.result_text = tk.Text(result_frame, wrap="word", state="disabled")
        result_scroll = ttk.Scrollbar(result_frame, orient="vertical", command=self.result_text.yview)
        self.result_text.configure(yscrollcommand=result_scroll.set)
        self.result_text.pack(side="left", fill="both", expand=True, padx=(PAD, 0), pady=PAD)
        result_scroll.pack(side="left", fill="y", pady=PAD, padx=(0, PAD))

    def load_design(self) -> None:
        path = self.path_var.get()
        if not path or not os.path.exists(path):
            messagebox.showwarning(APP_TITLE, "請先選擇或建立一個 DOE 檔案。")
            return
        try:
            design, _run_matrix, _warnings = read_design_and_runs(path)
        except ValueError as exc:
            self._reset_design_state()
            self._set_status(f"讀取 DOE 工作表失敗：{exc}")
            return

        self.design = design
        # 用目前檔案的結構覆寫「2. 因子數量設定」欄位，這樣按「調整因子數量」時是
        # 從目前結構改起，不會顯示上一次殘留在欄位裡的舊值。
        current_mat = sum(1 for f in design.factors if f.group == GROUP_MAT)
        current_other = design.k - current_mat
        self.mat_factor_var.set(str(current_mat))
        self.other_factor_var.set(str(current_other))
        try:
            selected_temp = read_selected_response_temperature(path)
            selected_desc = f"目前勾選要分析的溫度：{selected_temp}"
        except ValueError as exc:
            selected_desc = f"目前量測列 select 欄勾選狀態有誤：{exc}"
        self._set_status(f"已載入 DOE 工作表：k={design.k}，溫度階段={design.temperatures}。{selected_desc}")

    def _read_factor_count_inputs(
        self, require_responses: bool
    ) -> Optional[Tuple[int, int, Optional[int], Optional[int]]]:
        """讀取並驗證「2. 因子數量設定」目前的欄位值（直接顯示在主視窗，不彈出另一
        個對話視窗）。材料/非材料因子個數留空時視為 0（不是錯誤）。驗證失敗會跳出
        錯誤訊息並回傳 None，呼叫端據此中止；通過驗證回傳 `(mat, other, legs,
        responses)`。`require_responses=False` 時（調整因子數量）不檢查 response
        欄位，回傳的 responses 固定是 None——既有檔案的量測列數量固定沿用檔案本
        身，response 數量只在建立新範本時才有意義。
        """

        mat_str = self.mat_factor_var.get().strip()
        other_str = self.other_factor_var.get().strip()
        try:
            mat = int(mat_str) if mat_str else 0
            other = int(other_str) if other_str else 0
        except ValueError:
            messagebox.showerror(APP_TITLE, "材料因子與非材料因子個數請輸入整數，或留空視為 0。")
            return None
        if not (0 <= mat <= 64) or not (0 <= other <= 64):
            messagebox.showerror(APP_TITLE, "因子個數請輸入 0~64 之間的整數。")
            return None
        if mat < 1 and other < 1:
            messagebox.showerror(APP_TITLE, "材料因子跟非材料因子至少要有一個大於 0。")
            return None

        legs: Optional[int] = None
        legs_str = self.legs_var.get().strip()
        if legs_str:
            try:
                legs = int(legs_str)
            except ValueError:
                messagebox.showerror(APP_TITLE, "leg 數量請輸入整數，或留空自動計算。")
                return None
            if not (2 <= legs <= 1024):
                messagebox.showerror(APP_TITLE, "leg 數量請輸入 2~1024 之間的整數，或留空自動計算。")
                return None

        responses: Optional[int] = None
        if require_responses:
            responses_str = self.responses_var.get().strip()
            if responses_str:
                try:
                    responses = int(responses_str)
                except ValueError:
                    messagebox.showerror(APP_TITLE, "response 數量請輸入整數。")
                    return None
                if not (1 <= responses <= 64):
                    messagebox.showerror(APP_TITLE, "response 數量請輸入 1~64 之間的整數。")
                    return None

        return mat, other, legs, responses

    def adjust_factor_count(self) -> None:
        path = self.path_var.get()
        if not path or not os.path.exists(path):
            messagebox.showwarning(APP_TITLE, "請先選擇或建立一個 DOE 檔案。")
            return

        parsed = self._read_factor_count_inputs(require_responses=False)
        if parsed is None:
            return  # 驗證失敗，錯誤訊息已經在 _read_factor_count_inputs 裡顯示過
        num_mat, num_other, num_legs, _num_responses = parsed
        self._do_generate_template(path, num_mat, num_other, num_legs, force=False)

    def _do_generate_template(
        self, path: str, num_mat_factors: int, num_other_factors: int, num_legs: Optional[int], force: bool
    ) -> None:
        try:
            report = generate_template(
                path, num_mat_factors=num_mat_factors, num_other_factors=num_other_factors,
                num_legs=num_legs, force=force,
            )
        except ValueError as exc:
            if not force and "--force" in str(exc):
                if messagebox.askyesno("確認覆寫", f"{exc}\n\n是否要繼續？"):
                    self._do_generate_template(path, num_mat_factors, num_other_factors, num_legs, force=True)
                return
            messagebox.showerror(APP_TITLE, str(exc))
            return

        if report.warnings:
            self._set_status(
                f"因子數量已調整（材料 {report.mat_factor_count} 個、非材料 {report.other_factor_count} 個，"
                f"N={report.run_count}）。保留量測值 {report.preserved_cells} 筆，捨棄 {report.dropped_cells} 筆。"
            )
            messagebox.showwarning(APP_TITLE, "\n".join(report.warnings))
        else:
            self._set_status(f"因子數量未變更（材料 {report.mat_factor_count} 個、非材料 {report.other_factor_count} 個）。")
        self.load_design()

    def _reset_design_state(self) -> None:
        self.design = None

    def run_analysis(self) -> None:
        path = self.path_var.get()
        if not path or not os.path.exists(path):
            messagebox.showwarning(APP_TITLE, "請先選擇或建立一個 DOE 檔案。")
            return

        sn_direction = SN_DIRECTION_BY_LABEL[self.sn_override_var.get()]

        try:
            # 要分析哪個溫度直接看 Excel 量測列的 select 欄（A欄）打 v 在哪一列，
            # 取代舊版溫度下拉選單，沒勾或勾了超過一個都會丟 ValueError 說明原因。
            temperature = read_selected_response_temperature(path)
            # 每次分析都重新讀檔（不沿用 load_design() 快取的 design），避免使用者在
            # Excel 改完資料忘記按「重新整理」時，分析用到過期的因子/量測內容。
            design, run_matrix, read_warnings = read_design_and_runs(path, sn_direction=sn_direction)
            oa = generate_orthogonal_array(design.num_columns, min_runs=len(run_matrix.run_labels))
            result = run_full_analysis(
                design, oa, run_matrix, temperature, include_interactions=self.interactions_var.get()
            )
            recommendations = generate_recommendations(design, result)
            # baseline leg 比較是選用功能：讀最上面那一列的 1/2 標記，完全沒標記時
            # compare_to_baseline_leg() 會回傳 enabled=False，Summary 跟結果面板都
            # 不會顯示這段，不影響沒用這個功能的人。
            baseline_leg, compare_legs = read_baseline_compare_marks(path)
            baseline_compare = compare_to_baseline_leg(design, run_matrix, temperature, baseline_leg, compare_legs)
            # 分析報告直接寫回同一份 DOE 檔案（Summary/MainEffect/... 分頁），不再
            # 另外產生一份 report .xlsx。
            write_report(path, design, oa, run_matrix, result, recommendations, baseline_compare=baseline_compare)
        except ValueError as exc:
            messagebox.showerror(APP_TITLE, str(exc))
            return
        except Exception as exc:  # noqa: BLE001 — GUI 事件處理器需攔截未預期例外並顯示，而非讓 Tk 靜默吞掉
            messagebox.showerror(APP_TITLE, f"分析過程發生未預期錯誤：{exc}")
            return

        self.last_report_path = path
        self.open_report_button.state(["!disabled"])
        self._render_analysis_result(
            design, run_matrix, temperature, read_warnings, result, recommendations,
            baseline_compare, baseline_leg, compare_legs, path,
        )
        self._set_status(f"分析完成，報告已寫入同一份檔案：{path}")

    def _render_analysis_result(
        self, design, run_matrix, temperature, read_warnings, result, recommendations,
        baseline_compare, baseline_leg, compare_legs, out_path,
    ) -> None:
        lines = ["=== 本次分析使用的因子與 leg（依 Excel 的 select／leg 勾選列，請對照確認）==="]
        factor_labels = [
            f"{f.factor_id}（材料組）" if f.group == GROUP_MAT else f.factor_id
            for f in design.factors
        ]
        lines.append(f"已勾選因子（{len(design.factors)} 個）：{', '.join(factor_labels)}")
        lines.append(f"已勾選 leg（{len(run_matrix.run_labels)} 個）：{', '.join(run_matrix.run_labels)}")
        lines.append(f"已勾選要分析的 response（溫度）：{design.response_name} @ {temperature}{design.response_unit or ''}")
        if baseline_leg or compare_legs:
            lines.append(
                f"baseline 比較列標記（最上面那列，選用功能）：基準={baseline_leg or '（未標記）'}，"
                f"候選={', '.join(compare_legs) if compare_legs else '（未標記）'}"
            )
        else:
            lines.append("baseline 比較列標記（最上面那列，選用功能）：未標記（不啟用 baseline 比較）")
        lines.append("")
        lines.append(f"=== {design.response_name} @ {temperature}{design.response_unit or ''} 顯著因子排序 ===")
        for row in result.sensitivity.rows:
            lines.append(f"{row.rank}. {row.factor_id}（{row.name}）score={row.score:.3f}")

        lines.append("")
        lines.append("=== 迴歸模型品質 ===")
        reg = result.regression
        model_metrics = [f"R²={reg.r_squared:.4f}"]
        if reg.adjusted_r_squared is not None:
            model_metrics.append(f"調整 R²={reg.adjusted_r_squared:.4f}")
        if reg.predicted_r_squared is not None:
            model_metrics.append(f"預測 R²={reg.predicted_r_squared:.4f}")
        if reg.rmse is not None:
            model_metrics.append(f"RMSE={reg.rmse:.4g}")
        model_metrics.append(f"殘差自由度={reg.residual_df}")
        lines.append("；".join(model_metrics))
        lines.append(f"效應篩選方法：{result.effect_screening.method}")
        active_effects = [row for row in result.effect_screening.rows if row.active is True]
        if active_effects:
            lines.append(
                "判定為 active 的效應："
                + "、".join(f"{row.label}（effect={row.effect:.4g}）" for row in active_effects)
            )
        else:
            lines.append("判定為 active 的效應：無，或目前資料不足以可靠判定。")

        combinations = result.combinations
        if not combinations.skipped and combinations.rows:
            lines.append("")
            lines.append(f"=== 模型預測最佳組合（{SN_DIRECTION_LABELS[design.sn_direction]}）===")
            best = combinations.rows[0]
            for spec in combinations.column_specs:
                lines.append(
                    f"- column{spec.column} {spec.label}（{spec.display_name}）："
                    f"{_spec_level_display(design, spec, best.coded_levels[spec.column])}"
                )
            observed = "、".join(best.observed_run_labels) if best.observed_run_labels else "尚未實跑"
            lines.append(f"預測 {design.response_name}={best.predicted:.4g}；既有 run：{observed}")
            if combinations.confirmation_row is not None:
                confirm = combinations.confirmation_row
                settings = "；".join(
                    f"{spec.label}={_spec_level_display(design, spec, confirm.coded_levels[spec.column])}"
                    for spec in combinations.column_specs
                )
                lines.append(
                    f"建議確認實驗：排名 {confirm.rank} 的首個未實跑組合，"
                    f"預測值={confirm.predicted:.4g}（{settings}）。"
                )
        elif combinations.skipped:
            lines.append("")
            lines.append(f"=== 模型預測最佳組合 ===\n已略過：{combinations.skip_reason}")

        all_warnings = list(read_warnings) + list(result.warnings)
        if all_warnings:
            lines.append("")
            lines.append("=== 警告 ===")
            for w in all_warnings:
                lines.append(f"⚠ {w}")

        if recommendations:
            lines.append("")
            lines.append("=== 建議 ===")
            for rec in recommendations:
                lines.append(f"- {rec.factor_id}（{rec.name}）建議選 {rec.recommended_level} 水準（{rec.recommended_label}）— {rec.rationale}")

        if result.leg_ranking:
            lines.append("")
            lines.append(f"=== 依 leg 排序（{SN_DIRECTION_LABELS[design.sn_direction]}，rank 1 最好）===")
            for row in result.leg_ranking:
                lines.append(f"{row.rank}. {row.run_label}：{design.response_name}={row.response_value:.4g}")

        if baseline_compare.enabled:
            lines.append("")
            lines.append("=== 與 baseline 比較 ===")
            if baseline_compare.skipped:
                lines.append(f"已略過：{baseline_compare.skip_reason}")
            else:
                lines.append(
                    f"基準 leg：{baseline_compare.baseline_run_label}"
                    f"（{design.response_name}={baseline_compare.baseline_response:.4g}）"
                )
                for w in baseline_compare.warnings:
                    lines.append(f"⚠ {w}")
                for row in baseline_compare.rows:
                    pct_text = f"{row.improvement_pct:+.1f}%" if row.improvement_pct is not None else "N/A"
                    lines.append(
                        f"{row.rank}. {row.run_label}：{design.response_name}={row.response_value:.4g}"
                        f"（Δ={row.response_delta:+.4g}，改善 {pct_text}）"
                    )
                best = baseline_compare.rows[0]
                lines.append(f"  改善幅度最大的候選 leg（{best.run_label}）因子調整方向：")
                for fa in best.factor_adjustments:
                    pct_text = f"，{fa.pct_change:+.1f}%" if fa.pct_change is not None else ""
                    lines.append(f"    {fa.factor_id}（{fa.name}）：{fa.baseline_value} → {fa.leg_value}{pct_text}（{fa.direction}）")

        lines.append("")
        temp_suffix = _temp_suffix(temperature)
        lines.append(f"完整報告（同一份檔案的 Summary_{temp_suffix}/MainEffect_{temp_suffix}/... 分頁）：{out_path}")
        self._set_text(self.result_text, "\n".join(lines))

def cli_generate_template(argv: List[str]) -> int:
    parser = argparse.ArgumentParser(description="產生 DOE 分析範本 .xlsx，或調整既有檔案的因子數量")
    parser.add_argument("--out", required=True, help="輸出檔案路徑")
    parser.add_argument("--factors", type=int, default=None, help="舊版單一參數寫法，等同 --other-factors（不要跟 --mat-factors/--other-factors 一起用）")
    parser.add_argument("--mat-factors", type=int, default=0, help="材料相關參數個數（共用同一組 Low/High 排列，例如同一個材料的 Shrinkage/Ex/PRXY）")
    parser.add_argument("--other-factors", type=int, default=0, help="非材料因子個數（各自獨立一個正交表欄位）")
    parser.add_argument("--legs", type=int, default=None, help="leg（run）數量；留空時依因子數自動算最小需求")
    parser.add_argument(
        "--responses", type=int, default=None,
        help="量測列（溫度階段）數量；留空時預設 4 個（僅新建檔案時生效，既有檔案的量測列數量不受影響）",
    )
    parser.add_argument("--force", action="store_true", help="確認覆寫（結構變更時需要）")
    args = parser.parse_args(argv)

    try:
        report = generate_template(
            args.out, num_factors=args.factors, num_mat_factors=args.mat_factors,
            num_other_factors=args.other_factors, num_legs=args.legs,
            num_responses=args.responses, force=args.force,
        )
    except ValueError as exc:
        print(f"錯誤：{exc}", file=sys.stderr)
        return 1

    if report.created:
        print(f"已建立新範本：{args.out}")
        print(f"材料因子 {report.mat_factor_count} 個、非材料因子 {report.other_factor_count} 個，run 數 N={report.run_count}")
        print("已帶入預設因子/量測參數，可直接執行 analyze_doe.py 試跑，")
        print("或先在 Excel 把因子名稱/數值、量測結果換成你自己的實際資料再分析。")
    elif report.warnings:
        print(f"已依新的因子/leg 數量重新產生範本：{args.out}")
        print(f"材料因子 {report.mat_factor_count} 個、非材料因子 {report.other_factor_count} 個，run 數 N={report.run_count}")
        print(f"保留既有量測值：{report.preserved_cells} 筆；無法對應而捨棄：{report.dropped_cells} 筆")
        for w in report.warnings:
            print(f"⚠ {w}")
    else:
        print(f"因子/leg 數量未變更，DOE 工作表維持原樣（僅更新「說明」工作表）：{args.out}")

    return 0

}


def cli_analyze_doe(argv: List[str]) -> int:
    parser = argparse.ArgumentParser(description="分析 DOE .xlsx，輸出指定溫度的顯著因子排序與建議")
    parser.add_argument("--in", dest="in_path", required=True, help="已填好的 DOE .xlsx 路徑")
    parser.add_argument("--temperature", type=float, required=True, help="要分析的溫度（需與 Factors 工作表設定的溫度階段之一相符）")
    parser.add_argument("--interactions", action="store_true", help="迴歸模型加入未混淆的交互作用項")
    parser.add_argument("--sn-direction", choices=sorted(SN_DIRECTION_CHOICES), default=None, help="S/N 最佳化方向（預設 smaller）")
    parser.add_argument("--correlation-threshold", type=float, default=0.3, help="因子間相關係數警告門檻（預設 0.3）")
    parser.add_argument("--top-recommendations", type=int, default=3, help="建議清單列出前幾名因子（預設 3）")
    args = parser.parse_args(argv)

    if not os.path.exists(args.in_path):
        print(f"錯誤：找不到檔案 {args.in_path}", file=sys.stderr)
        return 1

    sn_direction = SN_DIRECTION_CHOICES[args.sn_direction] if args.sn_direction is not None else None
    try:
        design, run_matrix, read_warnings = read_design_and_runs(args.in_path, sn_direction=sn_direction)
    except ValueError as exc:
        print(f"錯誤：{exc}", file=sys.stderr)
        return 1

    oa = generate_orthogonal_array(design.num_columns, min_runs=len(run_matrix.run_labels))

    for w in read_warnings:
        print(f"⚠ {w}")

    try:
        result = run_full_analysis(
            design, oa, run_matrix, args.temperature,
            include_interactions=args.interactions,
            correlation_threshold=args.correlation_threshold,
        )
    except ValueError as exc:
        print(f"錯誤：{exc}", file=sys.stderr)
        return 1

    recommendations = generate_recommendations(design, result, top_n=args.top_recommendations)

    # baseline leg 比較是選用功能：讀 DOE 表最上面那一列的 1/2 標記，完全沒標記時
    # 回傳 enabled=False，報告跟輸出都不會出現這段，不影響沒用這個功能的人。
    baseline_leg, compare_legs = read_baseline_compare_marks(args.in_path)
    baseline_compare = compare_to_baseline_leg(design, run_matrix, args.temperature, baseline_leg, compare_legs)

    write_report(args.in_path, design, oa, run_matrix, result, recommendations, baseline_compare=baseline_compare)

    print(f"\n=== {design.response_name} @ {args.temperature}{design.response_unit or ''} 顯著因子排序 ===")
    for row in result.sensitivity.rows:
        print(f"{row.rank}. {row.factor_id}（{row.name}）score={row.score:.3f}")

    if result.warnings:
        print("\n=== 警告 ===")
        for w in result.warnings:
            print(f"⚠ {w}")

    if recommendations:
        print("\n=== 建議 ===")
        for rec in recommendations:
            print(f"- {rec.factor_id}（{rec.name}）建議選 {rec.recommended_level} 水準（{rec.recommended_label}）— {rec.rationale}")

    if baseline_compare.enabled:
        print("\n=== 與 baseline 比較 ===")
        if baseline_compare.skipped:
            print(f"已略過：{baseline_compare.skip_reason}")
        else:
            print(f"基準 leg：{baseline_compare.baseline_run_label}（{design.response_name}={baseline_compare.baseline_response:.4g}）")
            for w in baseline_compare.warnings:
                print(f"⚠ {w}")
            for row in baseline_compare.rows:
                pct_text = f"{row.improvement_pct:+.1f}%" if row.improvement_pct is not None else "N/A"
                print(f"{row.rank}. {row.run_label}：{design.response_name}={row.response_value:.4g}（Δ={row.response_delta:+.4g}，改善 {pct_text}）")
            best = baseline_compare.rows[0]
            print(f"  改善幅度最大的候選 leg（{best.run_label}）因子調整方向：")
            for fa in best.factor_adjustments:
                pct_text = f"，{fa.pct_change:+.1f}%" if fa.pct_change is not None else ""
                print(f"    {fa.factor_id}（{fa.name}）：{fa.baseline_value} → {fa.leg_value}{pct_text}（{fa.direction}）")

    print(f"\n分析報告已寫入：{args.in_path}")
    return 0


def run_smoke_test() -> int:
    """建立主視窗並跑一輪「建立範本→重新整理→執行分析→調整因子數量」，確認 GUI
    能正常啟動、不會卡在 mainloop()（比照 13_cte_calibrate/cte_calibrate_gui.pyw
    的 run_smoke_test() 慣例）。

    新範本一建立就已經用 `doe_excel.DEFAULT_MAT_FACTOR_VALUES`/
    `DEFAULT_OTHER_FACTOR_VALUES`/`DEFAULT_WARPAGE_BY_TEMP` 填好可直接分析的預設
    參數（不再是 Low/High 佔位文字），所以不用像舊版煙霧測試那樣手動改寫 Excel
    儲存格內容，直接載入、分析即可。

    調整因子數量會走到 messagebox.showwarning（真的彈出視窗等人按 OK），這裡暫時
    替換成不彈窗的假函式，煙霧測試才能自動跑完、不會卡住等真人互動。
    """

    import tempfile

    original_showwarning = messagebox.showwarning
    original_showerror = messagebox.showerror
    original_askyesno = messagebox.askyesno
    messagebox.showwarning = lambda *a, **k: None
    messagebox.showerror = lambda *a, **k: None
    messagebox.askyesno = lambda *a, **k: True

    root = tk.Tk()
    try:
        root.withdraw()
        app = DoeApp(root)

        tmp_dir = tempfile.mkdtemp(prefix="doe_gui_smoke_")
        path = os.path.join(tmp_dir, "smoke.xlsx")
        # 2 個材料性質共用一個設計欄位 + 1 個非材料因子，實際驗證 group-aware
        # 分析與報表路徑（只有 1 個材料列時無法測出重複欄位問題）。
        report = generate_template(path, num_mat_factors=2, num_other_factors=1)
        if report.mat_factor_count != 2 or report.other_factor_count != 1:
            raise AssertionError("建立範本後材料/非材料因子數不正確。")

        app.path_var.set(path)
        app.load_design()
        if app.design is None or app.design.k != 3:
            raise AssertionError("讀回 DOE 工作表後 design 未正確載入。")
        if app.design.factors[0].group != GROUP_MAT or app.design.factors[1].group != GROUP_MAT:
            raise AssertionError("材料因子的 group 未正確讀回。")
        if app.design.num_columns != 2:
            raise AssertionError("兩個材料性質沒有共用同一個設計欄位。")

        # 新範本第一個溫度的量測列 select 欄預設已打 v，不用另外指定要分析哪個溫度。
        selected_temp = read_selected_response_temperature(path)
        if selected_temp != app.design.temperatures[0]:
            raise AssertionError("新範本的量測列 select 欄預設勾選的溫度不是第一個溫度。")
        app.run_analysis()
        if not app.last_report_path or not os.path.exists(app.last_report_path):
            raise AssertionError("執行分析後未產生報告檔案。")
        report_wb = load_workbook(path, read_only=True)
        try:
            suffix = _temp_suffix(selected_temp)
            expected_sheets = {
                "說明",
                *(_report_sheet_name(base, suffix) for base in REPORT_SHEET_NAMES),
            }
            missing_sheets = expected_sheets.difference(report_wb.sheetnames)
            if missing_sheets:
                raise AssertionError(f"分析後缺少報告分頁：{sorted(missing_sheets)}")
        finally:
            report_wb.close()

        # 順手驗證「調整因子數量」（材料2+非材料1 -> 材料2+非材料2，結構性變更需要
        # force）也能跑；因子數量設定直接改「2. 因子數量設定」的欄位值（不彈出對話
        # 視窗），走跟使用者按鈕一樣的路徑，而不是直接呼叫內部的 _do_generate_template。
        app.mat_factor_var.set("2")
        app.other_factor_var.set("2")
        app.adjust_factor_count()
        if app.design is None or app.design.k != 4:
            raise AssertionError("調整因子數量後 design 未正確載入。")
    except Exception as exc:
        print(f"[smoke-test] 失敗：{exc}", file=sys.stderr)
        return 1
    finally:
        root.destroy()
        messagebox.showwarning = original_showwarning
        messagebox.showerror = original_showerror
        messagebox.askyesno = original_askyesno

    print("[smoke-test] 通過：GUI 可正常建立範本、把佔位文字換成實際值、重新整理、執行分析並調整因子數量。")
    return 0


def configure_style(root: tk.Tk) -> None:
    style = ttk.Style(root)
    if "vista" in style.theme_names():
        style.theme_use("vista")


def main() -> int:
    if "--smoke-test" in sys.argv[1:]:
        return run_smoke_test()
    if len(sys.argv) > 1 and sys.argv[1] == "generate-template":
        return cli_generate_template(sys.argv[2:])
    if len(sys.argv) > 1 and sys.argv[1] == "analyze":
        return cli_analyze_doe(sys.argv[2:])

    root = tk.Tk()
    root.title(APP_TITLE)
    root.geometry(WINDOW_GEOMETRY)
    root.minsize(WINDOW_MIN_WIDTH, WINDOW_MIN_HEIGHT)
    configure_style(root)
    DoeApp(root)
    root.mainloop()
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
