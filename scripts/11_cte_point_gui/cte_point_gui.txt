PYTHONW_EXE = Path(
    r"C:\Users\python3.8\win64\pythonw.exe"
)


def _executable_path_key(path: Path) -> str:
    """Return a case-insensitive absolute identity for a Windows executable."""

    return os.path.normcase(os.path.abspath(str(path)))


def _show_bootstrap_error(detail: str) -> None:
    """Show an error even when the current interpreter has no console."""

    if sys.stderr is not None:
        sys.stderr.write(detail + "\n")
    try:
        import ctypes

        ctypes.windll.user32.MessageBoxW(0, detail, "ANSYS CTE 溫度點內插工具", 0x10)
    except (AttributeError, OSError):
        pass


def _ensure_configured_python() -> None:
    """Relaunch this file with the configured portable Python when necessary."""

    configured_pythonw = PYTHONW_EXE.expanduser()
    configured_python = configured_pythonw.with_name("python.exe")
    current_key = _executable_path_key(Path(sys.executable))
    configured_keys = {
        _executable_path_key(configured_pythonw),
        _executable_path_key(configured_python),
    }
    if current_key in configured_keys:
        return

    if not configured_pythonw.is_file():
        _show_bootstrap_error(
            "找不到指定的 Python 執行檔：\n{}\n\n"
            "請用文字編輯器開啟 cte_point_gui.pyw，修改檔案最上方的 PYTHONW_EXE。".format(
                configured_pythonw
            )
        )
        raise SystemExit(1)

    current_is_console_python = Path(sys.executable).name.lower() == "python.exe"
    target_executable = (
        configured_python
        if current_is_console_python and configured_python.is_file()
        else configured_pythonw
    )
    script_path = Path(__file__).resolve()
    command = [str(target_executable), str(script_path)] + sys.argv[1:]
    try:
        if current_is_console_python:
            return_code = subprocess.call(command, cwd=str(script_path.parent))
            raise SystemExit(return_code)
        subprocess.Popen(command, cwd=str(script_path.parent))
    except OSError as exc:
        _show_bootstrap_error(
            "無法使用指定的 Python 啟動程式：\n{}\n\n{}".format(target_executable, exc)
        )
        raise SystemExit(1)
    raise SystemExit(0)


if __name__ == "__main__":
    _ensure_configured_python()

    """可直接顯示給使用者的輸入錯誤。"""

@dataclass(frozen=True)
class TransformReport:
    added: tuple[Decimal, ...]
    skipped: tuple[str, ...]
    point_count: int
    generated_mptemp: bool = False
    mptemp_numbering_corrections: tuple[
        MptempNumberingCorrection, ...
    ] = ()

    def summary(self) -> str:
        added_text = "、".join(format_decimal(value) for value in self.added)
        if added_text:
            first_line = f"已新增 {len(self.added)} 點：{added_text} °C。"
        else:
            first_line = "沒有需要新增的溫度點。"

        parts: list[str] = []
        if self.generated_mptemp:
            parts.append("已依 MPDATA 註解自動建立 MPTEMP 與 UIMP。")
        if self.mptemp_numbering_corrections:
            parts.append(
                "已自動修正 "
                f"{len(self.mptemp_numbering_corrections)} 個 MPTEMP 編號。"
            )
        parts.append(first_line)
        if self.skipped:
            parts.append("略過：" + "；".join(self.skipped) + "。")
        parts.append(f"輸出共有 {self.point_count} 組 MPTEMP / MPDATA,CTEX 資料。")
        return " ".join(parts)

    def mptemp_numbering_notice(self) -> str:
        details = []
        for correction in self.mptemp_numbering_corrections:
            original = correction.original or "（空白）"
            details.append(
                f"第 {correction.line_number} 列："
                f"{original} → {correction.corrected}"
            )
        return (
            "發現 MPTEMP 編號不是從 1 開始連續遞增，"
            "已依資料列順序自動修正。\n\n"
            + "\n".join(details)
        )

def _normalise_number_token(token: str) -> str:
    return re.sub(
        r"(?<=[0-9.])\s*[DdEe]\s*([+-]?)\s*(?=\d)",
        lambda match: "E" + match.group(1),
        token.strip(),
        flags=re.IGNORECASE,
    )


def _decimal_from_token(token: str) -> Decimal:
    try:
        return Decimal(_normalise_number_token(token))
    except InvalidOperation as exc:
        raise CteToolError(f"無法辨識數值：{token!r}") from exc


def _normalise_apdl_expression(expression: str) -> str:
    expression = _normalise_number_token(expression)
    return re.sub(r"\s+", "", expression)


def _evaluate_numeric_expression(expression: str) -> Decimal:
    normalised = _normalise_apdl_expression(expression).replace("^", "**")
    try:
        tree = ast.parse(normalised, mode="eval")
    except SyntaxError as exc:
        raise CteToolError(f"無法解析數值運算式：{expression!r}") from exc

    def evaluate(node: ast.AST) -> Decimal:
        if isinstance(node, ast.Expression):
            return evaluate(node.body)
        if isinstance(node, ast.Constant) and isinstance(node.value, (int, float)):
            return Decimal(str(node.value))
        if isinstance(node, ast.UnaryOp) and isinstance(node.op, (ast.UAdd, ast.USub)):
            value = evaluate(node.operand)
            return value if isinstance(node.op, ast.UAdd) else -value
        if isinstance(node, ast.BinOp):
            left = evaluate(node.left)
            right = evaluate(node.right)
            if isinstance(node.op, ast.Add):
                return left + right
            if isinstance(node.op, ast.Sub):
                return left - right
            if isinstance(node.op, ast.Mult):
                return left * right
            if isinstance(node.op, ast.Div):
                if right == 0:
                    raise CteToolError(f"數值運算式除以零：{expression!r}")
                return left / right
            if isinstance(node.op, ast.Pow) and right == right.to_integral_value():
                return left ** int(right)
        raise CteToolError(f"MPTEMP 僅支援數值運算式：{expression!r}")

    return evaluate(tree)


def _try_evaluate_constant(expression: str) -> Decimal | None:
    try:
        return _evaluate_numeric_expression(expression)
    except CteToolError:
        return None


def _split_command_line(line_index: int, raw: str) -> CommandLine:
    if "!" in raw:
        before_comment, comment = raw.split("!", 1)
        code = before_comment.rstrip()
    else:
        code = raw.rstrip()
        comment = None
    return CommandLine(
        line_index=line_index,
        raw=raw,
        code=code,
        comment=comment,
        fields=tuple(code.split(",")),
    )


def _parse_leading_comment_temperature(comment: str | None) -> Decimal | None:
    if comment is None:
        return None
    match = LEADING_NUMBER_RE.match(comment)
    if not match:
        return None
    try:
        return _decimal_from_token(match.group(1))
    except CteToolError:
        return None


def _parse_offset_expression(expression: str) -> tuple[Decimal, str, Decimal] | None:
    match = OFFSET_EXPRESSION_RE.match(expression)
    if not match:
        return None
    base_token, sign, offset_token = match.groups()
    base = _decimal_from_token(base_token)
    offset = _decimal_from_token(offset_token)
    if sign == "-":
        offset = -offset
    return base, _normalise_number_token(base_token), offset


def _nearly_equal(left: Decimal, right: Decimal) -> bool:
    return abs(left - right) <= DECIMAL_TOLERANCE


def _infer_temperatures(
    mptemp_lines: list[CommandLine],
    mpdata_lines: list[CommandLine],
) -> tuple[list[Decimal], TemperatureStyle]:
    absolute_temperatures = [
        _evaluate_numeric_expression(line.fields[2]) for line in mptemp_lines
    ]
    comment_temperatures = [
        _parse_leading_comment_temperature(line.comment) for line in mpdata_lines
    ]
    offset_forms = [_parse_offset_expression(line.fields[2]) for line in mptemp_lines]

    if all(value is not None for value in comment_temperatures):
        comments = [value for value in comment_temperatures if value is not None]
        differences = [
            absolute - comment
            for absolute, comment in zip(absolute_temperatures, comments)
        ]
        if all(_nearly_equal(value, differences[0]) for value in differences[1:]):
            if all(value is not None for value in offset_forms):
                parsed_offsets = [value for value in offset_forms if value is not None]
                first_base = parsed_offsets[0][0]
                same_base = all(
                    _nearly_equal(item[0], first_base) for item in parsed_offsets[1:]
                )
                offsets_match_comments = all(
                    _nearly_equal(item[2], comment)
                    for item, comment in zip(parsed_offsets, comments)
                )
                if same_base and offsets_match_comments:
                    return comments, TemperatureStyle(
                        kind="offset_expression",
                        base_value=first_base,
                        base_token=parsed_offsets[0][1],
                    )
            return comments, TemperatureStyle(
                kind="absolute_numeric",
                base_value=differences[0],
                base_token=format_decimal(differences[0]),
            )

    if all(value is not None for value in offset_forms):
        parsed_offsets = [value for value in offset_forms if value is not None]
        first_base = parsed_offsets[0][0]
        if all(_nearly_equal(item[0], first_base) for item in parsed_offsets[1:]):
            return [item[2] for item in parsed_offsets], TemperatureStyle(
                kind="offset_expression",
                base_value=first_base,
                base_token=parsed_offsets[0][1],
            )

    return absolute_temperatures, TemperatureStyle(kind="direct")


def _collect_table(text: str) -> tuple[list[str], list[ExistingPoint], TemperatureStyle]:
    lines = text.splitlines()
    mptemp_lines: list[CommandLine] = []
    mpdata_lines: list[CommandLine] = []

    for line_index, raw in enumerate(lines):
        parsed = _split_command_line(line_index, raw)
        if not parsed.fields:
            continue

        command = parsed.fields[0].strip().upper()
        if command == "MPTEMP":
            if len(parsed.fields) < 3:
                if any(field.strip() for field in parsed.fields[1:]):
                    raise CteToolError(
                        f"第 {line_index + 1} 列的 MPTEMP 欄位不足。"
                    )
                continue
            location = parsed.fields[1].strip()
            temperature = parsed.fields[2].strip()
            if not location and not temperature:
                continue
            if not location or not temperature:
                raise CteToolError(
                    f"第 {line_index + 1} 列的 MPTEMP 缺少位置或溫度。"
                )
            if any(field.strip() for field in parsed.fields[3:]):
                raise CteToolError(
                    f"第 {line_index + 1} 列一次含多個 MPTEMP 溫度；"
                    "本工具目前要求每列一個溫度點。"
                )
            _evaluate_numeric_expression(temperature)
            mptemp_lines.append(parsed)
            continue

        if command == "MPDATA" and len(parsed.fields) >= 2:
            if parsed.fields[1].strip().upper() != "CTEX":
                continue
            if len(parsed.fields) < 5 or not parsed.fields[4].strip():
                raise CteToolError(
                    f"第 {line_index + 1} 列的 MPDATA,CTEX 缺少 C1 數值。"
                )
            if any(field.strip() for field in parsed.fields[5:]):
                raise CteToolError(
                    f"第 {line_index + 1} 列一次含多個 MPDATA 數值；"
                    "本工具目前要求每列一個 CTE 值。"
                )
            mpdata_lines.append(parsed)

    if not mptemp_lines:
        raise CteToolError("找不到含溫度資料的 MPTEMP 列。")
    if not mpdata_lines:
        raise CteToolError("找不到 MPDATA,CTEX 列。")
    if len(mptemp_lines) != len(mpdata_lines):
        raise CteToolError(
            "MPTEMP 與 MPDATA,CTEX 的點數不同："
            f"{len(mptemp_lines)} 對 {len(mpdata_lines)}。"
        )

    temperatures, style = _infer_temperatures(mptemp_lines, mpdata_lines)
    for previous, current in zip(temperatures, temperatures[1:]):
        if current <= previous:
            raise CteToolError(
                "既有 MPTEMP 溫度必須嚴格遞增，否則無法唯一決定內插區間。"
            )

    points = [
        ExistingPoint(
            temperature=temperature,
            cte_expression=mpdata.fields[4].strip(),
            mptemp=mptemp,
            mpdata=mpdata,
        )
        for temperature, mptemp, mpdata in zip(
            temperatures, mptemp_lines, mpdata_lines
        )
    ]
    return lines, points, style


def parse_target_temperatures(text: str) -> list[Decimal]:
    normalised = text.replace("−", "-").replace("＋", "+")
    normalised = TARGET_UNIT_RE.sub(" ", normalised)
    normalised = TARGET_BULLET_RE.sub("", normalised)
    tokens = [
        token
        for token in TARGET_DELIMITER_RE.split(normalised.strip())
        if token
    ]
    if not tokens:
        raise CteToolError("請至少輸入一個要加入的溫度。")

    values: list[Decimal] = []
    seen: set[Decimal] = set()
    for token in tokens:
        try:
            value = _decimal_from_token(token)
        except CteToolError as exc:
            raise CteToolError(
                f"無法辨識溫度項目 {token!r}。"
                "可使用逗號、頓號、分號、空白、Tab 或換行分隔。"
            ) from exc
        if not value.is_finite():
            raise CteToolError(f"溫度必須是有限數值：{token!r}")
        if value not in seen:
            values.append(value)
            seen.add(value)
    return values


def format_decimal(value: Decimal) -> str:
    if value == 0:
        return "0"
    fixed = format(value, "f")
    if "." in fixed:
        fixed = fixed.rstrip("0").rstrip(".")
    return "0" if fixed in {"-0", "+0"} else fixed


def _format_significant(value: Decimal, significant_digits: int = 12) -> str:
    text = format(value, f".{significant_digits}g")
    return text.replace("e", "E")


def _format_engineering(value: Decimal) -> str:
    if value == 0:
        return "0"
    adjusted = value.copy_abs().adjusted()
    exponent = adjusted - (adjusted % 3)
    mantissa = value.scaleb(-exponent)
    return f"{_format_significant(mantissa)}E{exponent:+d}"


def _cte_interpolation_value(expression: str) -> tuple[Decimal, str]:
    normalised = _normalise_apdl_expression(expression)
    constant_value = _try_evaluate_constant(normalised)
    if constant_value is not None:
        return constant_value, normalised

    leading_number = LEADING_NUMBER_RE.match(normalised)
    if leading_number is None:
        raise CteToolError(
            f"無法從 CTE 運算式 {expression!r} 取得內插數值；"
            "含變數的運算式必須以數值開頭。"
        )

    numeric_token = _normalise_number_token(leading_number.group(1))
    return _decimal_from_token(numeric_token), numeric_token


def _interpolate_cte(
    left: ExistingPoint,
    right: ExistingPoint,
    target: Decimal,
) -> str:
    denominator = right.temperature - left.temperature
    if denominator == 0:
        raise CteToolError("內插區間的兩個溫度相同。")
    with localcontext() as context:
        context.prec = 28
        ratio = (target - left.temperature) / denominator

    left_value, left_numeric_token = _cte_interpolation_value(
        left.cte_expression
    )
    right_value, right_numeric_token = _cte_interpolation_value(
        right.cte_expression
    )
    if left_value == right_value:
        return left_numeric_token

    with localcontext() as context:
        context.prec = 28
        result = left_value + ratio * (right_value - left_value)

    left_scientific = SIMPLE_SCIENTIFIC_RE.fullmatch(left_numeric_token)
    right_scientific = SIMPLE_SCIENTIFIC_RE.fullmatch(right_numeric_token)
    if (
        left_scientific
        and right_scientific
        and int(left_scientific.group(2)) == int(right_scientific.group(2))
    ):
        exponent = int(left_scientific.group(2))
        mantissa = result.scaleb(-exponent)
        return f"{_format_significant(mantissa)}E{exponent:+d}"
    return _format_engineering(result)


def _replace_field_token(field: str, token: str, *, align_right: bool = True) -> str:
    width = len(field)
    leading = len(field) - len(field.lstrip())
    trailing = len(field) - len(field.rstrip())
    usable_width = max(width - trailing, len(token))
    if align_right:
        body = token.rjust(usable_width)
    else:
        body = (" " * leading) + token
    return body + (" " * trailing)


def _temperature_expression(temperature: Decimal, style: TemperatureStyle) -> str:
    if style.kind == "offset_expression":
        sign = "+" if temperature >= 0 else "-"
        magnitude = format_decimal(abs(temperature))
        return f"{style.base_token} {sign} {magnitude}"
    if style.kind == "absolute_numeric":
        return format_decimal(style.base_value + temperature)
    return format_decimal(temperature)


def _format_mptemp_field_like(
    template_field: str,
    temperature: Decimal,
    style: TemperatureStyle,
) -> str:
    expression = _temperature_expression(temperature, style)
    if style.kind != "offset_expression":
        return _replace_field_token(template_field, expression)

    layout = MPTEMP_FIELD_LAYOUT_RE.fullmatch(template_field)
    if layout is None:
        return _replace_field_token(template_field, expression)

    (
        leading,
        template_base,
        between_base_and_sign,
        _template_sign,
        before_magnitude,
        template_magnitude,
        trailing,
    ) = layout.groups()
    base_width = len(leading) + len(template_base)
    if temperature < 0:
        trailing_width = len(trailing)
        signed_temperature = "-" + format_decimal(abs(temperature))
        value_width = max(
            len(template_field) - base_width - trailing_width,
            len(signed_temperature),
        )
        return (
            style.base_token.rjust(base_width)
            + signed_temperature.rjust(value_width)
            + trailing
        )

    magnitude_width = len(before_magnitude) + len(template_magnitude)
    magnitude = format_decimal(abs(temperature))
    return (
        style.base_token.rjust(base_width)
        + between_base_and_sign
        + "+"
        + magnitude.rjust(magnitude_width)
        + trailing
    )


def _format_cte_field_like(template_field: str, expression: str) -> str:
    template_layout = CTE_FIELD_LAYOUT_RE.fullmatch(template_field)
    expression_layout = NORMALISED_CTE_RE.fullmatch(
        _normalise_apdl_expression(expression)
    )
    if template_layout is None or expression_layout is None:
        return _replace_field_token(template_field, expression)

    (
        leading,
        template_mantissa,
        before_exponent_marker,
        exponent_marker,
        after_exponent_marker,
        template_exponent,
        _template_suffix,
    ) = template_layout.groups()
    mantissa, exponent, suffix = expression_layout.groups()
    mantissa_width = len(leading) + len(template_mantissa)
    exponent_width = len(template_exponent)
    return (
        mantissa.rjust(mantissa_width)
        + before_exponent_marker
        + exponent_marker.upper()
        + after_exponent_marker
        + exponent.rjust(exponent_width)
        + suffix
    )


def _append_comment(code: str, comment: str, comment_column: int) -> str:
    gap = max(2, comment_column - len(code))
    return code + (" " * gap) + "!" + comment


def _correct_mptemp_numbering(
    text: str,
) -> tuple[str, tuple[MptempNumberingCorrection, ...]]:
    lines = text.splitlines()
    corrections: list[MptempNumberingCorrection] = []
    expected_location = 0

    for line_index, raw in enumerate(lines):
        parsed = _split_command_line(line_index, raw)
        if (
            len(parsed.fields) < 3
            or parsed.fields[0].strip().upper() != "MPTEMP"
            or not parsed.fields[2].strip()
        ):
            continue

        expected_location += 1
        original_location = parsed.fields[1].strip()
        if original_location == str(expected_location):
            continue

        corrections.append(
            MptempNumberingCorrection(
                line_number=line_index + 1,
                original=original_location,
                corrected=expected_location,
            )
        )
        first_comma = raw.find(",")
        second_comma = raw.find(",", first_comma + 1)
        replacement = _replace_field_token(
            parsed.fields[1], str(expected_location)
        )
        lines[line_index] = (
            raw[: first_comma + 1] + replacement + raw[second_comma:]
        )

    if not corrections:
        return text, ()

    line_ending = "\r\n" if "\r\n" in text else "\n"
    result = line_ending.join(lines)
    if text.endswith(("\r", "\n")):
        result += line_ending
    return result, tuple(corrections)


def _new_point_comment(
    temperature: Decimal, reference_comment: str | None
) -> str:
    indent = " "
    if reference_comment is not None:
        match = LEADING_NUMBER_RE.match(reference_comment)
        if match is not None:
            indent = reference_comment[: match.start(1)]
    return f"{indent}{format_decimal(temperature)} C {NEW_MARKER}"


def _comment_column(lines: list[CommandLine], default_column: int) -> int:
    existing_columns = [
        line.raw.index("!") for line in lines if line.comment is not None
    ]
    return max([default_column, *existing_columns])


def _mptemp_temperature_templates(
    points: list[ExistingPoint],
) -> dict[int, str]:
    all_fields = [point.mptemp.fields[2] for point in points]
    fallback = max(all_fields, key=len)
    templates: dict[int, str] = {}
    for sign in (-1, 1):
        matching = [
            point.mptemp.fields[2]
            for point in points
            if (-1 if point.temperature < 0 else 1) == sign
        ]
        templates[sign] = max(matching, key=len) if matching else fallback
    return templates


def _temperature_template(templates: dict[int, str], temperature: Decimal) -> str:
    return templates[-1 if temperature < 0 else 1]


def _normalise_mptemp_line(
    source: CommandLine,
    location: int,
    temperature: Decimal,
    style: TemperatureStyle,
    location_template: str,
    temperature_template: str,
    comment_column: int,
) -> str:
    fields = list(source.fields)
    fields[1] = _replace_field_token(location_template, str(location))
    fields[2] = _format_mptemp_field_like(
        temperature_template, temperature, style
    )
    code = ",".join(fields).rstrip()
    if source.comment is None:
        return code
    return _append_comment(code, source.comment, comment_column)


def _normalise_mpdata_prefix(
    fields: list[str], field_templates: tuple[str, ...]
) -> None:
    for index, template in enumerate(field_templates):
        token = fields[index].strip()
        fields[index] = _replace_field_token(template, token)


def _normalise_mpdata_line(
    source: CommandLine,
    expression: str,
    field_templates: tuple[str, ...],
    value_template: str,
    comment_column: int,
) -> str:
    fields = list(source.fields)
    _normalise_mpdata_prefix(fields, field_templates)
    fields[4] = _format_cte_field_like(value_template, expression)
    code = ",".join(fields).rstrip()
    if source.comment is None:
        return code
    return _append_comment(code, source.comment, comment_column)


def _new_mptemp_line(
    template: CommandLine,
    location: int,
    temperature: Decimal,
    style: TemperatureStyle,
    location_template: str,
    temperature_template: str,
    comment_column: int,
    reference_comment: str | None,
) -> str:
    fields = list(template.fields)
    fields[1] = _replace_field_token(location_template, str(location))
    fields[2] = _format_mptemp_field_like(
        temperature_template, temperature, style
    )
    code = ",".join(fields).rstrip()
    comment = _new_point_comment(temperature, reference_comment)
    return _append_comment(code, comment, comment_column)


def _new_mpdata_line(
    template: CommandLine,
    temperature: Decimal,
    expression: str,
    field_templates: tuple[str, ...],
    value_template: str,
    comment_column: int,
) -> str:
    fields = list(template.fields)
    _normalise_mpdata_prefix(fields, field_templates)
    fields[4] = _format_cte_field_like(value_template, expression)
    code = ",".join(fields).rstrip()
    comment = _new_point_comment(temperature, template.comment)
    return _append_comment(code, comment, comment_column)


def _generate_mptemp_from_mpdata(text: str) -> tuple[str, bool]:
    raw_lines = text.splitlines()
    parsed_lines = [
        _split_command_line(line_index, raw)
        for line_index, raw in enumerate(raw_lines)
    ]

    has_mptemp_data = any(
        len(line.fields) >= 3
        and line.fields[0].strip().upper() == "MPTEMP"
        and bool(line.fields[1].strip())
        and bool(line.fields[2].strip())
        for line in parsed_lines
    )
    if has_mptemp_data:
        return text, False

    has_any_mptemp_command = any(
        line.fields and line.fields[0].strip().upper() == "MPTEMP"
        for line in parsed_lines
    )
    if has_any_mptemp_command:
        return text, False

    mpdata_lines = [
        line
        for line in parsed_lines
        if len(line.fields) >= 2
        and line.fields[0].strip().upper() == "MPDATA"
        and line.fields[1].strip().upper() == "CTEX"
    ]
    if not mpdata_lines:
        return text, False

    temperatures: list[Decimal] = []
    materials: list[str] = []
    for line in mpdata_lines:
        temperature = _parse_leading_comment_temperature(line.comment)
        if temperature is None:
            raise CteToolError(
                f"第 {line.line_index + 1} 列只有 MPDATA,CTEX，"
                "但無法從 ! 後方讀到溫度。"
            )
        temperatures.append(temperature)

        if len(line.fields) < 3 or not line.fields[2].strip():
            raise CteToolError(
                f"第 {line.line_index + 1} 列缺少 MPDATA 材料編號，"
                "無法建立 UIMP。"
            )
        materials.append(line.fields[2].strip())

    material_names = {material.upper() for material in materials}
    if len(material_names) != 1:
        raise CteToolError(
            "只有 MPDATA,CTEX 時，所有資料列必須使用相同材料編號，"
            "才能自動建立單一 UIMP。"
        )

    generated_style = TemperatureStyle(
        kind="offset_expression",
        base_value=GENERATED_TEMPERATURE_BASE,
        base_token=format_decimal(GENERATED_TEMPERATURE_BASE),
    )
    generated_lines = [GENERATED_MPTEMP_RESET_LINE]
    for location, temperature in enumerate(temperatures, start=1):
        location_field = _replace_field_token(
            GENERATED_MPTEMP_LOCATION_FIELD, str(location)
        )
        temperature_template = (
            GENERATED_MPTEMP_NEGATIVE_FIELD
            if temperature < 0
            else GENERATED_MPTEMP_POSITIVE_FIELD
        )
        temperature_field = _format_mptemp_field_like(
            temperature_template, temperature, generated_style
        )
        generated_lines.append(
            ",".join(("MPTEMP", location_field, temperature_field)).rstrip()
        )

    material_field = _replace_field_token(
        GENERATED_UIMP_MATERIAL_FIELD, materials[0]
    )
    generated_lines.append(
        ",".join(
            (
                "UIMP",
                material_field,
                GENERATED_UIMP_REFT_FIELD,
                "",
                "",
                "",
            )
        )
    )
    generated_lines.append("")

    insertion_index = mpdata_lines[0].line_index
    output_lines = (
        raw_lines[:insertion_index]
        + generated_lines
        + raw_lines[insertion_index:]
    )
    result = "\n".join(output_lines)
    if text.endswith(("\n", "\r")):
        result += "\n"
    return result, True


def _find_existing_temperature(
    points: list[ExistingPoint], target: Decimal
) -> ExistingPoint | None:
    return next(
        (point for point in points if _nearly_equal(point.temperature, target)),
        None,
    )


def _find_bracketing_points(
    points: list[ExistingPoint], target: Decimal
) -> tuple[ExistingPoint, ExistingPoint]:
    for left, right in zip(points, points[1:]):
        if left.temperature < target < right.temperature:
            return left, right
    raise CteToolError(
        f"{format_decimal(target)} °C 位於既有溫度範圍外；"
        "本工具只做內插，不做外插。"
    )


def transform_cdb(text: str, targets: list[Decimal]) -> tuple[str, TransformReport]:
    if not text.strip():
        raise CteToolError("請先貼上 MPDATA,CTEX；MPTEMP 可省略。")

    prepared_text, generated_mptemp = _generate_mptemp_from_mpdata(text)
    prepared_text, input_numbering_corrections = _correct_mptemp_numbering(
        prepared_text
    )
    lines, existing_points, style = _collect_table(prepared_text)
    skipped: list[str] = []
    new_points: list[TablePoint] = []

    for target in targets:
        if target == SPECIAL_REQUEST_TEMPERATURE:
            special_match = _find_existing_temperature(
                existing_points, SPECIAL_EXISTING_TEMPERATURE
            )
            if special_match is not None:
                skipped.append("30 → 已有 30.01（依特例視為同一點）")
                continue

        match = _find_existing_temperature(existing_points, target)
        if match is not None:
            skipped.append(f"{format_decimal(target)}（已存在）")
            continue

        left, right = _find_bracketing_points(existing_points, target)
        new_points.append(
            TablePoint(
                temperature=target,
                cte_expression=_interpolate_cte(left, right, target),
                existing=None,
            )
        )

    if len(existing_points) + len(new_points) > MAX_MATERIAL_POINTS:
        raise CteToolError(
            f"加入後共有 {len(existing_points) + len(new_points)} 點，"
            f"超過 MPTEMP/MPDATA 的 {MAX_MATERIAL_POINTS} 點上限。"
        )

    if not new_points:
        report = TransformReport(
            added=(),
            skipped=tuple(skipped),
            point_count=len(existing_points),
            generated_mptemp=generated_mptemp,
            mptemp_numbering_corrections=input_numbering_corrections,
        )
        return prepared_text, report

    merged = [
        TablePoint(
            temperature=point.temperature,
            cte_expression=point.cte_expression,
            existing=point,
        )
        for point in existing_points
    ]
    merged.extend(new_points)
    merged.sort(key=lambda point: point.temperature)

    mptemp_sources = [point.mptemp for point in existing_points]
    mpdata_sources = [point.mpdata for point in existing_points]
    location_template = max(
        (line.fields[1] for line in mptemp_sources), key=len
    )
    temperature_templates = _mptemp_temperature_templates(existing_points)
    mptemp_default_comment_column = max(
        len(line.fields[0])
        + 1
        + len(location_template)
        + 1
        + len(_temperature_template(temperature_templates, point.temperature))
        + 2
        for line, point in zip(mptemp_sources, existing_points)
    )
    mptemp_comment_column = _comment_column(
        mptemp_sources, mptemp_default_comment_column
    )

    mpdata_field_templates = tuple(
        max((line.fields[index] for line in mpdata_sources), key=len)
        for index in range(4)
    )
    mpdata_value_template = max(
        (line.fields[4] for line in mpdata_sources), key=len
    )
    mpdata_default_comment_column = (
        sum(len(field) for field in mpdata_field_templates)
        + len(mpdata_field_templates)
        + len(mpdata_value_template)
        + 2
    )
    mpdata_comment_column = _comment_column(
        mpdata_sources, mpdata_default_comment_column
    )

    insert_before: dict[int, list[str]] = {}
    replacements: dict[int, str] = {}

    for location, point in enumerate(merged, start=1):
        if not point.is_new:
            assert point.existing is not None
            mptemp_temperature_template = _temperature_template(
                temperature_templates, point.temperature
            )
            replacements[point.existing.mptemp.line_index] = _normalise_mptemp_line(
                point.existing.mptemp,
                location,
                point.temperature,
                style,
                location_template,
                mptemp_temperature_template,
                mptemp_comment_column,
            )
            replacements[point.existing.mpdata.line_index] = _normalise_mpdata_line(
                point.existing.mpdata,
                point.cte_expression,
                mpdata_field_templates,
                mpdata_value_template,
                mpdata_comment_column,
            )
            continue

        right_existing = next(
            existing
            for existing in existing_points
            if existing.temperature > point.temperature
        )
        mptemp_temperature_template = _temperature_template(
            temperature_templates, point.temperature
        )
        mptemp_line = _new_mptemp_line(
            right_existing.mptemp,
            location,
            point.temperature,
            style,
            location_template,
            mptemp_temperature_template,
            mptemp_comment_column,
            right_existing.mpdata.comment,
        )
        mpdata_line = _new_mpdata_line(
            right_existing.mpdata,
            point.temperature,
            point.cte_expression,
            mpdata_field_templates,
            mpdata_value_template,
            mpdata_comment_column,
        )
        insert_before.setdefault(right_existing.mptemp.line_index, []).append(mptemp_line)
        insert_before.setdefault(right_existing.mpdata.line_index, []).append(mpdata_line)

    output_lines: list[str] = []
    for line_index, raw in enumerate(lines):
        output_lines.extend(insert_before.get(line_index, ()))
        output_lines.append(replacements.get(line_index, raw))

    result = "\n".join(output_lines)
    if prepared_text.endswith(("\n", "\r")):
        result += "\n"
    result, output_numbering_corrections = _correct_mptemp_numbering(result)
    numbering_corrections = (
        input_numbering_corrections + output_numbering_corrections
    )

    added = tuple(point.temperature for point in sorted(new_points, key=lambda p: p.temperature))
    report = TransformReport(
        added=added,
        skipped=tuple(skipped),
        point_count=len(merged),
        generated_mptemp=generated_mptemp,
        mptemp_numbering_corrections=numbering_corrections,
    )
    return result, report


def _ctex_param_name(number: int) -> str:
    return f"{CTEX_PARAM_PREFIX}{number}"


def _ctex_highlight_color(number: int) -> str:
    return CTEX_HIGHLIGHT_PALETTE[(number - 1) % len(CTEX_HIGHLIGHT_PALETTE)]


def _ctex_row_group(cte_expression: str) -> int | None:
    match = CTEX_SUFFIX_RE.search(cte_expression)
    return int(match.group(1)) if match else None


def apply_ctex_parameters(
    text: str, temperatures: list[Decimal]
) -> tuple[str, CtexParameterReport]:
    if not text.strip():
        raise CteToolError("請先貼上含 MPTEMP／MPDATA,CTEX 的 CDB 內容。")
    if not temperatures:
        raise CteToolError("請至少輸入一個要加上 ctexN 參數的溫度點。")

    prepared_text, generated_mptemp = _generate_mptemp_from_mpdata(text)
    prepared_text, _input_numbering_corrections = _correct_mptemp_numbering(
        prepared_text
    )

    _lines, existing_points, _style = _collect_table(prepared_text)

    position_by_temperature = {
        point.temperature: position
        for position, point in enumerate(existing_points, start=1)
    }

    not_found: list[Decimal] = []
    selected_positions: set[int] = set()
    for target in temperatures:
        match = _find_existing_temperature(existing_points, target)
        if match is None:
            not_found.append(target)
            continue
        selected_positions.add(position_by_temperature[match.temperature])

    if not_found:
        formatted = "、".join(format_decimal(value) for value in not_found)
        raise CteToolError(
            f"找不到溫度點：{formatted}（必須是表格中既有的溫度）。"
        )

    # Current per-row group membership, derived from each row's existing "+ctexN" suffix.
    row_group: dict[int, int] = {}
    for position, point in enumerate(existing_points, start=1):
        group = _ctex_row_group(point.cte_expression)
        if group is not None:
            row_group[position] = group
    group_members: dict[int, set[int]] = {}
    for position, group in row_group.items():
        group_members.setdefault(group, set()).add(position)

    # Same set of points as an existing batch -> reuse its name; otherwise take the next number.
    reused_group = next(
        (
            group
            for group, members in group_members.items()
            if members == selected_positions
        ),
        None,
    )
    max_existing_number = max([0, *row_group.values()])
    group_number = reused_group if reused_group is not None else max_existing_number + 1
    param_name = _ctex_param_name(group_number)

    final_group_by_position = dict(row_group)
    for position in selected_positions:
        final_group_by_position[position] = group_number
    referenced_groups = set(final_group_by_position.values())

    # Strip every existing ctexN declaration block (plus its trailing blank separator
    # line); only the still-referenced groups get rewritten below.
    lines = prepared_text.splitlines()
    kept_lines = []
    index = 0
    while index < len(lines):
        code = _split_command_line(0, lines[index]).code.strip()
        if not CTEX_DECLARATION_LINE_RE.match(code):
            kept_lines.append(lines[index])
            index += 1
            continue
        while index < len(lines) and CTEX_DECLARATION_LINE_RE.match(
            _split_command_line(0, lines[index]).code.strip()
        ):
            index += 1
        if index < len(lines) and lines[index].strip() == "":
            index += 1
    lines = kept_lines

    cleaned_text = "\n".join(lines)
    _cleaned_lines, existing_points, _style = _collect_table(cleaned_text)

    field_by_position: dict[int, str] = {}
    for position, point in enumerate(existing_points, start=1):
        line_index = point.mpdata.line_index
        parsed = _split_command_line(line_index, lines[line_index])
        fields = list(parsed.fields)
        original_field = fields[4]
        if position in selected_positions:
            layout = CTE_FIELD_LAYOUT_RE.fullmatch(original_field)
            if layout is None:
                raise CteToolError(
                    f"第 {line_index + 1} 列的 MPDATA,CTEX 數值不是"
                    "「數字 E 指數」格式，無法套用 ctexN 參數。"
                )
            mantissa = layout.group(2)
            exponent = layout.group(6)
            expression = f"{mantissa}E{exponent}+{param_name}"
            fields[4] = _format_cte_field_like(original_field, expression)
        code = ",".join(fields).rstrip()
        field_by_position[position] = fields[4]
        lines[line_index] = code

    mpdata_line_indexes = [point.mpdata.line_index for point in existing_points]
    existing_comment_columns = [
        lines[idx].index("!")
        for idx in mpdata_line_indexes
        if "!" in lines[idx]
    ]
    needed_width = max(
        (len(lines[idx]) + 2 for idx in mpdata_line_indexes), default=0
    )
    comment_column = max([needed_width, *existing_comment_columns])

    for point in existing_points:
        line_index = point.mpdata.line_index
        if point.mpdata.comment is None:
            continue
        lines[line_index] = _append_comment(
            lines[line_index], point.mpdata.comment, comment_column
        )

    # Groups whose declaration line survived the strip above (i.e. it wasn't the
    # literal "=0 E..." placeholder, e.g. a user-edited real value like
    # "ctex1=10 E-6") already have a declaration in the text; don't duplicate it.
    surviving_declared_groups: set[int] = set()
    for raw in lines:
        code = _split_command_line(0, raw).code.strip()
        match = CTEX_DECLARATION_VALUE_RE.match(code)
        if match is not None:
            surviving_declared_groups.add(int(match.group(1)))

    declarations = []
    for group in sorted(referenced_groups - surviving_declared_groups):
        representative_position = min(
            position
            for position, assigned_group in final_group_by_position.items()
            if assigned_group == group
        )
        layout = CTE_FIELD_LAYOUT_RE.fullmatch(field_by_position[representative_position])
        exponent = layout.group(6) if layout is not None else "-6"
        declarations.append(f"{_ctex_param_name(group)}=0 E{exponent}")

    insertion_index = existing_points[0].mpdata.line_index
    declaration_block = declarations + [""] if declarations else []
    output_lines = (
        lines[:insertion_index] + declaration_block + lines[insertion_index:]
    )

    result = "\n".join(output_lines)
    if prepared_text.endswith(("\n", "\r")):
        result += "\n"

    report = CtexParameterReport(
        point_count=len(selected_positions),
        param_name=param_name,
        reused_existing=reused_group is not None,
        generated_mptemp=generated_mptemp,
    )
    return result, report


@dataclass(frozen=True)
class CurvePoint:
    temperature: Decimal
    base_value: Decimal
    variable_name: str | None


@dataclass
class CurveRecord:
    id: int
    name: str
    points: tuple[CurvePoint, ...]
    variable_values: dict[str, Decimal] = field(default_factory=dict)
    visible: bool = True
    tg_enabled: bool = False
    tg_temperature: Decimal | None = None
    source_text: str = ""


def _extract_curve_points(text: str) -> list[CurvePoint]:
    _lines, existing_points, _style = _collect_table(text)
    points: list[CurvePoint] = []
    for point in existing_points:
        base_value, _token = _cte_interpolation_value(point.cte_expression)
        group = _ctex_row_group(point.cte_expression)
        variable_name = _ctex_param_name(group) if group is not None else None
        points.append(CurvePoint(point.temperature, base_value, variable_name))
    return points


def _resolve_curve_xy(
    points: tuple[CurvePoint, ...],
    compute_variables: bool,
    variable_values: dict[str, Decimal],
) -> tuple[list[float], list[float]]:
    xs: list[float] = []
    ys: list[float] = []
    for point in points:
        extra = (
            variable_values.get(point.variable_name, Decimal(0))
            if (compute_variables and point.variable_name is not None)
            else Decimal(0)
        )
        xs.append(float(point.temperature))
        ys.append(float(point.base_value + extra))
    return xs, ys


def _parse_ctex_declarations(text: str) -> dict[str, Decimal]:
    values: dict[str, Decimal] = {}
    for raw in text.splitlines():
        code = _split_command_line(0, raw).code.strip()
        match = CTEX_DECLARATION_VALUE_RE.match(code)
        if match is None:
            continue
        number, mantissa, exponent = match.groups()
        values[_ctex_param_name(int(number))] = _decimal_from_token(
            f"{mantissa}E{exponent}"
        )
    return values


TRANSFER_BASE_VALUE_DECIMAL_PLACES = 4


def _round_mantissa_text(token: str, decimal_places: int = TRANSFER_BASE_VALUE_DECIMAL_PLACES) -> str:
    """Round a numeric token's mantissa to a fixed number of decimal places.

    Interpolated base values can come back with a dozen+ significant digits;
    that precision isn't meaningful for a CTE curve and makes the generated
    block hard to read, so round just the mantissa (keeping whatever exponent
    or suffix the token already carries).
    """

    quantizer = Decimal(1).scaleb(-decimal_places)
    match = NORMALISED_CTE_RE.fullmatch(_normalise_apdl_expression(token))
    if match is None:
        value = _decimal_from_token(token)
        rounded = value.quantize(quantizer, rounding=ROUND_HALF_UP)
        return format_decimal(rounded)

    mantissa_text, exponent_text, suffix = match.groups()
    mantissa = _decimal_from_token(mantissa_text).quantize(
        quantizer, rounding=ROUND_HALF_UP
    )
    return f"{format_decimal(mantissa)}E{exponent_text}{suffix}"


def _correction_suffix(correction: Decimal) -> str:
    if correction == 0:
        return ""
    if correction > 0:
        return f"+{_format_engineering(correction)}"
    return f"-{_format_engineering(-correction)}"


def transfer_calibration(source_text: str, target_text: str) -> tuple[str, str]:
    """Transfer curve 1's per-point calibration onto curve 2's base values.

    Curve 1 (temp1/ctex1) carries an already-calibrated absolute correction per
    temperature point (its declared ctexN value). That correction has physical
    meaning only at the exact temperature it was calibrated for, so it is never
    interpolated or scaled -- it is copied onto curve 2 unchanged, written as an
    explicit "base+correction" addition rather than a pre-summed number.
    """

    if not source_text.strip():
        raise CteToolError("請輸入曲線1（temp1/ctex1，含校正參數）的 CDB 內容。")
    if not target_text.strip():
        raise CteToolError("請輸入曲線2（temp2/cte2，基礎曲線）的 CDB 內容。")

    prepared_source, _generated_source = _generate_mptemp_from_mpdata(source_text)
    prepared_source, _ = _correct_mptemp_numbering(prepared_source)
    _src_lines, source_points, _src_style = _collect_table(prepared_source)
    source_declared = _parse_ctex_declarations(prepared_source)

    prepared_target, _generated_target = _generate_mptemp_from_mpdata(target_text)
    prepared_target, _ = _correct_mptemp_numbering(prepared_target)
    _tgt_lines, target_points, _tgt_style = _collect_table(prepared_target)
    if len(target_points) < 2:
        raise CteToolError("曲線2（基礎曲線）至少需要兩個溫度點才能內插。")

    temperatures: list[Decimal] = []
    base_tokens: list[str] = []
    suffixes: list[str] = []
    corrected_count = 0
    for point in source_points:
        group = _ctex_row_group(point.cte_expression)
        correction = (
            source_declared.get(_ctex_param_name(group), Decimal(0))
            if group is not None
            else Decimal(0)
        )
        if correction != 0:
            corrected_count += 1

        match = _find_existing_temperature(target_points, point.temperature)
        if match is not None:
            _base_value, base_token = _cte_interpolation_value(match.cte_expression)
        else:
            left, right = _find_bracketing_points(target_points, point.temperature)
            base_token = _interpolate_cte(left, right, point.temperature)

        temperatures.append(point.temperature)
        base_tokens.append(_round_mantissa_text(base_token))
        suffixes.append(_correction_suffix(correction))

    # Right-align the base value of every row to a shared column width -- the
    # interpolated values vary wildly in precision (plain integers up to
    # 13-significant-digit results), so a fixed template field can't anticipate
    # the widest one; it has to be measured from this batch's actual tokens.
    base_width = max(len(token) for token in base_tokens)
    value_tokens = [
        base.rjust(base_width) + suffix
        for base, suffix in zip(base_tokens, suffixes)
    ]
    value_width = max(len(token) for token in value_tokens)
    value_tokens = [token.ljust(value_width) for token in value_tokens]

    material_field = GENERATED_MPDATA_MATERIAL_FIELD
    mpdata_codes = [
        ",".join(
            ("MPDATA", "CTEX", material_field, GENERATED_MPDATA_LOCATION_FIELD, token)
        )
        for token in value_tokens
    ]
    comment_column = max(len(code) for code in mpdata_codes) + 3
    mpdata_lines = [
        _append_comment(code, f" {format_decimal(temperature)} C", comment_column)
        for code, temperature in zip(mpdata_codes, temperatures)
    ]
    result_text, _generated_result = _generate_mptemp_from_mpdata(
        "\n".join(mpdata_lines)
    )

    summary = (
        f"已將曲線1 的校正參數轉移到曲線2，共產生 {len(temperatures)} 個溫度點，"
        f"其中 {corrected_count} 點套用了非零校正。"
    )
    return result_text, summary


def _read_text_file(path: Path) -> str:
    raw = path.read_bytes()
    for encoding in ("utf-8-sig", "utf-8", "cp950"):
        try:
            return raw.decode(encoding)
        except UnicodeDecodeError:
            continue
    raise CteToolError(f"無法用 UTF-8 或 CP950 讀取檔案：{path}")


class LineNumberedText(ttk.Frame):
    def __init__(self, parent: tk.Misc, text_font: tkfont.Font) -> None:
        super().__init__(parent)
        self._redraw_pending = False
        self._column_anchor: tuple[int, int] | None = None
        self._column_ranges: list[tuple[str, str]] = []
        self._column_caret_marks: list[str] = []
        self._column_caret_widgets: list[tk.Frame] = []
        self._column_release_fallback_pending = False
        self.text_font = text_font

        self.gutter = tk.Canvas(
            self,
            width=LINE_NUMBER_GUTTER_WIDTH,
            highlightthickness=0,
            borderwidth=0,
            background=LINE_NUMBER_BACKGROUND,
        )
        self.text = tk.Text(
            self, wrap=tk.NONE, undo=True, font=self.text_font
        )
        self.vertical = ttk.Scrollbar(
            self, orient=tk.VERTICAL, command=self._on_vertical_scroll
        )
        self.horizontal = ttk.Scrollbar(
            self, orient=tk.HORIZONTAL, command=self._on_horizontal_scroll
        )
        self.text.configure(
            yscrollcommand=self._on_text_scroll,
            xscrollcommand=self._on_horizontal_text_scroll,
        )

        self.gutter.grid(row=0, column=0, sticky="ns")
        self.text.grid(row=0, column=1, sticky="nsew")
        self.vertical.grid(row=0, column=2, sticky="ns")
        self.horizontal.grid(row=1, column=1, sticky="ew")
        self.rowconfigure(0, weight=1)
        self.columnconfigure(1, weight=1)

        self.text.bind("<<Modified>>", self._on_modified, add=True)
        self.text.bind("<Configure>", self._schedule_redraw, add=True)
        self.text.bind("<KeyRelease>", self._schedule_redraw, add=True)
        self.text.bind("<ButtonRelease-1>", self._schedule_redraw, add=True)
        self.text.bind("<MouseWheel>", self._schedule_redraw, add=True)
        self.text.bind(
            "<Control-Alt-ButtonPress-1>",
            self._start_column_selection,
            add=True,
        )
        self.text.bind("<B1-Motion>", self._drag_column_selection, add=True)
        self.text.bind(
            "<ButtonRelease-1>", self._finish_column_selection, add=True
        )
        self.text.bind(
            "<ButtonPress-1>", self._clear_column_selection_on_click, add=True
        )
        self._column_key_bindtag = (
            f"{COLUMN_KEY_BINDTAG_PREFIX}{id(self.text)}"
        )
        self.text.bindtags(
            (self._column_key_bindtag, *self.text.bindtags())
        )
        self.text.bind_class(
            self._column_key_bindtag,
            "<KeyPress>",
            self._clear_column_selection_on_keypress,
            add=True,
        )
        self.text.bind_class(
            self._column_key_bindtag,
            "<KeyRelease>",
            self._on_column_key_release,
            add=True,
        )
        self.text.bind(
            "<Control-KeyPress-c>", self._copy_column_selection, add=True
        )
        self.text.bind(
            "<Control-KeyPress-C>", self._copy_column_selection, add=True
        )
        self.text.bind(
            "<Control-KeyPress-x>", self._cut_column_selection, add=True
        )
        self.text.bind(
            "<Control-KeyPress-X>", self._cut_column_selection, add=True
        )
        self.text.bind("<<Copy>>", self._copy_column_selection, add=True)
        self.text.bind("<<Cut>>", self._cut_column_selection, add=True)
        self.text.bind("<Delete>", self._delete_column_selection, add=True)
        self.text.bind("<BackSpace>", self._delete_column_selection, add=True)
        self.text.bind("<Escape>", self._cancel_column_selection, add=True)
        self.text.edit_modified(False)
        self._schedule_redraw()

    @staticmethod
    def _line_and_column(index: str) -> tuple[int, int]:
        line, column = index.split(".", 1)
        return int(line), int(column)

    def _index_from_pointer(self, event: tk.Event) -> str:
        return self.text.index(f"@{event.x},{event.y}")

    def _start_column_selection(self, event: tk.Event) -> str:
        self.clear_column_selection()
        self.text.focus_set()
        pointer_index = self._index_from_pointer(event)
        self._column_anchor = self._line_and_column(pointer_index)
        self.text.mark_set(tk.INSERT, pointer_index)
        return "break"

    def _drag_column_selection(self, event: tk.Event) -> str | None:
        if self._column_anchor is None:
            return None
        self._update_column_selection(self._index_from_pointer(event))
        return "break"

    def _finish_column_selection(self, event: tk.Event) -> str | None:
        if self._column_anchor is None:
            return None
        self._update_column_selection(self._index_from_pointer(event))
        self._column_anchor = None
        return "break"

    def _update_column_selection(self, pointer_index: str) -> None:
        if self._column_anchor is None:
            return

        anchor_line, anchor_column = self._column_anchor
        pointer_line, pointer_column = self._line_and_column(pointer_index)
        first_line, last_line = sorted((anchor_line, pointer_line))
        first_column, last_column = sorted(
            (anchor_column, pointer_column)
        )

        self.text.tag_remove(tk.SEL, "1.0", tk.END)
        self._column_ranges.clear()
        self._clear_column_carets()
        if first_column == last_column:
            self._create_column_carets(
                first_line, last_line, first_column
            )
            return

        for line_number in range(first_line, last_line + 1):
            line_end = self.text.index(f"{line_number}.end")
            line_length = self._line_and_column(line_end)[1]
            range_start_column = min(first_column, line_length)
            range_end_column = min(last_column, line_length)
            if range_start_column == range_end_column:
                continue
            range_start = f"{line_number}.{range_start_column}"
            range_end = f"{line_number}.{range_end_column}"
            self.text.tag_add(tk.SEL, range_start, range_end)
            self._column_ranges.append((range_start, range_end))

        if self._column_ranges:
            self.text.mark_set(tk.INSERT, self._column_ranges[-1][1])
        self.text.see(pointer_index)

    def _create_column_carets(
        self, first_line: int, last_line: int, column: int
    ) -> None:
        caret_positions: list[str] = []
        for line_number in range(first_line, last_line + 1):
            line_end = self.text.index(f"{line_number}.end")
            line_length = self._line_and_column(line_end)[1]
            caret_column = min(column, line_length)
            caret_positions.append(f"{line_number}.{caret_column}")
        self._set_column_carets(caret_positions)

    def _set_column_carets(self, caret_positions: list[str]) -> None:
        self._clear_column_carets()
        for caret_index, caret_position in enumerate(caret_positions):
            mark_name = f"{COLUMN_CARET_MARK_PREFIX}{caret_index}"
            self.text.mark_set(mark_name, caret_position)
            self.text.mark_gravity(mark_name, tk.RIGHT)
            caret_widget = tk.Frame(
                self.text,
                width=COLUMN_CARET_WIDTH,
                background=self.text.cget("insertbackground"),
                borderwidth=0,
            )
            self._column_caret_marks.append(mark_name)
            self._column_caret_widgets.append(caret_widget)

        if self._column_caret_marks:
            self.text.mark_set(tk.INSERT, self._column_caret_marks[-1])
            self.text.see(self._column_caret_marks[-1])
        self._redraw_column_carets()

    def _redraw_column_carets(self) -> None:
        for mark_name, caret_widget in zip(
            self._column_caret_marks,
            self._column_caret_widgets,
        ):
            bounding_box = self.text.bbox(mark_name)
            if bounding_box is None:
                caret_widget.place_forget()
                continue
            x_position, y_position, _width, height = bounding_box
            caret_widget.place(
                x=x_position,
                y=y_position,
                width=COLUMN_CARET_WIDTH,
                height=height,
            )
            caret_widget.lift()

    def _clear_column_carets(self) -> None:
        for mark_name in self._column_caret_marks:
            self.text.mark_unset(mark_name)
        for caret_widget in self._column_caret_widgets:
            caret_widget.destroy()
        self._column_caret_marks.clear()
        self._column_caret_widgets.clear()

    def clear_column_selection(self) -> None:
        if (
            not self._column_ranges
            and not self._column_caret_marks
            and self._column_anchor is None
        ):
            return
        self.text.tag_remove(tk.SEL, "1.0", tk.END)
        self._column_anchor = None
        self._column_ranges.clear()
        self._clear_column_carets()
        self._column_release_fallback_pending = False

    def _clear_column_selection_on_click(self, _event: tk.Event) -> None:
        if self._column_anchor is None:
            self.clear_column_selection()

    @staticmethod
    def _is_space_keyevent(event: tk.Event) -> bool:
        return (
            event.keycode == WINDOWS_VK_SPACE
            or event.keysym in {"space", "KP_Space"}
            or event.char == " "
        )

    def _clear_column_selection_on_keypress(
        self, event: tk.Event
    ) -> str | None:
        if not self._column_ranges and not self._column_caret_marks:
            return None
        if self._is_space_keyevent(event):
            self._column_release_fallback_pending = False
            return self._on_column_space(event)
        if event.keysym in {"Delete", "BackSpace"}:
            self._column_release_fallback_pending = False
            return self._delete_column_selection(event)
        if event.keysym == "Escape":
            self._column_release_fallback_pending = False
            return self._cancel_column_selection(event)
        if event.keysym in {
            "Control_L",
            "Control_R",
            "Alt_L",
            "Alt_R",
            "Shift_L",
            "Shift_R",
        }:
            return None
        if (
            event.state & TK_CONTROL_STATE_MASK
            and event.keysym.lower() in {"c", "x"}
        ):
            return None
        if (
            event.keycode == WINDOWS_VK_PACKET
            or event.keysym in UNRESOLVED_KEYSYMS
        ):
            # Windows/Tk reports Space's KeyPress as this unresolved
            # placeholder whenever Alt is still held (it avoids colliding
            # with the OS "Alt+Space opens system menu" accelerator), even
            # though Ctrl is also held. The matching KeyRelease always
            # resolves correctly, so defer judgement to
            # _on_column_key_release instead of discarding the carets here.
            self._column_release_fallback_pending = True
            return None
        self._column_release_fallback_pending = False
        self.clear_column_selection()
        return None

    def _on_column_key_release(self, event: tk.Event) -> str | None:
        if not self._column_release_fallback_pending:
            return None
        self._column_release_fallback_pending = False
        if not self._column_ranges and not self._column_caret_marks:
            return None
        if self._is_space_keyevent(event):
            return self._on_column_space(event)
        if event.keysym in {"Delete", "BackSpace"}:
            return self._delete_column_selection(event)
        return None

    def _on_column_space(self, _event: tk.Event) -> str | None:
        if self._column_ranges:
            return self._insert_at_column_ranges(" ")
        if self._column_caret_marks:
            return self._insert_at_column_carets(" ")
        return None

    def _copy_column_selection(self, _event: tk.Event) -> str | None:
        if not self._column_ranges:
            return None
        selected_lines = [
            self.text.get(range_start, range_end)
            for range_start, range_end in self._column_ranges
        ]
        self.text.clipboard_clear()
        self.text.clipboard_append("\n".join(selected_lines))
        return "break"

    def _cut_column_selection(self, event: tk.Event) -> str | None:
        if not self._column_ranges:
            return None
        self._copy_column_selection(event)
        return self._delete_column_selection(event)

    def _delete_column_selection(self, event: tk.Event) -> str | None:
        if not self._column_ranges and not self._column_caret_marks:
            return None
        if self._column_caret_marks:
            return self._delete_at_column_carets(event.keysym)
        self.text.edit_separator()
        for range_start, range_end in reversed(self._column_ranges):
            self.text.delete(range_start, range_end)
        self.text.edit_separator()
        self.clear_column_selection()
        return "break"

    def _insert_at_column_carets(self, value: str) -> str:
        self.text.edit_separator()
        for mark_name in reversed(self._column_caret_marks):
            self.text.insert(mark_name, value)
        self.text.edit_separator()
        self._redraw_column_carets()
        return "break"

    def _insert_at_column_ranges(self, value: str) -> str:
        caret_positions = [
            range_start for range_start, _range_end in self._column_ranges
        ]
        self.text.tag_remove(tk.SEL, "1.0", tk.END)
        self._column_ranges.clear()
        self._set_column_carets(caret_positions)
        return self._insert_at_column_carets(value)

    def _delete_at_column_carets(self, keysym: str) -> str:
        self.text.edit_separator()
        for mark_name in reversed(self._column_caret_marks):
            caret_index = self.text.index(mark_name)
            line_number, column = self._line_and_column(caret_index)
            if keysym == "BackSpace":
                if column == 0:
                    continue
                self.text.delete(f"{line_number}.{column - 1}", caret_index)
                continue
            line_length = self._line_and_column(
                self.text.index(f"{line_number}.end")
            )[1]
            if column < line_length:
                self.text.delete(caret_index, f"{line_number}.{column + 1}")
        self.text.edit_separator()
        self._redraw_column_carets()
        return "break"

    def _cancel_column_selection(self, _event: tk.Event) -> str | None:
        if (
            not self._column_ranges
            and not self._column_caret_marks
            and self._column_anchor is None
        ):
            return None
        self.clear_column_selection()
        return "break"

    def _on_vertical_scroll(self, *args: str) -> None:
        self.text.yview(*args)
        self._schedule_redraw()

    def _on_horizontal_scroll(self, *args: str) -> None:
        self.text.xview(*args)
        self._schedule_redraw()

    def _on_text_scroll(self, first: str, last: str) -> None:
        self.vertical.set(first, last)
        self._schedule_redraw()

    def _on_horizontal_text_scroll(self, first: str, last: str) -> None:
        self.horizontal.set(first, last)
        self._schedule_redraw()

    def _on_modified(self, _event: tk.Event | None = None) -> None:
        if self.text.edit_modified():
            self.text.edit_modified(False)
            self._schedule_redraw()

    def _schedule_redraw(self, _event: tk.Event | None = None) -> None:
        if self._redraw_pending:
            return
        self._redraw_pending = True
        self.after_idle(self.redraw_line_numbers)

    def redraw_line_numbers(self) -> None:
        self._redraw_pending = False
        self.gutter.delete("all")
        current_index = self.text.index("@0,0")
        line_number = int(current_index.split(".", 1)[0])
        final_line = int(self.text.index("end-1c").split(".", 1)[0])
        digit_count = max(MIN_LINE_NUMBER_DIGITS, len(str(final_line)))
        required_width = max(
            LINE_NUMBER_GUTTER_WIDTH,
            self.text_font.measure("0" * digit_count)
            + LINE_NUMBER_GUTTER_PADDING,
        )
        if int(self.gutter.cget("width")) != required_width:
            self.gutter.configure(width=required_width)
        gutter_right = required_width - (LINE_NUMBER_GUTTER_PADDING // 2)

        while True:
            line_index = f"{line_number}.0"
            display_info = self.text.dlineinfo(line_index)
            if display_info is None:
                break
            y_position = display_info[1]
            self.gutter.create_text(
                gutter_right,
                y_position,
                anchor="ne",

    def _on_text_zoom(self, event: tk.Event) -> str:
        if event.delta == 0:
            return "break"
        direction = 1 if event.delta > 0 else -1
        current_size = int(self.text_font.cget("size"))
        new_size = max(
            MIN_TEXT_FONT_SIZE,
            min(
                MAX_TEXT_FONT_SIZE,
                current_size + direction * TEXT_ZOOM_STEP,
            ),
        )
        if new_size != current_size:
            self.text_font.configure(size=new_size)
            self.separator_font.configure(size=new_size)
            self.editor._schedule_redraw()
            self.status_var.set(f"文字大小：{new_size} pt。")
        return "break"

    def _initialize_pane_ratio(self, event: tk.Event) -> None:
        if self._pane_ratio_initialized:
            return
        panes = event.widget
        if not isinstance(panes, ttk.Panedwindow) or panes.winfo_width() <= 1:
            return
        self._pane_ratio_initialized = True
        panes.sashpos(0, round(panes.winfo_width() * TARGET_PANEL_RATIO))

    def _initialize_target_vertical_pane_ratio(self, event: tk.Event) -> None:
        if self._target_vertical_pane_ratio_initialized:
            return
        panes = event.widget
        if not isinstance(panes, ttk.Panedwindow) or panes.winfo_height() <= 1:
            return
        self._target_vertical_pane_ratio_initialized = True
        panes.sashpos(0, round(panes.winfo_height() * 0.5))

    @staticmethod
    def _split_editor_sections(text: str) -> tuple[str, str | None]:
        separator_index = text.find(OUTPUT_SEPARATOR)
        if separator_index < 0:
            return text, None
        source = text[:separator_index].rstrip("\r\n")
        output = text[separator_index + len(OUTPUT_SEPARATOR) :].lstrip("\r\n")
        return source, output

    def _set_editor(self, source: str, output: str | None = None) -> None:
        self.editor.clear_column_selection()
        if output is None:
            content = source
        else:
            clean_source = source.rstrip("\r\n")
            clean_output = output.rstrip("\r\n")
            content = (
                clean_source
                + "\n\n"
                + OUTPUT_SEPARATOR
                + "\n"
                + clean_output
            )

        self.combined_text.delete("1.0", tk.END)
        self.combined_text.insert("1.0", content)
        self.combined_text.tag_remove("output_separator", "1.0", tk.END)
        separator_start = self.combined_text.search(
            OUTPUT_SEPARATOR, "1.0", stopindex=tk.END
        )
        if separator_start:
            separator_end = f"{separator_start}+{len(OUTPUT_SEPARATOR)}c"
            self.combined_text.tag_add(
                "output_separator", separator_start, separator_end
            )
        self.combined_text.edit_reset()
        self.editor.redraw_line_numbers()
        self._highlight_ctex_variables()

    def _highlight_ctex_variables(self, _event: tk.Event | None = None) -> None:
        text_widget = self.combined_text
        for tag in text_widget.tag_names():
            if tag.startswith(CTEX_HIGHLIGHT_TAG_PREFIX):
                text_widget.tag_remove(tag, "1.0", tk.END)

        content = text_widget.get("1.0", "end-1c")
        for match in CTEX_TOKEN_RE.finditer(content):
            number = int(match.group(1))
            tag = f"{CTEX_HIGHLIGHT_TAG_PREFIX}{number}"
            text_widget.tag_configure(
                tag, background=_ctex_highlight_color(number)
            )
            text_widget.tag_add(
                tag, f"1.0+{match.start()}c", f"1.0+{match.end()}c"
            )

    def load_path(self, path: Path) -> None:
        try:
            text = _read_text_file(path)
        except (OSError, CteToolError) as exc:
            messagebox.showerror(APP_TITLE, str(exc), parent=self.root)
            return
        self._set_editor(text)
        self.status_var.set(f"已載入：{path}")

    def paste_input(self) -> None:
        try:
            clipboard_text = self.root.clipboard_get()
        except tk.TclError:
            messagebox.showwarning(APP_TITLE, "剪貼簿中沒有文字。", parent=self.root)
            return
        self._set_editor(clipboard_text)
        self.status_var.set("已將剪貼簿文字貼到 CDB 編輯區。")

    def _current_base_text(self) -> str:
        editor_content = self.combined_text.get("1.0", "end-1c")
        source, output = self._split_editor_sections(editor_content)
        if output is not None and output.strip():
            return output
        return source

    def transform(self) -> None:
        try:
            target_text = self.target_text.get("1.0", "end-1c")
            targets = (
                parse_target_temperatures(target_text)
                if target_text.strip()
                else []
            )
            base = self._current_base_text()
            result, report = transform_cdb(base, targets)
        except CteToolError as exc:
            self.status_var.set(f"輸入錯誤：{exc}")
            messagebox.showerror(APP_TITLE, str(exc), parent=self.root)
            return
        self._set_editor(base, result)
        self._record_curve_from_text(result)
        self.status_var.set(report.summary())
        if report.mptemp_numbering_corrections:
            messagebox.showwarning(
                APP_TITLE,
                report.mptemp_numbering_notice(),
                parent=self.root,
            )

    def generate_ctex_parameters(self) -> None:
        try:
            target_text = self.ctex_text.get("1.0", "end-1c")
            targets = parse_target_temperatures(target_text)
            base = self._current_base_text()
            result, report = apply_ctex_parameters(base, targets)
        except CteToolError as exc:
            self.status_var.set(f"輸入錯誤：{exc}")
            messagebox.showerror(APP_TITLE, str(exc), parent=self.root)
            return
        self._set_editor(base, result)
        self.status_var.set(report.summary())

    def _find_matching_curve_record(
        self, points: list[CurvePoint]
    ) -> CurveRecord | None:
        signature = tuple((point.temperature, point.base_value) for point in points)
        for record in self._curve_records:
            existing_signature = tuple(
                (point.temperature, point.base_value) for point in record.points
            )
            if existing_signature == signature:
                return record
        return None

    def _create_curve_record(self, text: str, points: list[CurvePoint]) -> CurveRecord:
        declared = _parse_ctex_declarations(text)
        variable_names = {
            point.variable_name for point in points if point.variable_name is not None
        }
        variable_values = {
            name: declared.get(name, Decimal(0)) for name in variable_names
        }
        record = CurveRecord(
            id=self._next_curve_id,
            name=f"curve{self._next_curve_id}",
            points=tuple(points),
            variable_values=variable_values,
            source_text=text,
        )
        self._next_curve_id += 1
        self._curve_records.append(record)
        return record

    def _append_curve_record(self, text: str, points: list[CurvePoint]) -> None:
        existing_record = self._find_matching_curve_record(points)
        if existing_record is not None:
            # Same temperature/CTE data as an existing curve -- only a variable
            # name was added/changed, so update it in place instead of
            # appending a near-duplicate curve.
            declared = _parse_ctex_declarations(text)
            variable_names = {
                point.variable_name for point in points if point.variable_name is not None
            }
            existing_record.points = tuple(points)
            existing_record.source_text = text
            for name in variable_names:
                if name in declared:
                    # The declaration in the text is authoritative on capture.
                    existing_record.variable_values[name] = declared[name]
                else:
                    existing_record.variable_values.setdefault(name, Decimal(0))
        else:
            self._create_curve_record(text, points)

        self.chart_panel.refresh()

    def _record_curve_from_text(self, text: str) -> None:
        try:
            points = _extract_curve_points(text)
        except CteToolError:
            return
        self._append_curve_record(text, points)

    def capture_current_curve(self) -> None:
        text = self._current_base_text()
        try:
            points = _extract_curve_points(text)
        except CteToolError as exc:
            self.status_var.set(f"輸入錯誤：{exc}")
            messagebox.showerror(APP_TITLE, str(exc), parent=self.root)
            return
        self._append_curve_record(text, points)
        self.status_var.set("已擷取目前曲線並加入圖表。")
        self.show_chart_tab()

    def add_new_curve(self) -> None:
        text = self._current_base_text()
        try:
            points = _extract_curve_points(text)
        except CteToolError as exc:
            self.status_var.set(f"輸入錯誤：{exc}")
            messagebox.showerror(APP_TITLE, str(exc), parent=self.root)
            return
        self._create_curve_record(text, points)
        self.chart_panel.refresh()
        self.status_var.set("已新增一條曲線。")
        self.show_chart_tab()

    def show_chart_tab(self) -> None:
        self.notebook.select(self._chart_tab)

    def _on_notebook_tab_changed(self, _event: tk.Event) -> None:
        self.transfer_panel.refresh_curve_choices()

    def copy_output(self) -> None:
        editor_content = self.combined_text.get("1.0", "end-1c")
        _source, result = self._split_editor_sections(editor_content)
        if result is None or not result.strip():
            messagebox.showwarning(APP_TITLE, "目前沒有可複製的輸出。", parent=self.root)
            return
        self.root.clipboard_clear()
        self.root.clipboard_append(result)
        self.root.update_idletasks()
        self.status_var.set("結果已複製到剪貼簿。")

    @staticmethod
    def _on_keypad_decimal(event: tk.Event) -> str | None:
        is_keypad_decimal = (
            event.keycode in {WINDOWS_VK_SEPARATOR, WINDOWS_VK_DECIMAL}
            or event.keysym in KEYPAD_DECIMAL_KEYSYMS
        )
        if not is_keypad_decimal:
            return None

        widget = event.widget
        if not isinstance(widget, tk.Text):
            return None
        if widget.tag_ranges(tk.SEL):
            widget.delete(tk.SEL_FIRST, tk.SEL_LAST)
        widget.insert(tk.INSERT, ".")
        return "break"

    def _on_transform_shortcut(self, _event: tk.Event) -> str:
        if self.notebook.select() == str(self._transfer_tab):
            self.transfer_panel._compute()
        else:
            self.transform()
        return "break"

    @staticmethod
    def _make_scrollable_list(parent: ttk.Frame, *, height: int) -> ttk.Frame:
        canvas = tk.Canvas(parent, height=height, highlightthickness=0)
        scrollbar = ttk.Scrollbar(parent, orient=tk.VERTICAL, command=canvas.yview)
        inner = ttk.Frame(canvas)
        inner.bind(
            "<Configure>",
            lambda _e: canvas.configure(scrollregion=canvas.bbox("all")),
        )
        window_id = canvas.create_window((0, 0), window=inner, anchor="nw")
        canvas.configure(yscrollcommand=scrollbar.set)
        canvas.bind(
            "<Configure>", lambda e: canvas.itemconfig(window_id, width=e.width)
        )
        canvas.pack(side=tk.LEFT, fill=tk.BOTH, expand=True, padx=(6, 0), pady=6)
        scrollbar.pack(side=tk.LEFT, fill=tk.Y, pady=6)
        return inner

    @staticmethod
    def _axis_padding(values: list[float]) -> float:
        span = max(values) - min(values)
        if span > 0:
            return span * 0.1
        magnitude = max(abs(value) for value in values)
        return magnitude * 0.1 if magnitude > 0 else 1.0

    @staticmethod
    def _variable_exponent(record: CurveRecord, name: str) -> int:
        for point in record.points:
            if point.variable_name == name and point.base_value != 0:
                adjusted = point.base_value.copy_abs().adjusted()
                return int(adjusted - (adjusted % 3))
        return -6

        menu.tk_popup(event.x_root, event.y_root)

    def _copy_figure_to_clipboard(self) -> None:
        try:
            import io

            import win32clipboard
            from PIL import Image
        except ImportError:
            messagebox.showwarning(
                APP_TITLE,
                "此電腦缺少 pywin32 或 Pillow，無法把圖片複製到剪貼簿（檔案仍已儲存）。",
                parent=self,
            )
            return

        buffer = io.BytesIO()
        self._fig.savefig(
            buffer,
            format="png",
            bbox_inches="tight",
            pad_inches=CHART_CLIPBOARD_PAD_INCHES,
        )
        buffer.seek(0)
        bitmap = io.BytesIO()
        Image.open(buffer).convert("RGB").save(bitmap, "BMP")
        dib_data = bitmap.getvalue()[14:]  # strip the BMP file header for CF_DIB

        win32clipboard.OpenClipboard()
        try:
            win32clipboard.EmptyClipboard()
            win32clipboard.SetClipboardData(win32clipboard.CF_DIB, dib_data)
        finally:
            win32clipboard.CloseClipboard()
        self.app.status_var.set("圖表已複製到剪貼簿。")

    def refresh(self) -> None:
        self._redraw_chart()
        self._rebuild_curve_list_panel()

    def _redraw_chart(self) -> None:
        self._fig.clear()
        ax = self._fig.add_subplot(111)
        compute = self.app._compute_variables_var.get()
        all_x: list[float] = [0.0]
        all_y: list[float] = [0.0]
        for record in self.app._curve_records:
            if not record.visible:
                continue
            xs, ys = _resolve_curve_xy(
                record.points, compute, record.variable_values
            )
            if not xs:
                continue
            (line,) = ax.plot(xs, ys, marker="o", lw=1.4, ms=4, label=record.name)
            all_x.extend(xs)
            all_y.extend(ys)

            if record.tg_enabled and record.tg_temperature is not None:
                tg_value = float(record.tg_temperature)
                ax.axvline(
                    tg_value,
                    color=line.get_color(),
                    ls="--",
                    lw=1.2,
                    label=f"{record.name} Tg={format_decimal(record.tg_temperature)}°C",
                )
                all_x.append(tg_value)

        ax.axhline(0, color="#999999", lw=0.8, ls="--", zorder=0)
        ax.axvline(0, color="#999999", lw=0.8, ls="--", zorder=0)
        ax.scatter(
            [0.0], [0.0], marker="o", s=50, color="red", zorder=5, label="原點 (0, 0)"
        )

        x_pad = self._axis_padding(all_x)
        y_pad = self._axis_padding(all_y)
        ax.set_xlim(min(all_x) - x_pad, max(all_x) + x_pad)
        ax.set_ylim(min(all_y) - y_pad, max(all_y) + y_pad)

        ax.set_xlabel("Temperature (°C)", fontsize=11)
        ax.set_ylabel("CTE", fontsize=11)
        ax.grid(True, alpha=0.3)
        ax.legend(fontsize=9, loc="best", framealpha=0.85)
        self._fig.tight_layout()
        self._canvas.draw()

    def _rebuild_curve_list_panel(self) -> None:
        for child in self._curve_list_frame.winfo_children():
            child.destroy()
        records = self.app._curve_records
        for index, record in enumerate(records):
            block = ttk.Frame(self._curve_list_frame)
            block.pack(fill=tk.X, pady=(2, 4), padx=2)

            header = ttk.Frame(block)
            header.pack(fill=tk.X)

            visible_var = tk.BooleanVar(value=record.visible)

            def toggle_visible(record=record, visible_var=visible_var) -> None:
                record.visible = visible_var.get()
                self._redraw_chart()

            ttk.Checkbutton(
                header, variable=visible_var, command=toggle_visible
            ).pack(side=tk.LEFT)

            name_var = tk.StringVar(value=record.name)

            def commit_name(_event=None, record=record, name_var=name_var) -> None:
                record.name = name_var.get().strip() or record.name
                self._redraw_chart()

            name_entry = ttk.Entry(header, textvariable=name_var, width=14)
            name_entry.pack(side=tk.LEFT, fill=tk.X, expand=True, padx=(2, 4))
            name_entry.bind("<Return>", commit_name)
            name_entry.bind("<FocusOut>", commit_name)

            def delete_curve(record=record) -> None:
                self.app._curve_records.remove(record)
                self.refresh()

            ttk.Button(header, text="×", width=2, command=delete_curve).pack(
                side=tk.LEFT
            )

            for name in sorted(record.variable_values):
                exponent = self._variable_exponent(record, name)
                var_row = ttk.Frame(block)
                var_row.pack(fill=tk.X, padx=(22, 0), pady=(2, 0))
                ttk.Label(var_row, text=name, width=10).pack(side=tk.LEFT)
                mantissa = record.variable_values[name].scaleb(-exponent)
                value_var = tk.StringVar(value=format_decimal(mantissa))
                entry = ttk.Entry(var_row, textvariable=value_var, width=10)
                entry.pack(side=tk.LEFT)
                ttk.Label(var_row, text=f"E{exponent:+d}").pack(side=tk.LEFT, padx=(2, 0))

                def commit_value(
                    _event=None,
                    record=record,
                    name=name,
                    value_var=value_var,
                    exponent=exponent,
                ) -> None:
                    try:
                        mantissa = _decimal_from_token(value_var.get())
                    except CteToolError:
                        value_var.set(
                            format_decimal(record.variable_values[name].scaleb(-exponent))
                        )
                        return
                    record.variable_values[name] = mantissa.scaleb(exponent)
                    self._redraw_chart()

                entry.bind("<Return>", commit_value)
                entry.bind("<FocusOut>", commit_value)

            tg_row = ttk.Frame(block)
            tg_row.pack(fill=tk.X, padx=(22, 0), pady=(4, 0))

            tg_enabled_var = tk.BooleanVar(value=record.tg_enabled)

            def toggle_tg(record=record, tg_enabled_var=tg_enabled_var) -> None:
                record.tg_enabled = tg_enabled_var.get()
                self._redraw_chart()

            ttk.Checkbutton(
                tg_row, text="Tg", variable=tg_enabled_var, command=toggle_tg
            ).pack(side=tk.LEFT)

            tg_value_var = tk.StringVar(
                value=format_decimal(record.tg_temperature)
                if record.tg_temperature is not None
                else ""
            )

            def commit_tg(
                _event=None, record=record, tg_value_var=tg_value_var
            ) -> None:
                text = tg_value_var.get().strip()
                if not text:
                    record.tg_temperature = None
                else:
                    try:
                        record.tg_temperature = _decimal_from_token(text)
                    except CteToolError:
                        tg_value_var.set(
                            format_decimal(record.tg_temperature)
                            if record.tg_temperature is not None
                            else ""
                        )
                        return
                self._redraw_chart()

            tg_entry = ttk.Entry(tg_row, textvariable=tg_value_var, width=10)
            tg_entry.pack(side=tk.LEFT, padx=(4, 0))
            tg_entry.bind("<Return>", commit_tg)
            tg_entry.bind("<FocusOut>", commit_tg)
            ttk.Label(tg_row, text="°C").pack(side=tk.LEFT, padx=(2, 0))

            if index < len(records) - 1:
                ttk.Separator(self._curve_list_frame, orient=tk.HORIZONTAL).pack(
                    fill=tk.X, pady=(0, 4)
                )


class TransferPanel(ttk.Frame):
    def __init__(self, master: tk.Misc, app: CtePointApp) -> None:
        super().__init__(master)
        self.app = app

        options = ttk.Frame(self)
        options.pack(fill=tk.X, padx=8, pady=(8, 4))
        ttk.Button(
            options, text="計算並產生 temp2' / cte2'", command=self._compute
        ).pack(side=tk.LEFT)

        curves_panes = ttk.Panedwindow(self, orient=tk.HORIZONTAL)
        curves_panes.pack(fill=tk.BOTH, expand=True, padx=8, pady=(0, 4))

        source_frame = ttk.LabelFrame(
            curves_panes, text="曲線1：temp1 / ctex1（含校正參數，來源）"
        )
        target_frame = ttk.LabelFrame(
            curves_panes, text="曲線2：temp2 / cte2（基礎曲線，待套用校正）"
        )
        curves_panes.add(source_frame, weight=1)
        curves_panes.add(target_frame, weight=1)

        self.source_choice_var, self.source_combobox, self.source_editor = (
            self._build_curve_input(source_frame)
        )
        self.target_choice_var, self.target_combobox, self.target_editor = (
            self._build_curve_input(target_frame)
        )

        result_frame = ttk.LabelFrame(
            self, text="計算結果：temp2' / cte2'（完整 CDB 區塊）"
        )
        result_frame.pack(fill=tk.BOTH, expand=True, padx=8, pady=(0, 8))
        result_controls = ttk.Frame(result_frame)
        result_controls.pack(fill=tk.X, padx=6, pady=(6, 0))
        ttk.Button(result_controls, text="複製結果", command=self._copy_result).pack(
            side=tk.LEFT
        )
        self.result_editor = LineNumberedText(result_frame, self.app.text_font)
        self.result_editor.pack(fill=tk.BOTH, expand=True, padx=6, pady=6)
        self.result_text_widget = self.result_editor.text
        self.app._bind_common_text_shortcuts(self.result_text_widget)

        self.refresh_curve_choices()

    def _build_curve_input(
        self, parent: ttk.LabelFrame
    ) -> tuple[tk.StringVar, ttk.Combobox, LineNumberedText]:
        picker_row = ttk.Frame(parent)
        picker_row.pack(fill=tk.X, padx=6, pady=(6, 2))
        ttk.Label(picker_row, text="從已擷取曲線載入：").pack(side=tk.LEFT)
        choice_var = tk.StringVar()
        combobox = ttk.Combobox(
            picker_row, textvariable=choice_var, state="readonly", width=16
        )
        combobox.pack(side=tk.LEFT, padx=(2, 4))

        editor = LineNumberedText(parent, self.app.text_font)
        self.app._bind_common_text_shortcuts(editor.text)

        def load_selected(_event=None, choice_var=choice_var, editor=editor) -> None:
            self._load_curve(choice_var.get(), editor)

        ttk.Button(picker_row, text="載入", command=load_selected).pack(side=tk.LEFT)

        editor.pack(fill=tk.BOTH, expand=True, padx=6, pady=(0, 6))
        return choice_var, combobox, editor

    def _load_curve(self, name: str, editor: LineNumberedText) -> None:
        record = next((r for r in self.app._curve_records if r.name == name), None)
        if record is None:
            messagebox.showwarning(APP_TITLE, "請先在清單中選擇一條曲線。", parent=self)
            return
        if not record.source_text.strip():
            messagebox.showwarning(
                APP_TITLE, f"曲線「{record.name}」沒有可載入的 CDB 文字。", parent=self
            )
            return
        text_widget = editor.text
        text_widget.delete("1.0", tk.END)
        text_widget.insert("1.0", record.source_text)
        editor.redraw_line_numbers()

    def refresh_curve_choices(self) -> None:
        names = [record.name for record in self.app._curve_records]
        self.source_combobox.configure(values=names)
        self.target_combobox.configure(values=names)

    def _copy_result(self) -> None:
        result = self.result_text_widget.get("1.0", "end-1c")
        if not result.strip():
            messagebox.showwarning(APP_TITLE, "目前沒有可複製的結果。", parent=self)
            return
        self.clipboard_clear()
        self.clipboard_append(result)
        self.update_idletasks()
        self.app.status_var.set("校正轉移結果已複製到剪貼簿。")

    def _compute(self) -> None:
        try:
            source_text = self.source_editor.text.get("1.0", "end-1c")
            target_text = self.target_editor.text.get("1.0", "end-1c")
            result_text, summary = transfer_calibration(source_text, target_text)
        except CteToolError as exc:
            self.app.status_var.set(f"輸入錯誤：{exc}")
            messagebox.showerror(APP_TITLE, str(exc), parent=self)
            return
        self.result_text_widget.delete("1.0", tk.END)
        self.result_text_widget.insert("1.0", result_text)
        self.result_editor.redraw_line_numbers()
        self.app.status_var.set(summary)


def main() -> None:
    root = tk.Tk()
    try:
        ttk.Style(root).theme_use("vista")
    except tk.TclError:
        pass
    app = CtePointApp(root)
    if len(sys.argv) > 1:
        app.load_path(Path(sys.argv[1]))
    root.mainloop()


if __name__ == "__main__":
    main()
