
(defun c:SRTEST (/ e d o c ce ex)
  (vl-load-com)
  (setq e (handent "3740")
        d (entget e))
  (princ (strcat "\n類型: " (cdr (assoc 0 d))
                 "  圖層: " (cdr (assoc 8 d))))
  (if (assoc 70 d)
    (princ (strcat "  旗標70: " (itoa (cdr (assoc 70 d))))))
  (if (assoc 90 d)
    (princ (strcat "  頂點數: " (itoa (cdr (assoc 90 d))))))
  (princ "\n[1] 轉 VLA 物件...")
  (setq o (vlax-ename->vla-object e))
  (princ "OK\n[2] vla-Copy...")
  (setq c (vla-Copy o))
  (princ "OK\n[3] 改圖層...")
  (vla-put-Layer c "0")
  (princ "OK\n[4] 取 ename...")
  (setq ce (vlax-vla-object->ename c))
  (princ "OK\n[5] 寫 XData...")
  (regapp "SRTEST")
  (entmod (append (entget ce)
                  (list (list -3 (list "SRTEST" (cons 1000 "x"))))))
  (princ "OK\n[6] Explode...")
  (if (wcmatch (cdr (assoc 0 (entget ce))) "*POLYLINE")
    (progn
      (setq ex (vlax-invoke c 'Explode))
      (princ (strcat "OK，炸成 " (itoa (length ex)) " 個"))
      (foreach x ex (vla-Delete x)))
    (princ "略過（不是多段線）"))
  (vla-Delete c)
  (princ "\n全部通過")
  (princ))
