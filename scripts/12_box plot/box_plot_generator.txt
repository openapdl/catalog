股價圖 hiLowLines + upDownBars 畫箱體/鬚線，另疊加 XY 散點圖把每個溫度點的
原始樣本與平均值畫上去。這是跟 ``box_plot_generator.pyw``（誤差線版）不同
的另一套技巧，兩者不相容、不要混用。

最前面：改 ``TEMPERATURES``／``SAMPLE_COUNT`` 即可擴增，樣本資料

補資料；其餘列號／輔助資料／圖表都會自動跟著調整。主工作表的樣本格寫的
是 Excel ``RAND()`` 公式（不是寫死的數字），開啟活頁簿後按 F9 可以重新
抽樣。

    python tsm_classic_box_plot_generator.pyw --generate output.xlsx
"""

def _load_xlsxwriter():
    try:
        import xlsxwriter
        from xlsxwriter.utility import xl_col_to_name, xl_range_abs, xl_rowcol_to_cell
    except ImportError as exc:
        raise RuntimeError(
            "找不到 XlsxWriter。請確認 PYTHONW_EXE 指定的 Python "
            "已安裝 xlsxwriter。"
        ) from exc

    package_path = Path(xlsxwriter.__file__).resolve()
    allowed_directory = PYTHONW_EXE.parent.resolve()
    try:
        package_path.relative_to(allowed_directory)
    except ValueError:
        raise RuntimeError(
            "XlsxWriter 必須位於指定的可攜式 Python 資料夾內：{}；"
            "目前載入位置：{}".format(allowed_directory, package_path)
        )

    return xlsxwriter, xl_col_to_name, xl_range_abs, xl_rowcol_to_cell


def _read_font_name(script_dir):
    config_path = script_dir / "fontname.txt"
    if not config_path.exists():
        return DEFAULT_FONT

    try:
        value = config_path.read_text(encoding="utf-8").strip()
    except (OSError, UnicodeError):
        return DEFAULT_FONT

    value = value.strip("\"'").strip()
    return value or DEFAULT_FONT


def _quote_sheet_name(name):
    return "'{}'".format(name.replace("'", "''"))


def _sheet_range(sheet_name, first_row, first_col, last_row, last_col):
    _, _, xl_range_abs, _ = _load_xlsxwriter()
    return "={}!{}".format(
        _quote_sheet_name(sheet_name),
        xl_range_abs(first_row - 1, first_col - 1, last_row - 1, last_col - 1),
    )


def _sheet_cell(sheet_name, row, col):
    _, _, _, xl_rowcol_to_cell = _load_xlsxwriter()
    return "{}!{}".format(
        _quote_sheet_name(sheet_name),
        xl_rowcol_to_cell(row - 1, col - 1, True, True),
    )


def _cell(row, col, absolute=True):
    _, _, _, xl_rowcol_to_cell = _load_xlsxwriter()
    return xl_rowcol_to_cell(row - 1, col - 1, row_abs=absolute, col_abs=absolute)


def _column_name(col):
    _, xl_col_to_name, _, _ = _load_xlsxwriter()
    return xl_col_to_name(col - 1)


def _display_width(text):
    """Approximate rendered width: CJK characters count as roughly 2x ASCII."""
    width = 0
    for char in str(text):
        width += 2 if ord(char) > 0x2E80 else 1
    return width


def _autofit_width(texts, padding=2, minimum=6, maximum=40):
    longest = max((_display_width(text) for text in texts), default=minimum)
    return max(minimum, min(maximum, longest + padding))


def _percentile_inc(values, fraction):
    ordered = sorted(values)
    if len(ordered) == 1:
        return ordered[0]

    import math

    position = (len(ordered) - 1) * fraction
    lower_index = int(math.floor(position))
    upper_index = int(math.ceil(position))
    if lower_index == upper_index:
        return ordered[lower_index]

    weight = position - lower_index
    return ordered[lower_index] + (ordered[upper_index] - ordered[lower_index]) * weight


def _calculate_statistics(values):
    q1 = _percentile_inc(values, 0.25)
    median = _percentile_inc(values, 0.50)
    q3 = _percentile_inc(values, 0.75)
    iqr = q3 - q1
    lower_fence = q1 - 1.5 * iqr
    upper_fence = q3 + 1.5 * iqr
    non_outliers = [value for value in values if lower_fence <= value <= upper_fence]
    stddev = (
        sum((value - sum(values) / len(values)) ** 2 for value in values)
        / (len(values) - 1)
    ) ** 0.5

    return {
        "stddev": stddev,
        "max": max(values),
        "min": min(values),
        "average": sum(values) / float(len(values)),
        "median": median,
        "q3": q3,
        "q1": q1,
        "iqr": iqr,
        "lower_fence": lower_fence,
        "upper_fence": upper_fence,
        "lower_whisker": min(non_outliers),
        "upper_whisker": max(non_outliers),
        "count": len(values),
        "outlier_count": sum(
            1 for value in values if value < lower_fence or value > upper_fence
        ),
    }


def _create_formats(workbook, font_name):
    centered = {"align": "center", "valign": "vcenter"}
    border = {"border": 1}

    def fmt(extra):
        merged = dict({"font_name": font_name}, **extra)
        return workbook.add_format(merged)

    return {
        "title_header": fmt(
            dict(
                {"bold": True, "font_color": "#FFFFFF", "bg_color": NAVY_HEADER_COLOR},
                **centered
            )
        ),
        "index_cell": fmt(dict({"bg_color": PALE_ACCENT_COLOR}, **dict(centered, **border))),
        "sample_label": fmt(dict({"bg_color": PALE_ACCENT_COLOR}, **dict(centered, **border))),
        "data_number": fmt(
            dict({"bg_color": "#FFFFFF", "num_format": "0.00"}, **dict(centered, **border))
        ),
        "stat_label_cn": fmt(dict({"bg_color": PALE_LABEL_COLOR}, **dict(centered, **border))),
        "stat_label_en": fmt(dict({"bg_color": PALE_LABEL_COLOR}, **dict(centered, **border))),
        "stat_number": fmt(
            dict({"bg_color": "#FFFFFF", "num_format": "0.00"}, **dict(centered, **border))
        ),
        "stat_integer": fmt(
            dict({"bg_color": "#FFFFFF", "num_format": "0"}, **dict(centered, **border))
        ),
        "spec_label_cn": fmt(dict({"bg_color": PALE_ACCENT_COLOR}, **dict(centered, **border))),
        "spec_label_en": fmt(
            dict(
                {"bg_color": PALE_LABEL_COLOR, "align": "left", "valign": "vcenter"},
                **border
            )
        ),
        "spec_number": fmt(dict({"bg_color": PALE_ACCENT_COLOR}, **dict(centered, **border))),
        "cal_number": fmt(
            dict({"bg_color": "#FFFFFF", "num_format": "0.00"}, **dict(centered, **border))
        ),
        "leg_blank": fmt(dict({"bg_color": "#FFFFFF"}, **dict(centered, **border))),
        "helper_index": fmt({}),
        "helper_value": fmt({}),
    }


def _write_main_sheet(workbook, worksheet, formats):
    temperature_count = len(TEMPERATURES)
    last_col = 2 + temperature_count  # column B + 每個溫度點各一欄。

    worksheet.merge_range(
        TITLE_ROW - 1, 1, TITLE_ROW - 1, last_col - 1, "TSM Data", formats["title_header"]
    )

    worksheet.write(INDEX_ROW - 1, 1, "數量", formats["sample_label"])
    for index in range(temperature_count):
        worksheet.write_number(INDEX_ROW - 1, 2 + index, index + 1, formats["index_cell"])

    worksheet.write(TEMPERATURE_ROW - 1, 1, "樣本", formats["sample_label"])
    for index, temperature in enumerate(TEMPERATURES):
        worksheet.write_number(
            TEMPERATURE_ROW - 1, 2 + index, float(temperature), formats["sample_label"]
        )

    for sample_index in range(SAMPLE_COUNT):
        row = SAMPLE_FIRST_ROW + sample_index
        worksheet.write(row - 1, 1, "#{}".format(sample_index + 1), formats["sample_label"])
        for temperature_index in range(temperature_count):
            value = SAMPLE_DATA[temperature_index][sample_index]
            span = RANDOM_SAMPLE_MAX - RANDOM_SAMPLE_MIN
            sign = "-" if RANDOM_SAMPLE_MIN < 0 else "+"
            formula = "=RAND()*{}{}{}".format(span, sign, abs(RANDOM_SAMPLE_MIN))
            worksheet.write_formula(
                row - 1, 2 + temperature_index, formula, formats["data_number"], value
            )

    worksheet.write(STATS_HEADER_ROW - 1, 0, "統計指標", formats["title_header"])
    worksheet.merge_range(
        STATS_HEADER_ROW - 1,
        1,
        STATS_HEADER_ROW - 1,
        last_col - 1,
        "統計指標",
        formats["title_header"],
    )

    stat_rows = [
        (STDDEV_ROW, "標準差", "StdDev"),
        (MAX_ROW, "最大值", "Max"),
        (MIN_ROW, "最小值", "Min"),
        (AVERAGE_ROW, "平均值", "Avg"),
        (MEDIAN_ROW, "中位數", "Median"),
        (Q3_ROW, "75%百分位", "Q3"),
        (Q1_ROW, "25%百分位", "Q1"),
        (IQR_ROW, "四分位距 (IQR)", "IQR"),
        (LOWER_FENCE_ROW, "下界 (Q1 - 1.5 × IQR)", "Lower Fence"),
        (UPPER_FENCE_ROW, "上界 (Q3 + 1.5 × IQR)", "Upper Fence"),
        (LOWER_WHISKER_ROW, "下鬚（最小非離群值）", "Lower Whisker"),
        (UPPER_WHISKER_ROW, "上鬚（最大非離群值）", "Upper Whisker"),
        (COUNT_ROW, "有效數值筆數", "Count"),
        (OUTLIER_ROW, "離群值數量", "Outliers"),
    ]
    for row, label_cn, label_en in stat_rows:
        worksheet.write(row - 1, 0, label_cn, formats["stat_label_cn"])
        worksheet.write(row - 1, 1, label_en, formats["stat_label_en"])

    statistics = [_calculate_statistics(column) for column in SAMPLE_DATA]

    for temperature_index, statistic in enumerate(statistics):
        col = 3 + temperature_index  # 1-based: C, D, E, ...
        col_name = _column_name(col)
        data_range = "{0}{1}:{0}{2}".format(col_name, SAMPLE_FIRST_ROW, SAMPLE_LAST_ROW)
        q1_cell = _cell(Q1_ROW, col)
        q3_cell = _cell(Q3_ROW, col)
        iqr_cell = _cell(IQR_ROW, col)
        lower_fence_cell = _cell(LOWER_FENCE_ROW, col)
        upper_fence_cell = _cell(UPPER_FENCE_ROW, col)
        average_cell = _cell(AVERAGE_ROW, col)
        upper_whisker_cell = _cell(UPPER_WHISKER_ROW, col)
        lower_whisker_cell = _cell(LOWER_WHISKER_ROW, col)

        formulas_and_values = [
            (STDDEV_ROW, "=STDEV({})".format(data_range), statistic["stddev"], "stat_number"),
            (MAX_ROW, "=MAX({})".format(data_range), statistic["max"], "stat_number"),
            (MIN_ROW, "=MIN({})".format(data_range), statistic["min"], "stat_number"),
            (
                AVERAGE_ROW,
                "=AVERAGE({})".format(data_range),
                statistic["average"],
                "stat_number",
            ),
            (
                MEDIAN_ROW,
                "=MEDIAN({})".format(data_range),
                statistic["median"],
                "stat_number",
            ),
            (
                Q3_ROW,
                "=QUARTILE({},3)".format(data_range),
                statistic["q3"],
                "stat_number",
            ),
            (
                Q1_ROW,
                "=QUARTILE({},1)".format(data_range),
                statistic["q1"],
                "stat_number",
            ),
            (
                IQR_ROW,
                "={}-{}".format(q3_cell, q1_cell),
                statistic["iqr"],
                "stat_number",
            ),
            (
                LOWER_FENCE_ROW,
                "={}-1.5*{}".format(q1_cell, iqr_cell),
                statistic["lower_fence"],
                "stat_number",
            ),
            (
                UPPER_FENCE_ROW,
                "={}+1.5*{}".format(q3_cell, iqr_cell),
                statistic["upper_fence"],
                "stat_number",
            ),
            (
                LOWER_WHISKER_ROW,
                '=IFERROR(_xlfn.AGGREGATE(15,6,{0}/(({0}>={1})*ISNUMBER({0})),1),"")'.format(
                    data_range, lower_fence_cell
                ),
                statistic["lower_whisker"],
                "stat_number",
            ),
            (
                UPPER_WHISKER_ROW,
                '=IFERROR(_xlfn.AGGREGATE(14,6,{0}/(({0}<={1})*ISNUMBER({0})),1),"")'.format(
                    data_range, upper_fence_cell
                ),
                statistic["upper_whisker"],
                "stat_number",
            ),
            (
                COUNT_ROW,
                "=COUNT({})".format(data_range),
                statistic["count"],
                "stat_integer",
            ),
            (
                OUTLIER_ROW,
                '=COUNTIF({0},"<"&{1})+COUNTIF({0},">"&{2})'.format(
                    data_range, lower_fence_cell, upper_fence_cell
                ),
                statistic["outlier_count"],
                "stat_integer",
            ),
        ]
        for row, formula, cached_value, format_key in formulas_and_values:
            worksheet.write_formula(
                row - 1, col - 1, formula, formats[format_key], cached_value
            )

        # IF(avg>0,ABS(avg),IF(avg<0,-ABS(avg),0)) 化簡後恆等於 avg 本身
        # （avg>0 時 ABS(avg)=avg；avg<0 時 -ABS(avg)=avg），只有 avg=0 時為 0。
        cal_cons_value = statistic["average"]
        worksheet.write_formula(
            CAL_CONS_ROW - 1,
            col - 1,
            "=IF({0}>0,ABS({0}),IF({0}<0,-ABS({0}),0))".format(average_cell),
            formats["cal_number"],
            cal_cons_value,
        )
        for step in range(1, STRICT_TARGET_STEPS + 1):
            row = CAL_STR_FIRST_ROW + step - 1
            if statistic["average"] > 0:
                cached_value = statistic["average"] + (
                    statistic["upper_whisker"] - statistic["average"]
                ) * step / float(STRICT_TARGET_STEPS)
            elif statistic["average"] < 0:
                cached_value = statistic["average"] + (
                    statistic["lower_whisker"] - statistic["average"]
                ) * step / float(STRICT_TARGET_STEPS)
            else:
                cached_value = 0.0
            formula = (
                "=IF({avg}>0,{avg}+({upper}-{avg})*{step}/{steps},"
                "IF({avg}<0,{avg}+({lower}-{avg})*{step}/{steps},0))"
            ).format(
                avg=average_cell,
                upper=upper_whisker_cell,
                lower=lower_whisker_cell,
                step=step,
                steps=STRICT_TARGET_STEPS,
            )
            worksheet.write_formula(
                row - 1, col - 1, formula, formats["cal_number"], cached_value
            )

    worksheet.write(CAL_HEADER_ROW - 1, 0, "校正目標", formats["title_header"])
    worksheet.merge_range(
        CAL_HEADER_ROW - 1,
        1,
        CAL_HEADER_ROW - 1,
        last_col - 1,
        "Cal Target",
        formats["title_header"],
    )

    worksheet.write(SPEC_UPPER_ROW - 1, 0, "規格上限", formats["spec_label_cn"])
    worksheet.write(SPEC_UPPER_ROW - 1, 1, "upper spec", formats["spec_label_en"])
    worksheet.write(SPEC_LOWER_ROW - 1, 0, "規格下限", formats["spec_label_cn"])
    worksheet.write(SPEC_LOWER_ROW - 1, 1, "lower spec", formats["spec_label_en"])
    for temperature_index in range(temperature_count):
        col = 2 + temperature_index
        worksheet.write_number(
            SPEC_UPPER_ROW - 1, col, SPEC_UPPER_VALUE, formats["spec_number"]
        )
        worksheet.write_number(
            SPEC_LOWER_ROW - 1, col, SPEC_LOWER_VALUE, formats["spec_number"]
        )

    worksheet.write(CAL_CONS_ROW - 1, 0, "校正目標（保守）", formats["spec_label_cn"])
    worksheet.write(CAL_CONS_ROW - 1, 1, "Cal Target (Cons.)", formats["spec_label_en"])
    for step in range(1, STRICT_TARGET_STEPS + 1):
        row = CAL_STR_FIRST_ROW + step - 1
        worksheet.write(
            row - 1, 0, "校正目標（嚴謹{}）".format(step), formats["spec_label_cn"]
        )
        worksheet.write(
            row - 1, 1, "Cal Target (Str.{})".format(step), formats["spec_label_en"]
        )

    for leg_row_offset, leg_name in enumerate(LEG_SERIES_NAMES):
        row = LEG_FIRST_ROW + leg_row_offset
        worksheet.write(row - 1, 1, leg_name, formats["spec_label_en"])
        for temperature_index in range(temperature_count):
            col = 2 + temperature_index
            worksheet.write_blank(row - 1, col, None, formats["leg_blank"])

    column_a_texts = (
        ["統計指標", "校正目標"]
        + [label for _, label, _ in stat_rows]
        + ["規格上限", "規格下限", "校正目標（保守）"]
        + ["校正目標（嚴謹{}）".format(step) for step in range(1, STRICT_TARGET_STEPS + 1)]
    )
    column_b_texts = (
        ["數量", "樣本"]
        + ["#{}".format(index + 1) for index in range(SAMPLE_COUNT)]
        + [label for _, _, label in stat_rows]
        + ["upper spec", "lower spec", "Cal Target (Cons.)"]
        + ["Cal Target (Str.{})".format(step) for step in range(1, STRICT_TARGET_STEPS + 1)]
        + LEG_SERIES_NAMES
    )
    worksheet.set_column(0, 0, _autofit_width(column_a_texts))
    worksheet.set_column(1, 1, _autofit_width(column_b_texts))

    flat_values = [value for column in SAMPLE_DATA for value in column]
    has_negative = any(value < 0 for value in flat_values) or SPEC_LOWER_VALUE < 0
    max_abs_value = max(
        [abs(value) for value in flat_values] + [SPEC_UPPER_VALUE, abs(SPEC_LOWER_VALUE)]
    )
    integer_digits = max(1, len("{:.0f}".format(max_abs_value)))
    sample_number_text = (
        ("-" if has_negative else "") + ("9" * integer_digits) + "." + ("9" * 2)
    )
    data_column_width = _autofit_width(
        [str(temperature) for temperature in TEMPERATURES] + [sample_number_text]
    )
    worksheet.set_column(2, last_col - 1, data_column_width)

    return statistics, last_col


def _write_helper_sheet(workbook, worksheet, formats):
    for temperature_index in range(len(TEMPERATURES)):
        x_col = 2 + 2 * temperature_index  # 1-based: B, D, F, ...
        y_col = x_col + 1
        main_col = 3 + temperature_index  # 工作表1 的溫度欄：C, D, E, ...
        main_col_name = _column_name(main_col)
        for sample_index in range(SAMPLE_COUNT):
            row = sample_index + 1
            worksheet.write_number(
                row - 1, x_col - 1, temperature_index + 1, formats["helper_index"]
            )
            main_row = SAMPLE_FIRST_ROW + sample_index
            formula = "={}".format(
                _sheet_cell(MAIN_SHEET, main_row, main_col)
            )
            worksheet.write_formula(
                row - 1,
                y_col - 1,
                formula,
                formats["helper_value"],
                SAMPLE_DATA[temperature_index][sample_index],
            )

            "values": box_row_range(MEDIAN_ROW),

def _write_workbook(path, script_dir):
    xlsxwriter, _, _, _ = _load_xlsxwriter()

    workbook = xlsxwriter.Workbook(str(path))
    try:
        workbook.set_properties(
            {
                "title": "TSM Box Plot (classic plot)",
                "subject": "Excel 2010 compatible box plot",
                "comments": (
                    "Generated by tsm_classic_box_plot_generator.pyw"
                    "（hiLowLines + upDownBars + XY scatter 技巧）。"
                ),
            }
        )
        workbook.set_calc_mode("auto")
        formats = _create_formats(workbook, _read_font_name(script_dir))

        main_sheet = workbook.add_worksheet(MAIN_SHEET)
        helper_sheet = workbook.add_worksheet(HELPER_SHEET)

        _statistics, last_col = _write_main_sheet(workbook, main_sheet, formats)
        _write_helper_sheet(workbook, helper_sheet, formats)
        _add_chart(workbook, main_sheet, last_col)

        main_sheet.activate()
    finally:
        workbook.close()


def generate_workbook(output_path, overwrite=False):
    output_path = Path(output_path).expanduser()
    if output_path.suffix.lower() != ".xlsx":
        output_path = output_path.with_suffix(".xlsx")
    output_path = output_path.resolve()

    if not output_path.parent.exists():
        raise FileNotFoundError("輸出資料夾不存在：{}".format(output_path.parent))
    if output_path.exists() and not overwrite:
        raise FileExistsError("檔案已存在：{}".format(output_path))

    file_handle, temporary_name = tempfile.mkstemp(
        prefix=".tsm_classic_box_plot_build_",
        suffix=".xlsx",
        dir=str(output_path.parent),
    )
    os.close(file_handle)
    temporary_path = Path(temporary_name)

    try:
        _write_workbook(temporary_path, Path(__file__).resolve().parent)
        os.replace(str(temporary_path), str(output_path))
    finally:
        if temporary_path.exists():
            temporary_path.unlink()

    return output_path


def _configured_pythonw():
    return PYTHONW_EXE if PYTHONW_EXE.is_file() else None


def _ensure_allowed_runtime():
    current_directory = Path(sys.executable).resolve().parent
    allowed_directory = PYTHONW_EXE.resolve().parent
    if current_directory != allowed_directory:
        raise RuntimeError(
            "本程式只能使用指定的可攜式 Python：{}；"
            "目前執行環境：{}".format(PYTHONW_EXE, sys.executable)
        )


def _relaunch_for_double_click():
    """Use the configured pythonw.exe for a no-console double-click launch."""
    if len(sys.argv) > 1:
        return False

    script_path = Path(__file__).resolve()
    configured = _configured_pythonw()
    if configured is None:
        raise FileNotFoundError(
            "找不到指定的 pythonw.exe，請修改 PYTHONW_EXE：{}".format(PYTHONW_EXE)
        )

    subprocess.Popen(
        [str(configured), "-I", str(script_path), "--direct-run"],
        cwd=str(script_path.parent),
    )
    return True


def _clear_error_log(script_dir):
    error_log = script_dir / ERROR_LOG_FILENAME
    if error_log.exists():
        try:
            error_log.unlink()
        except OSError:
            pass


def _write_error_log(error):
    script_dir = Path(__file__).resolve().parent
    error_log = script_dir / ERROR_LOG_FILENAME
    details = "{}\n\n{}".format(error, traceback.format_exc())
    try:
        error_log.write_text(details, encoding="utf-8")
    except OSError:
        pass


def main(argv=None):
    _ensure_allowed_runtime()
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument(
        "--generate", metavar="OUTPUT.xlsx", help="直接產生指定的 Excel 檔案"
    )
    parser.add_argument(
        "--overwrite", action="store_true", help="允許覆寫既有輸出檔"
    )
    parser.add_argument("--direct-run", action="store_true", help=argparse.SUPPRESS)
    args = parser.parse_args(argv)

    if args.generate:
        result = generate_workbook(args.generate, overwrite=args.overwrite)
        print(result)
        return 0

    script_dir = Path(__file__).resolve().parent
    output_path = script_dir / DEFAULT_OUTPUT_FILENAME
    generate_workbook(output_path, overwrite=True)
    _clear_error_log(script_dir)
    return 0


if __name__ == "__main__":
    try:
        if _relaunch_for_double_click():
            sys.exit(0)
        sys.exit(main())
    except Exception as error:
        if "--generate" in sys.argv:
            print("錯誤：{}".format(error), file=sys.stderr)
        else:
            _write_error_log(error)
        sys.exit(1)
