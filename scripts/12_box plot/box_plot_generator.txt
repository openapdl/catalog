def _load_xlsxwriter():
    try:
        import xlsxwriter
        from xlsxwriter.utility import xl_col_to_name, xl_range_abs, xl_rowcol_to_cell
    except ImportError as exc:
        raise RuntimeError(
            "找不到 XlsxWriter。請確認 PYTHONW_EXE 指定的 Python "
            "已安裝 xlsxwriter。"
        ) from exc

    package_path = Path(xlsxwriter.__file__).resolve()
    allowed_directory = PYTHONW_EXE.parent.resolve()
    try:
        package_path.relative_to(allowed_directory)
    except ValueError:
        raise RuntimeError(
            "XlsxWriter 必須位於指定的可攜式 Python 資料夾內：{}；"
            "目前載入位置：{}".format(allowed_directory, package_path)
        )

    return xlsxwriter, xl_col_to_name, xl_range_abs, xl_rowcol_to_cell


def _read_font_name(script_dir):
    config_path = script_dir / "fontname.txt"
    if not config_path.exists():
        return DEFAULT_FONT

    try:
        value = config_path.read_text(encoding="utf-8").strip()
    except (OSError, UnicodeError):
        return DEFAULT_FONT

    value = value.strip("\"'").strip()
    return value or DEFAULT_FONT


def _quote_sheet_name(name):
    return "'{}'".format(name.replace("'", "''"))


def _sheet_range(sheet_name, first_row, first_col, last_row, last_col):
    _, _, xl_range_abs, _ = _load_xlsxwriter()
    return "={}!{}".format(
        _quote_sheet_name(sheet_name),
        xl_range_abs(first_row - 1, first_col - 1, last_row - 1, last_col - 1),
    )


def _sheet_cell(sheet_name, row, col):
    _, _, _, xl_rowcol_to_cell = _load_xlsxwriter()
    return "{}!{}".format(
        _quote_sheet_name(sheet_name),
        xl_rowcol_to_cell(row - 1, col - 1, True, True),
    )


def _cell(row, col, absolute=False):
    _, _, _, xl_rowcol_to_cell = _load_xlsxwriter()
    return xl_rowcol_to_cell(
        row - 1,
        col - 1,
        row_abs=absolute,
        col_abs=absolute,
    )


def _column_name(col):
    _, xl_col_to_name, _, _ = _load_xlsxwriter()
    return xl_col_to_name(col - 1)


def _display_width(text):
    """Approximate rendered width: CJK characters count as roughly 2x ASCII."""
    width = 0
    for char in str(text):
        width += 2 if ord(char) > 0x2E80 else 1
    return width


def _autofit_width(texts, padding=2, minimum=6, maximum=40):
    longest = max((_display_width(text) for text in texts), default=minimum)
    return max(minimum, min(maximum, longest + padding))


def normalize_temperatures(temperatures):
    cleaned = [str(value).strip() for value in temperatures if str(value).strip()]
    if len(cleaned) < 2:
        raise ValueError("至少需要 2 個溫度點。")
    if len(cleaned) > MAX_TEMPERATURES:
        raise ValueError("溫度點不可超過 {} 個。".format(MAX_TEMPERATURES))

    normalized_names = [value.casefold() for value in cleaned]
    duplicates = sorted(
        {
            value
            for value, normalized in zip(cleaned, normalized_names)
            if normalized_names.count(normalized) > 1
        }
    )
    if duplicates:
        raise ValueError("溫度點名稱不可重複：{}".format(", ".join(duplicates)))

    for value in cleaned:
        if len(value) > 255:
            raise ValueError("溫度點名稱過長：{}".format(value[:30]))

    return cleaned


def _percentile_inc(values, fraction):
    ordered = sorted(values)
    if len(ordered) == 1:
        return ordered[0]

    position = (len(ordered) - 1) * fraction
    lower_index = int(math.floor(position))
    upper_index = int(math.ceil(position))
    if lower_index == upper_index:
        return ordered[lower_index]

    weight = position - lower_index
    return ordered[lower_index] + (ordered[upper_index] - ordered[lower_index]) * weight


def _calculate_statistics(values):
    q1 = _percentile_inc(values, 0.25)
    median = _percentile_inc(values, 0.50)
    q3 = _percentile_inc(values, 0.75)
    iqr = q3 - q1
    lower_bound = q1 - 1.5 * iqr
    upper_bound = q3 + 1.5 * iqr
    non_outliers = [value for value in values if lower_bound <= value <= upper_bound]
    lower_whisker = min(non_outliers)
    upper_whisker = max(non_outliers)
    average = sum(values) / float(len(values))
    outliers = [value < lower_bound or value > upper_bound for value in values]

    return {
        "count": len(values),
        "min": min(values),
        "q1": q1,
        "median": median,
        "q3": q3,
        "max": max(values),
        "average": average,
        "iqr": iqr,
        "lower_bound": lower_bound,
        "upper_bound": upper_bound,
        "lower_whisker": lower_whisker,
        "upper_whisker": upper_whisker,
        "outlier_count": sum(outliers),
        "outliers": outliers,
    }


def _cached_target(stats, step=None):
    average = stats["average"]
    if step is None:
        return average
    if average > 0:
        end_value = stats["upper_whisker"]
    elif average < 0:
        end_value = stats["lower_whisker"]
    else:
        return 0.0
    return average + (end_value - average) * step / float(STRICT_TARGET_STEPS)


def _nice_axis_major_unit(axis_range, target_tick_count=8):
    """Round the Y 軸刻度間距到一個「漂亮」數字（1/2/5 × 10 的次方）。

    Excel 的格線是以 0 為基準、按 major unit 的整數倍畫出來的，不是從軸的
    min 開始算；只要 major unit 是這種漂亮數字，0（只要落在軸範圍內）就
    一定會剛好對齊到一條格線，Y=0 的刻度標籤才會確實被畫出來，而不是只是
    「範圍涵蓋 0」卻剛好沒有刻度線經過它。
    """
    if axis_range <= 0:
        return 1.0

    raw_step = axis_range / float(target_tick_count)
    magnitude = 10.0 ** math.floor(math.log10(raw_step))
    normalized = raw_step / magnitude
    if normalized <= 1:
        nice = 1
    elif normalized <= 2:
        nice = 2
    elif normalized <= 5:
        nice = 5
    else:
        nice = 10
    return nice * magnitude

        col = temperature_index + 2
        worksheet.write(
            INDEX_ROW - 1,
            col - 1,
            temperature_index + 1,
            formats["plain_center"],
        )

    table_columns = [
        {"header": "樣本", "header_format": formats["plain_center"]}
    ] + [
        {"header": temperature, "header_format": formats["plain_center"]}
        for temperature in temperatures
    ]
    worksheet.add_table(
        HEADER_ROW - 1,
        0,
        LEG_LAST_ROW - 1,
        last_data_col - 1,
        {
            "name": "tblBoxData2010",
            "style": "Table Style Medium 2",
            "columns": table_columns,
            "autofilter": False,
        },
    )

    for sample_index in range(SAMPLE_COUNT):
        excel_row = DATA_FIRST_ROW + sample_index
        worksheet.write(excel_row - 1, 0, "#{}".format(sample_index + 1), formats["plain_center"])
        for temperature_index in range(len(temperatures)):
            value = data[temperature_index][sample_index]
            worksheet.write_formula(
                excel_row - 1,
                temperature_index + 1,
                "=RAND()*500-250",
                formats["number"],
                value,
            )

    summary_rows = [
        (SUMMARY_MIN_ROW, "Min", MIN_ROW, "min"),
        (SUMMARY_MAX_ROW, "Max", MAX_ROW, "max"),
        (SUMMARY_AVERAGE_ROW, "Average", AVERAGE_ROW, "average"),
    ]
    for row, label, source_row, key in summary_rows:
        worksheet.write(row - 1, 0, label, formats["summary_label"])
        for temperature_index, statistic in enumerate(stats):
            col = temperature_index + 2
            worksheet.write_formula(
                row - 1,
                col - 1,
                "={}".format(_cell(source_row, col)),
                formats["summary_number"],
                statistic[key],
            )

    worksheet.write(
        TARGET_BASE_ROW - 1,
        0,
        "Cal Target (Cons.)",
        formats["target_label"],
    )
    for temperature_index, statistic in enumerate(stats):
        col = temperature_index + 2
        average_cell = _cell(AVERAGE_ROW, col)
        formula = "=IF({0}>0,ABS({0}),IF({0}<0,-ABS({0}),0))".format(average_cell)
        worksheet.write_formula(
            TARGET_BASE_ROW - 1,
            col - 1,
            formula,
            formats["target_number"],
            _cached_target(statistic),
        )

    for step in range(1, STRICT_TARGET_STEPS + 1):
        row = TARGET_FIRST_STRICT_ROW + step - 1
        worksheet.write(
            row - 1,
            0,
            "Cal Target (Str.{})".format(step),
            formats["target_label"],
        )
        for temperature_index, statistic in enumerate(stats):
            col = temperature_index + 2
            average_cell = _cell(AVERAGE_ROW, col)
            upper_cell = _cell(UPPER_WHISKER_ROW, col)
            lower_cell = _cell(LOWER_WHISKER_ROW, col)
            formula = (
                "=IF({avg}>0,{avg}+({upper}-{avg})*{step}/{steps},"
                "IF({avg}<0,{avg}+({lower}-{avg})*{step}/{steps},0))"
            ).format(
                avg=average_cell,
                upper=upper_cell,
                lower=lower_cell,
                step=step,
                steps=STRICT_TARGET_STEPS,
            )
            worksheet.write_formula(
                row - 1,
                col - 1,
                formula,
                formats["target_number"],
                _cached_target(statistic, step),
            )

    worksheet.write(SPEC_UPPER_ROW - 1, 0, "Upper Spec", formats["spec_label"])
    worksheet.write(SPEC_LOWER_ROW - 1, 0, "Lower Spec", formats["spec_label"])
    for temperature_index in range(len(temperatures)):
        col = temperature_index + 2
        worksheet.write_number(
            SPEC_UPPER_ROW - 1, col - 1, SPEC_UPPER_VALUE, formats["spec_number"]
        )
        worksheet.write_number(
            SPEC_LOWER_ROW - 1, col - 1, SPEC_LOWER_VALUE, formats["spec_number"]
        )

    for leg_row_offset, leg_name in enumerate(LEG_SERIES_NAMES):
        row = LEG_FIRST_ROW + leg_row_offset
        worksheet.write(row - 1, 0, leg_name, formats["leg_label"])
        for temperature_index in range(len(temperatures)):
            col = temperature_index + 2
            worksheet.write_formula(
                row - 1,
                col - 1,
                "=RAND()*500-250",
                formats["leg_number"],
                leg_data[leg_row_offset][temperature_index],
            )

    worksheet.write(STATS_HEADER_ROW - 1, 0, "統計指標", formats["stats_header"])
    for temperature_index, temperature in enumerate(temperatures):
        worksheet.write(
            STATS_HEADER_ROW - 1,
            temperature_index + 1,
            temperature,
            formats["stats_header"],
        )

    statistic_rows = [
        (COUNT_ROW, "有效數值筆數", "count"),
        (MIN_ROW, "最小值", "min"),
        (Q1_ROW, "第一四分位數 (Q1)", "q1"),
        (MEDIAN_ROW, "中位數", "median"),
        (Q3_ROW, "第三四分位數 (Q3)", "q3"),
        (MAX_ROW, "最大值", "max"),
        (AVERAGE_ROW, "平均值", "average"),
        (IQR_ROW, "四分位距 (IQR)", "iqr"),
        (LOWER_BOUND_ROW, "下界 (Q1 - 1.5 × IQR)", "lower_bound"),
        (UPPER_BOUND_ROW, "上界 (Q3 + 1.5 × IQR)", "upper_bound"),
        (LOWER_WHISKER_ROW, "下鬚（最小非離群值）", "lower_whisker"),
        (UPPER_WHISKER_ROW, "上鬚（最大非離群值）", "upper_whisker"),
        (OUTLIER_COUNT_ROW, "離群值數量", "outlier_count"),
    ]
    for row, label, _ in statistic_rows:
        worksheet.write(row - 1, 0, label, formats["stats_label"])

    column_a_texts = (
        ["樣本"]
        + ["#{}".format(sample_index + 1) for sample_index in range(SAMPLE_COUNT)]
        + [label for _, label, _, _ in summary_rows]
        + ["Cal Target (Cons.)"]
        + [
            "Cal Target (Str.{})".format(step)
            for step in range(1, STRICT_TARGET_STEPS + 1)

        + ["Upper Spec", "Lower Spec"]
        + LEG_SERIES_NAMES
        + ["統計指標"]
        + [label for _, label, _ in statistic_rows]

    worksheet.set_column(0, 0, _autofit_width(column_a_texts))

    flat_values = [value for column_values in data for value in column_values]
    has_negative = any(value < 0 for value in flat_values)
    max_abs_value = max((abs(value) for value in flat_values), default=0)
    integer_digits = max(1, len("{:.0f}".format(max_abs_value)))
    sample_number_text = (
        ("-" if has_negative else "") + ("9" * integer_digits) + "." + ("9" * 4)
    )
    data_column_width = _autofit_width(list(temperatures) + [sample_number_text])
    worksheet.set_column(1, last_data_col - 1, data_column_width)

    for temperature_index, statistic in enumerate(stats):
        col = temperature_index + 2
        col_name = _column_name(col)
        data_range = "${0}${1}:${0}${2}".format(
            col_name, DATA_FIRST_ROW, DATA_LAST_ROW
        )
        main_data_range = "{}!{}".format(_quote_sheet_name(MAIN_SHEET), data_range)
        count_cell = _cell(COUNT_ROW, col)
        q1_cell = _cell(Q1_ROW, col)
        q3_cell = _cell(Q3_ROW, col)
        iqr_cell = _cell(IQR_ROW, col)
        lower_bound_cell = _cell(LOWER_BOUND_ROW, col)
        upper_bound_cell = _cell(UPPER_BOUND_ROW, col)

        formulas = {
            COUNT_ROW: "=COUNT({})".format(main_data_range),
            MIN_ROW: '=IF({0}=0,"",MIN({1}))'.format(count_cell, main_data_range),
            Q1_ROW: '=IF({0}=0,"",QUARTILE({1},1))'.format(count_cell, main_data_range),
            MEDIAN_ROW: '=IF({0}=0,"",MEDIAN({1}))'.format(count_cell, main_data_range),
            Q3_ROW: '=IF({0}=0,"",QUARTILE({1},3))'.format(count_cell, main_data_range),
            MAX_ROW: '=IF({0}=0,"",MAX({1}))'.format(count_cell, main_data_range),
            AVERAGE_ROW: '=IF({0}=0,"",AVERAGE({1}))'.format(count_cell, main_data_range),
            IQR_ROW: '=IF({0}=0,"",{1}-{2})'.format(count_cell, q3_cell, q1_cell),
            LOWER_BOUND_ROW: '=IF({0}=0,"",{1}-1.5*{2})'.format(
                count_cell, q1_cell, iqr_cell
            ),
            UPPER_BOUND_ROW: '=IF({0}=0,"",{1}+1.5*{2})'.format(
                count_cell, q3_cell, iqr_cell
            ),
            LOWER_WHISKER_ROW: (
                '=IF({count}=0,"",IFERROR(_xlfn.AGGREGATE(15,6,'
                '{data}/(({data}>={bound})*ISNUMBER({data})),1),""))'
            ).format(count=count_cell, data=main_data_range, bound=lower_bound_cell),
            UPPER_WHISKER_ROW: (
                '=IF({count}=0,"",IFERROR(_xlfn.AGGREGATE(14,6,'
                '{data}/(({data}<={bound})*ISNUMBER({data})),1),""))'
            ).format(count=count_cell, data=main_data_range, bound=upper_bound_cell),
            OUTLIER_COUNT_ROW: (
                '=IF({count}=0,"",COUNTIF({data},"<"&{lower})+'
                'COUNTIF({data},">"&{upper}))'
            ).format(
                count=count_cell,
                data=main_data_range,
                lower=lower_bound_cell,
                upper=upper_bound_cell,
            ),
        }
        cached_values = {
            COUNT_ROW: statistic["count"],
            MIN_ROW: statistic["min"],
            Q1_ROW: statistic["q1"],
            MEDIAN_ROW: statistic["median"],
            Q3_ROW: statistic["q3"],
            MAX_ROW: statistic["max"],
            AVERAGE_ROW: statistic["average"],
            IQR_ROW: statistic["iqr"],
            LOWER_BOUND_ROW: statistic["lower_bound"],
            UPPER_BOUND_ROW: statistic["upper_bound"],
            LOWER_WHISKER_ROW: statistic["lower_whisker"],
            UPPER_WHISKER_ROW: statistic["upper_whisker"],
            OUTLIER_COUNT_ROW: statistic["outlier_count"],
        }

        for row, _, _ in statistic_rows:
            if row == COUNT_ROW:
                cell_format = formats["stats_count"]
            elif row == AVERAGE_ROW:
                cell_format = formats["stats_average"]
            elif row == OUTLIER_COUNT_ROW:
                cell_format = formats["stats_count"]
            else:
                cell_format = formats["stats_number"]
            worksheet.write_formula(
                row - 1,
                col - 1,
                formulas[row],
                cell_format,
                cached_values[row],
            )

    chart_width = max(1000, min(1450, 70 * len(temperatures)))
    chart_height = 620
    rows_per_chart = int(math.ceil(chart_height / 20.0)) + 1
    chart_gap_rows = 2
    # Zero-based row for the first chart, one blank row below the stats block;
    # the remaining charts stack directly below it with a small gap each.
    chart_start_rows = [
        OUTLIER_COUNT_ROW + 1 + chart_index * (rows_per_chart + chart_gap_rows)
        for chart_index in range(len(CHART_VARIANT_TITLES))
    ]
    chart_end_row = chart_start_rows[-1] + rows_per_chart

    worksheet.set_landscape()
    worksheet.set_paper(9)  # A4.
    worksheet.fit_to_pages(1, 2)
    worksheet.center_horizontally()
    worksheet.set_margins(left=0.25, right=0.25, top=0.5, bottom=0.5)
    worksheet.print_area(0, 0, chart_end_row, last_data_col - 1)

    return chart_start_rows, chart_width, chart_height

def _write_helper_sheet(worksheet, temperatures, data, stats, formats):
    headers = [
        "指標",
        "Q1 底座",
        "Q1 至中位數",
        "中位數至 Q3",
        "下鬚誤差",
        "上鬚誤差",
        "平均值",
    ] + ["離群值_{}".format(index) for index in range(1, SAMPLE_COUNT + 1)] + [
        "Q3",
        "BoxLegend",
        "ZeroError",
        "MedianBase",
        "MedianThickness",
    ]
    worksheet.write_row(0, 0, headers, formats["helper"])

    overall_min = min(statistic["min"] for statistic in stats)
    overall_max = max(statistic["max"] for statistic in stats)
    overall_range = overall_max - overall_min
    median_thickness = overall_range * MEDIAN_MARKER_THICKNESS_RATIO
    first_box_col_name = _column_name(BOX_STATS_FIRST_VALUE_COL)
    last_box_col_name = _column_name(BOX_STATS_FIRST_VALUE_COL + len(temperatures) - 1)
    median_thickness_formula = (
        "(MAX({s}!{first}{max_row}:{last}{max_row})-"
        "MIN({s}!{first}{min_row}:{last}{min_row}))*{ratio}"
    ).format(
        s=_quote_sheet_name(STATS_SHEET),
        first=first_box_col_name,
        last=last_box_col_name,
        max_row=BOX_MAX_ROW,
        min_row=BOX_MIN_ROW,
        ratio=MEDIAN_MARKER_THICKNESS_RATIO,
    )

    for temperature_index, (temperature, statistic) in enumerate(zip(temperatures, stats)):
        row = temperature_index + 2
        main_col = temperature_index + 2
        box_col = BOX_STATS_FIRST_VALUE_COL + temperature_index
        box_col_name = _column_name(box_col)

        worksheet.write_formula(
            row - 1,
            HELPER_CATEGORY_COL - 1,
            "={}".format(_sheet_cell(MAIN_SHEET, HEADER_ROW, main_col)),
            formats["helper"],
            temperature,
        )

        formulas_and_values = [
            (
                HELPER_Q1_COL,
                "={}!{}{}".format(_quote_sheet_name(STATS_SHEET), box_col_name, BOX_Q1_ROW),
                statistic["q1"],
            ),
            (
                HELPER_Q1_TO_MEDIAN_COL,
                "={s}!{c}{median}-{s}!{c}{q1}".format(
                    s=_quote_sheet_name(STATS_SHEET),
                    c=box_col_name,
                    median=BOX_MEDIAN_ROW,
                    q1=BOX_Q1_ROW,
                ),
                statistic["median"] - statistic["q1"],
            ),
            (
                HELPER_MEDIAN_TO_Q3_COL,
                "={s}!{c}{q3}-{s}!{c}{median}".format(
                    s=_quote_sheet_name(STATS_SHEET),
                    c=box_col_name,
                    q3=BOX_Q3_ROW,
                    median=BOX_MEDIAN_ROW,
                ),
                statistic["q3"] - statistic["median"],
            ),
            (
                HELPER_LOWER_ERROR_COL,
                "={s}!{c}{q1}-{s}!{c}{whisker}".format(
                    s=_quote_sheet_name(STATS_SHEET),
                    c=box_col_name,
                    q1=BOX_Q1_ROW,
                    whisker=BOX_LOWER_WHISKER_ROW,
                ),
                statistic["q1"] - statistic["lower_whisker"],
            ),
            (
                HELPER_UPPER_ERROR_COL,
                "={s}!{c}{whisker}-{s}!{c}{q3}".format(
                    s=_quote_sheet_name(STATS_SHEET),
                    c=box_col_name,
                    whisker=BOX_UPPER_WHISKER_ROW,
                    q3=BOX_Q3_ROW,
                ),
                statistic["upper_whisker"] - statistic["q3"],
            ),
            (
                HELPER_AVERAGE_COL,
                "={}!{}{}".format(
                    _quote_sheet_name(STATS_SHEET), box_col_name, BOX_AVERAGE_ROW
                ),
                statistic["average"],
            ),
        ]
        for col, formula, cached_value in formulas_and_values:
            worksheet.write_formula(
                row - 1,
                col - 1,
                formula,
                formats["helper"],
                cached_value,
            )

        for sample_index in range(SAMPLE_COUNT):
            col = HELPER_FIRST_OUTLIER_COL + sample_index
            main_row = DATA_FIRST_ROW + sample_index
            main_cell = _sheet_cell(MAIN_SHEET, main_row, main_col)
            lower_cell = "{}!${}${}".format(
                _quote_sheet_name(STATS_SHEET), box_col_name, BOX_LOWER_BOUND_ROW
            )
            upper_cell = "{}!${}${}".format(
                _quote_sheet_name(STATS_SHEET), box_col_name, BOX_UPPER_BOUND_ROW
            )
            formula = "=IF(OR({0}<{1},{0}>{2}),{0},NA())".format(
                main_cell, lower_cell, upper_cell
            )
            cached_value = (
                data[temperature_index][sample_index]
                if statistic["outliers"][sample_index]
                else "#N/A"
            )
            worksheet.write_formula(
                row - 1,
                col - 1,
                formula,
                formats["helper"],
                cached_value,
            )

        worksheet.write_formula(
            row - 1,
            HELPER_Q3_COL - 1,
            "={}!${}${}".format(
                _quote_sheet_name(STATS_SHEET), box_col_name, BOX_Q3_ROW
            ),
            formats["helper"],
            statistic["q3"],
        )
        worksheet.write_formula(
            row - 1,
            HELPER_BOX_LEGEND_COL - 1,
            "=NA()",
            formats["helper"],
            "#N/A",
        )
        worksheet.write_number(
            row - 1,
            HELPER_ZERO_ERROR_COL - 1,
            0,
            formats["helper"],
        )

        worksheet.write_formula(
            row - 1,
            HELPER_MEDIAN_THICKNESS_COL - 1,
            "={}".format(median_thickness_formula),
            formats["helper"],
            median_thickness,
        )
        worksheet.write_formula(
            row - 1,
            HELPER_MEDIAN_BASE_COL - 1,
            "={s}!{c}{median}-(({thickness})/2)".format(
                s=_quote_sheet_name(STATS_SHEET),
                c=box_col_name,
                median=BOX_MEDIAN_ROW,
                thickness=median_thickness_formula,
            ),
            formats["helper"],
            statistic["median"] - median_thickness / 2,
        )

    # XlsxWriter 3.0.2 only exposes normal hiding.  The ZIP post-processing
    # step changes this specific sheet to Excel's veryHidden state.
    worksheet.hide()


def _add_chart(
    workbook,
    worksheet,
    temperatures,
    chart_start_row,
    chart_width,
    chart_height,
    axis_min,
    axis_max,
    axis_major_unit,
    chart_variant="primary",
):
    include_nt_legs = chart_variant in ("all", "uncalibrated")
    include_t_legs = chart_variant in ("all", "calibrated")

    first_helper_row = 2
    last_helper_row = first_helper_row + len(temperatures) - 1
    categories = _sheet_range(
        HELPER_SHEET,
        first_helper_row,
        HELPER_CATEGORY_COL,
        last_helper_row,
        HELPER_CATEGORY_COL,
    )

    def helper_range(col):
        return _sheet_range(
            HELPER_SHEET,
            first_helper_row,
            col,
            last_helper_row,
            col,
        )

    chart = workbook.add_chart({"type": "line"})
    chart.set_style(2)

    series_index = [0]
    hidden_legend_series = []

    def add_series(target_chart, options, show_in_legend):
        target_chart.add_series(options)
        if not show_in_legend:
            hidden_legend_series.append(series_index[0])
        series_index[0] += 1

    # Exact colors resolved from the reference workbook's Office theme.
    whisker_color = "#B4C6E7"  # accent5, lighter 60%

def _encode_legacy_password(password):
    """Legacy Excel password hash (same algorithm XlsxWriter uses for sheet protection)."""
    digits = []
    for shift, char in enumerate(password, start=1):
        value = ord(char) << shift
        low_15 = value & 0x7FFF
        high_15 = (value & (0x7FFF << 15)) >> 15
        digits.append(low_15 | high_15)

    password_hash = 0x0000
    for digit in digits:
        password_hash ^= digit
    password_hash ^= len(password)
    password_hash ^= 0xCE4B

    return "%X" % password_hash


def _patch_generated_xlsx(path):
    """Apply two OOXML details unavailable in the bundled XlsxWriter 3.0.2."""
    handle, temporary_name = tempfile.mkstemp(
        prefix=".box_plot_patch_",
        suffix=".xlsx",
        dir=str(path.parent),
    )
    os.close(handle)
    temporary_path = Path(temporary_name)

    try:
        with zipfile.ZipFile(str(path), "r") as source_zip:
            with zipfile.ZipFile(str(temporary_path), "w") as target_zip:
                for info in source_zip.infolist():
                    content = source_zip.read(info.filename)
                    if info.filename == "xl/workbook.xml":
                        content = re.sub(
                            rb'(<sheet\b[^>]*\bname="_BoxPlot2010"[^>]*\bstate=")hidden(")',
                            rb"\1veryHidden\2",
                            content,
                            count=1,
                        )
                        if WORKBOOK_STRUCTURE_PASSWORD:
                            password_hash = _encode_legacy_password(
                                WORKBOOK_STRUCTURE_PASSWORD
                            )
                            protection_tag = (
                                '<workbookProtection lockStructure="1" '
                                'workbookPassword="{}"/>'
                            ).format(password_hash).encode("ascii")
                            # Schema order requires workbookProtection right after
                            # the self-closing <workbookPr .../> element.
                            content = re.sub(
                                rb"(<workbookPr\b[^>]*/>)",
                                lambda match: match.group(1) + protection_tag,
                                content,
                                count=1,
                            )
                    elif info.filename.startswith("xl/charts/chart") and info.filename.endswith(
                        ".xml"
                    ):
                        content = content.replace(
                            b'<c:gapWidth val="150"/>',
                            '<c:gapWidth val="{}"/>'.format(BOX_GAP_WIDTH).encode(
                                "ascii"
                            ),
                            1,
                        )
                    target_zip.writestr(info, content)
        os.replace(str(temporary_path), str(path))
    finally:
        if temporary_path.exists():
            temporary_path.unlink()


def _write_workbook(path, temperatures, script_dir):
    xlsxwriter, _, _, _ = _load_xlsxwriter()
    rng = random.Random()
    data = [
        [rng.random() * 500.0 - 250.0 for _ in range(SAMPLE_COUNT)]
        for _ in temperatures
    ]
    stats = [_calculate_statistics(values) for values in data]
    leg_data = [
        [rng.random() * 500.0 - 250.0 for _ in temperatures]
        for _ in LEG_SERIES_NAMES
    ]

    all_values = (
        [value for column_values in data for value in column_values]
        + [value for row_values in leg_data for value in row_values]
        + [SPEC_UPPER_VALUE, SPEC_LOWER_VALUE]
    )
    raw_min, raw_max = min(all_values), max(all_values)
    axis_margin = (raw_max - raw_min) * Y_AXIS_AUTO_MARGIN_RATIO
    axis_min = Y_AXIS_MIN if Y_AXIS_MIN is not None else raw_min - axis_margin
    axis_max = Y_AXIS_MAX if Y_AXIS_MAX is not None else raw_max + axis_margin
    # 不論自動或手動設定，Y=0 一律要落在範圍內，才能保證基準線一定畫得出來。
    axis_min = min(axis_min, 0.0)
    axis_max = max(axis_max, 0.0)
    # 格線間距用「漂亮數字」而非 Excel 自動選值，確保 0 一定剛好落在某條
    # 格線上、刻度標籤「0」才會真的被畫出來，不是只有範圍涵蓋到而已。
    axis_major_unit = _nice_axis_major_unit(axis_max - axis_min)

    workbook = xlsxwriter.Workbook(str(path))
    try:
        workbook.set_properties(
            {
                "title": "TSM Box Plot",
                "subject": "Excel 2010 compatible box plot",
                "author": "Box Plot Excel Generator",
                "comments": "Generated entirely by Python without an Excel template.",
            }
        )
        workbook.set_calc_mode("auto")
        formats = _create_formats(workbook, _read_font_name(script_dir))

        main_sheet = workbook.add_worksheet(MAIN_SHEET)
        stats_sheet = workbook.add_worksheet(STATS_SHEET)
        helper_sheet = workbook.add_worksheet(HELPER_SHEET)

        chart_start_rows, chart_width, chart_height = _write_main_sheet(
            workbook, main_sheet, temperatures, data, stats, leg_data, formats
        )
        _write_stats_sheet(stats_sheet, temperatures, stats, formats)
        _write_helper_sheet(helper_sheet, temperatures, data, stats, formats)
        chart_variants = ["primary", "all", "uncalibrated", "calibrated"]
        for chart_start_row, chart_variant in zip(chart_start_rows, chart_variants):
            _add_chart(
                workbook,
                main_sheet,
                temperatures,
                chart_start_row,
                chart_width,
                chart_height,
                axis_min,
                axis_max,
                axis_major_unit,
                chart_variant=chart_variant,
            )
        main_sheet.activate()
    finally:
        workbook.close()

    _patch_generated_xlsx(path)


def generate_workbook(output_path, temperatures, overwrite=False):
    temperatures = normalize_temperatures(temperatures)
    output_path = Path(output_path).expanduser()
    if output_path.suffix.lower() != ".xlsx":
        output_path = output_path.with_suffix(".xlsx")
    output_path = output_path.resolve()

    if not output_path.parent.exists():
        raise FileNotFoundError("輸出資料夾不存在：{}".format(output_path.parent))
    if output_path.exists() and not overwrite:
        raise FileExistsError("檔案已存在：{}".format(output_path))

    file_handle, temporary_name = tempfile.mkstemp(
        prefix=".box_plot_build_",
        suffix=".xlsx",
        dir=str(output_path.parent),
    )
    os.close(file_handle)
    temporary_path = Path(temporary_name)

    try:
        _write_workbook(temporary_path, temperatures, Path(__file__).resolve().parent)
        os.replace(str(temporary_path), str(output_path))
    finally:
        if temporary_path.exists():
            temporary_path.unlink()

    return output_path


def _parse_cli_temperatures(raw_value):
    if not raw_value:
        return DEFAULT_TEMPERATURES
    return [item.strip() for item in re.split(r"[,;\n]+", raw_value) if item.strip()]


def _configured_pythonw():
    return PYTHONW_EXE if PYTHONW_EXE.is_file() else None


def _ensure_allowed_runtime():
    current_directory = Path(sys.executable).resolve().parent
    allowed_directory = PYTHONW_EXE.resolve().parent
    if current_directory != allowed_directory:
        raise RuntimeError(
            "本程式只能使用指定的可攜式 Python：{}；"
            "目前執行環境：{}".format(PYTHONW_EXE, sys.executable)
        )


def _relaunch_for_double_click():
    """Use the configured pythonw.exe for a no-console double-click launch."""
    if len(sys.argv) > 1:
        return False

    script_path = Path(__file__).resolve()
    configured = _configured_pythonw()
    if configured is None:
        raise FileNotFoundError(
            "找不到指定的 pythonw.exe，請修改 PYTHONW_EXE：{}".format(
                PYTHONW_EXE
            )
        )

    subprocess.Popen(
        [str(configured), "-I", str(script_path), "--direct-run"],
        cwd=str(script_path.parent),
    )
    return True


def _clear_error_log(script_dir):
    error_log = script_dir / ERROR_LOG_FILENAME
    if error_log.exists():
        try:
            error_log.unlink()
        except OSError:
            pass


def _write_error_log(error):
    script_dir = Path(__file__).resolve().parent
    error_log = script_dir / ERROR_LOG_FILENAME
    details = "{}\n\n{}".format(error, traceback.format_exc())
    try:
        error_log.write_text(details, encoding="utf-8")
    except OSError:
        pass


def main(argv=None):
    _ensure_allowed_runtime()
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument(
        "--generate",
        metavar="OUTPUT.xlsx",
        help="直接產生指定的 Excel 檔案",
    )
    parser.add_argument(
        "--temperatures",
        help="以逗號或分號分隔的溫度點，最多 20 個",
    )
    parser.add_argument(
        "--overwrite",
        action="store_true",
        help="允許覆寫既有輸出檔",
    )
    parser.add_argument("--direct-run", action="store_true", help=argparse.SUPPRESS)
    args = parser.parse_args(argv)

    if args.generate:
        result = generate_workbook(
            args.generate,
            _parse_cli_temperatures(args.temperatures),
            overwrite=args.overwrite,
        )
        print(result)
        return 0

    script_dir = Path(__file__).resolve().parent
    output_path = script_dir / DEFAULT_OUTPUT_FILENAME
    generate_workbook(output_path, DEFAULT_TEMPERATURES, overwrite=True)
    _clear_error_log(script_dir)
    return 0


if __name__ == "__main__":
    try:
        if _relaunch_for_double_click():
            sys.exit(0)
        sys.exit(main())
    except Exception as error:
        if "--generate" in sys.argv:
            print("錯誤：{}".format(error), file=sys.stderr)
        else:
            _write_error_log(error)
        sys.exit(1)
