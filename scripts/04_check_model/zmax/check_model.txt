! ── 逐材料體積（供下方 PER-MATERIAL BOUNDING BOX 表格的 volume/percent 欄位使用）──
! 原本用 ETABLE,,VOLU + SSUM（POST1 專用指令）取得元素體積，實測發現 SSUM/*GET,,SSUM
! 在從未求解過（無 .rst / 無 SET 過結果）時會直接報錯（Unknown label ... 被忽略）。
! 改用 *VGET,,ELEM,,GEOM 直接指定 ENTNUM 空白時，*VGET 會依「元素編號 1,2,3...」依序抓，
! 跟目前選取的元素完全無關（官方文件寫明：不論該元素是否存在/被選取都會照編號往下抓），
! 導致抓到別的材料、甚至不存在的元素編號（"Some entities...were undefined" 警告即為此因）。
! 正確做法：先用 *VGET,,ELEM,,ELIST 取得「目前實際選取」的元素編號緊湊清單，
! 再逐一用 *GET,,ELEM,N,VOLU（單一元素查詢，PREP7 資料庫層級，不需要求解結果）取體積、加總。
*dim,vol_mat,array,mmax
vol_total = 0
*DO,m,1,mmax
    ALLSEL,ALL
    ESEL,S,MAT,,m
    *GET,ne,ELEM,0,COUNT
    *IF,ne,EQ,0,THEN
        vol_mat(m) = 0
        *CYCLE
    *ENDIF
    *del,elist_pre_   $  *del,evol_pre_
    *dim,elist_pre_,array,ne
    *vget,elist_pre_,elem,,elist
    *dim,evol_pre_,array,ne
    *do,i_pre_,1,ne
        *get,evol_pre_(i_pre_),elem,elist_pre_(i_pre_),volu
    *enddo
    *vscfun,vol_mat(m),sum,evol_pre_
    vol_total = vol_total + vol_mat(m)
*ENDDO

    *DO,safety_,1,ne
        CMSEL,S,cm_remain
        *GET,ne_remain_,ELEM,0,COUNT
        *IF,ne_remain_,EQ,0,THEN
            *EXIT
        *ENDIF
        *GET,seed_elem_,ELEM,0,NUM,MIN
        ESEL,S,ELEM,,seed_elem_
        *DO,igrow_,1,ne
            *GET,ne_before_,ELEM,0,COUNT
            NSLE,S,ALL
            ESLN,S,0
            CMSEL,R,cm_remain
            *GET,ne_after_,ELEM,0,COUNT

            *ENDIF
        *ENDDO

    ! 第二輪：重跑同一套決定性 flood-fill（種子挑選規則相同，順序與第一輪一致），
    ! 這次把每層的 bbox/體積存進陣列，最後依 zmin 排序輸出

        *GET,lay_zmin_(ib_),NODE,0,MNLOC,Z  $  *GET,lay_zmax_(ib_),NODE,0,MXLOC,Z

        ! 不管這層 XY 形狀多不規則（有沒有孔洞）都成立，不用另外做投影/取樣運算。
        ! 有些材料（如接觸/膠合面元素）厚度為 0，此時 AREA 無意義，直接填 0 避免除以零中斷腳本

            area_k_ = 0
        *ELSE
            area_k_ = lay_vol_(orig_i_)/(lay_zmax_(orig_i_)-lay_zmin_(k_))
        *ENDIF
        *VLEN,1        ! 只印這一列：*VWRITE 對陣列索引預設會從起始索引印到陣列結尾，不加這行會重複列印
        *VWRITE,m,k_,lay_xmin_(orig_i_),lay_ymin_(orig_i_),lay_zmin_(k_),lay_xmax_(orig_i_),lay_ymax_(orig_i_),lay_zmax_(orig_i_),lay_xmax_(orig_i_)-lay_xmin_(orig_i_),lay_ymax_(orig_i_)-lay_ymin_(orig_i_),lay_zmax_(orig_i_)-lay_zmin_(k_),lay_vol_(orig_i_),vol_pct_k_,area_k_
(F5.0,F5.0,9F10.4,F20.2,F10.4,F20.4)
    *enddo
*ENDDO

            *ENDDO

            *dim,lay_xmin_,array,nband_m_  $  *dim,lay_xmax_,array,nband_m_
            *dim,lay_ymin_,array,nband_m_  $  *dim,lay_ymax_,array,nband_m_
            *dim,lay_zmin_,array,nband_m_  $  *dim,lay_zmax_,array,nband_m_
            *dim,lay_vol_,array,nband_m_
            *dim,lay_perm_,array,nband_m_

            ALLSEL,ALL
            ESEL,S,MAT,,m
            CM,cm_remain,ELEM
            ib_ = 0
            *DO,safety_,1,ne
                CMSEL,S,cm_remain
                *GET,ne_remain_,ELEM,0,COUNT
                *IF,ne_remain_,EQ,0,THEN
                    *EXIT
                *ENDIF
                *GET,seed_elem_,ELEM,0,NUM,MIN
                ESEL,S,ELEM,,seed_elem_
                *DO,igrow_,1,ne
                    *GET,ne_before_,ELEM,0,COUNT
                    NSLE,S,ALL
                    ESLN,S,0
                    CMSEL,R,cm_remain
                    *GET,ne_after_,ELEM,0,COUNT
                    *IF,ne_after_,EQ,ne_before_,THEN
                        *EXIT
                    *ENDIF
                *ENDDO
                ib_ = ib_ + 1
                CM,cm_layer,ELEM

                NSLE,S
                *GET,lay_xmin_(ib_),NODE,0,MNLOC,X  $  *GET,lay_xmax_(ib_),NODE,0,MXLOC,X
                *GET,lay_ymin_(ib_),NODE,0,MNLOC,Y  $  *GET,lay_ymax_(ib_),NODE,0,MXLOC,Y
                *GET,lay_zmin_(ib_),NODE,0,MNLOC,Z  $  *GET,lay_zmax_(ib_),NODE,0,MXLOC,Z

                CMSEL,S,cm_layer
                *GET,ne_layer_,ELEM,0,COUNT
                *del,elist_band_  $  *del,evol_band_
                *dim,elist_band_,array,ne_layer_
                *vget,elist_band_,elem,,elist
                *dim,evol_band_,array,ne_layer_
                *do,i_band_,1,ne_layer_
                    *get,evol_band_(i_band_),elem,elist_band_(i_band_),volu
                *enddo
                *vscfun,lay_vol_(ib_),sum,evol_band_

                CMSEL,S,cm_remain
                CMSEL,U,cm_layer
                CM,cm_remain,ELEM
            *ENDDO

            *moper,lay_perm_,lay_zmin_,sort,,1

            *do,k_,1,nband_m_
                orig_i_ = lay_perm_(k_)
                vol_pct_k_ = lay_vol_(orig_i_)/vol_total*100
                *IF,(lay_zmax_(orig_i_)-lay_zmin_(k_)),EQ,0,THEN
                    area_k_ = 0
                *ELSE
                    area_k_ = lay_vol_(orig_i_)/(lay_zmax_(orig_i_)-lay_zmin_(k_))
                *ENDIF
                gb_idx_ = gb_idx_ + 1
                gb_m_(gb_idx_)    = m                   $  gb_k_(gb_idx_)    = k_
                gb_xmin_(gb_idx_) = lay_xmin_(orig_i_)  $  gb_xmax_(gb_idx_) = lay_xmax_(orig_i_)
                gb_ymin_(gb_idx_) = lay_ymin_(orig_i_)  $  gb_ymax_(gb_idx_) = lay_ymax_(orig_i_)
                gb_zmin_(gb_idx_) = lay_zmin_(k_)       $  gb_zmax_(gb_idx_) = lay_zmax_(orig_i_)
                gb_vol_(gb_idx_)  = lay_vol_(orig_i_)   $  gb_pct_(gb_idx_)  = vol_pct_k_
                gb_area_(gb_idx_) = area_k_
            *enddo
        *ENDDO

        ! ── 全表依 zmax 由大到小排序後才印出（*MOPER,,,SORT 只能升冪，取負再排序達成降冪）──
        *IF,gb_total_,GT,0,THEN
            *del,gb_negzmax_ $ *del,gb_perm_
            *dim,gb_negzmax_,array,gb_total_
            *do,k_,1,gb_total_
                gb_negzmax_(k_) = -gb_zmax_(k_)
            *enddo
            *dim,gb_perm_,array,gb_total_
            *moper,gb_perm_,gb_negzmax_,sort,,1

            *do,k_,1,gb_total_
                orig_i_ = gb_perm_(k_)
                *VLEN,1
                *VWRITE,gb_m_(orig_i_),gb_k_(orig_i_),gb_xmin_(orig_i_),gb_ymin_(orig_i_),gb_zmin_(orig_i_),gb_xmax_(orig_i_),gb_ymax_(orig_i_),gb_zmax_(orig_i_),gb_xmax_(orig_i_)-gb_xmin_(orig_i_),gb_ymax_(orig_i_)-gb_ymin_(orig_i_),gb_zmax_(orig_i_)-gb_zmin_(orig_i_),gb_vol_(orig_i_),gb_pct_(orig_i_),gb_area_(orig_i_)
(F5.0,F5.0,9F10.4,F20.2,F10.4,F20.4)
            *enddo
        *ENDIF
    *ELSE
        *VWRITE
(' ')
        *VWRITE,m_above
('  WARNING: m_above = ',F5.0,' has no elements, ABOVE MAT block skipped')
    *ENDIF
*ENDIF
