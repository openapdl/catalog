CSYS,0
x0 = 10      $  y0 = 10
w  = 1.1     $  l  = 2.1
pitch_x  = 0.5  $  pitch_y  = 0.5
pitch_xx = 20   $  pitch_yy = 30
nx = 5   $  ny = 5
nbx = 8  $  nby = 3
mat_new = 21                      ! ★新材料號

ESEL,NONE
*DO,bx,1,nbx
    *DO,i,1,nx
        xa = x0+(bx-1)*pitch_xx+(i-1)*(w+pitch_x)
        ESEL,A,CENT,X,xa,xa+w
    *ENDDO
*ENDDO
CM,_EX,ELEM

ESEL,NONE
*DO,by,1,nby
    *DO,j,1,ny
        ya = y0+(by-1)*pitch_yy+(j-1)*(l+pitch_y)
        ESEL,A,CENT,Y,ya,ya+l
    *ENDDO
*ENDDO

CMSEL,R,_EX
ESEL,R,CENT,Z,-0.013,0
CM,E_BUMP,ELEM

/PREP7                            ! ★EMODIF 只在 PREP7 有效
EMODIF,ALL,MAT,mat_new            ! ★已選元素改材料

CMDELE,_EX
ALLSEL,ALL





！========================

! ---- 1. 直條選了什麼 ----
CMSEL,S,_EX
*GET,n1,ELEM,0,COUNT
EPLOT                             ! 應該是 40 條分開的直條

! ---- 2. 交集之後 ----
CMSEL,S,E_BUMP
*GET,n2,ELEM,0,COUNT
EPLOT                             ! 應該是 600 顆分開的 bump

! ---- 3. 材料有沒有真的改到 ----
*GET,e1,ELEM,0,NUM,MIN
*GET,e2,ELEM,0,NUM,MAX
*GET,m1,ELEM,e1,ATTR,MAT
*GET,m2,ELEM,e2,ATTR,MAT

/GO
*MSG,INFO,n1,n2,m1,m2
EX=%I  BUMP=%I  first MAT=%I  last MAT=%I

！========================


CSYS,0
x0 = 10      $  y0 = 10
w  = 1.1     $  l  = 2.1
pitch_x  = 0.5  $  pitch_y  = 0.5
pitch_xx = 20   $  pitch_yy = 30
nx = 5   $  ny = 5
nbx = 8  $  nby = 3
mat_new = 21

/PREP7
*DO,by,1,nby
    *DO,bx,1,nbx
        *DO,j,1,ny
            *DO,i,1,nx
                xa = x0+(bx-1)*pitch_xx+(i-1)*(w+pitch_x)
                ya = y0+(by-1)*pitch_yy+(j-1)*(l+pitch_y)

                ESEL,S,CENT,X,xa,xa+w          ! 這一顆的 X 範圍
                ESEL,R,CENT,Y,ya,ya+l          ! 再篩 Y
                ESEL,R,CENT,Z,-0.013,0         ! 再篩厚度
                EMODIF,ALL,MAT,mat_new         ! 當場改材料
            *ENDDO
        *ENDDO
    *ENDDO
*ENDDO

ESEL,S,MAT,,mat_new                            ! 全部改完，用材料號一次選回來
CM,E_BUMP,ELEM
ALLSEL,ALL
