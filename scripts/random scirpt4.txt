
/PREP7

! =============== Unit bump 參數設定 ===============
! ── Die 位置與尺寸 ──
Die_position_x = 1.705
Die_position_y = 2.005
Die_size_x     = 11.490
Die_size_y     = 7.990

! ========= 單顆 bump 參數
bump_count_x    = 5
bump_count_y    = 5
total_bump_area = 8.06258338617285

! ========= bump 陣列複製設定
kp_start = 1000000        ! TODO: 使用者自行設定：keypoint 起始編號（腳本其他地方已固定用掉 id=1，避免這裡產生的 keypoint 撞號）
die_pitch_x = 14.25    ! TODO: 使用者自行設定：+X 方向陣列間距
die_pitch_y = 12.25    ! TODO: 使用者自行設定：+Y 方向陣列間距
die_count_x = 8    ! +X 方向總份數（含原始那一份）
die_count_y = 3     ! +Y 方向總份數（含原始那一份）

! ========= 背景 area 座標範圍（涵蓋 +X/+Y 方向複製後的完整陣列，範圍隨 die_count_x/y、die_pitch_x/y 自動延伸）
bg_x1 = Die_position_x
bg_y1 = Die_position_y
bg_x2 = Die_position_x + Die_size_x + (die_count_x-1)*die_pitch_x
bg_y2 = Die_position_y + Die_size_y + (die_count_y-1)*die_pitch_y

! ========= 其他設定
loc_tol = 1e-6   ! TODO: 使用者自行設定：ASEL,LOC 座標比對容差，避免浮點數誤差漏選

! ========= 腳本流程開始，建立unit bump area出來

bump_count_per_die = bump_count_x*bump_count_y

*del,bump_xy
*dim,bump_xy,array,bump_count_per_die,4

NUMSTR,KP,kp_start

bump_area = total_bump_area / (bump_count_x*bump_count_y)   ! [FIX] * 前不可有空白，否則該行被當成註解截斷（見APDL手冊3.7節）
bump_w    = sqrt(bump_area)
bump_h    = sqrt(bump_area)
pitch_x = (Die_size_x  - bump_count_x*bump_w) / (bump_count_x + 1)
pitch_y = (Die_size_y - bump_count_y*bump_h) / (bump_count_y + 1)
start_cx = Die_position_x + pitch_x + bump_w / 2
start_cy = Die_position_y + pitch_y + bump_h / 2

*DO,j,1,bump_count_y
    *DO,i,1,bump_count_x
        xc = start_cx + (i-1)*(bump_w + pitch_x)
        yc = start_cy + (j-1)*(bump_h + pitch_y)
        BLC5, xc, yc, bump_w, bump_h, 0

        bump_idx = (j-1)*bump_count_x + i
        bump_xy(bump_idx,1) = xc - bump_w/2   ! x1
        bump_xy(bump_idx,2) = yc - bump_h/2   ! y1
        bump_xy(bump_idx,3) = xc + bump_w/2   ! x2
        bump_xy(bump_idx,4) = yc + bump_h/2   ! y2
    *ENDDO
*ENDDO

! ========= 依座標重新選出 bump_base（不依賴面號連續區間，避免公司模型既有幾何造成編號不連續而漏選）
first_base_cell = 1
*DO,b,1,bump_count_per_die
    ALLSEL,ALL
    ASEL,S,LOC,X,bump_xy(b,1)-loc_tol,bump_xy(b,3)+loc_tol
    ASEL,R,LOC,Y,bump_xy(b,2)-loc_tol,bump_xy(b,4)+loc_tol
    *get,base_cell_cnt,area,0,count          ! 這格實際選到幾個面：0 代表這格沒建出面（BLC5 失敗），需留意
    *if,base_cell_cnt,eq,0,then
        *msg,WARN,b
這格(bump_idx=%I)沒有選到任何面，BLC5 可能未成功建立，請檢查
    *endif
    *if,base_cell_cnt,gt,0,then
        *if,first_base_cell,eq,1,then
            CM,bump_base,AREA                 ! 第一格：把目前選到的（可能不只一個）面直接存成 component
            first_base_cell = 0
        *else
            CM,bump_base_tmp,AREA             ! 這一格選到的所有面，先存成暫存 component
            CMSEL,A,bump_base,AREA            ! 加回目前累積結果
            CMSEL,A,bump_base_tmp,AREA        ! 再加上這一格新選到的
            CM,bump_base,AREA                 ! 重新存回累積 component（等於聯集）
        *endif
    *endif
*ENDDO
ALLSEL,ALL                                    ! bump_base 已存成 component，供下面 AGEN 陣列複製使用
! APLOT

! ========= 把unit bump area 進行陣列(+X 方向複製)
*get,area_max_x0,area,0,num,max
AGEN,die_count_x,bump_base, , ,die_pitch_x, , ,0,0,0
*get,area_max_x1,area,0,num,max
ASEL,S,AREA,,area_max_x0+1,area_max_x1
CMSEL,A,bump_base,AREA
CM,bump_x_array,AREA                              ! X 方向複製完成後的完整陣列

! ========= +Y 方向複製(以 X 方向結果為基礎)
*get,area_max_y0,area,0,num,max
AGEN,die_count_y,bump_x_array, , , ,die_pitch_y, ,0,0,0
*get,area_max_y1,area,0,num,max
ASEL,S,AREA,,area_max_y0+1,area_max_y1
CMSEL,A,bump_x_array,AREA
CM,bump_full_array,AREA                           ! 最終完整 bump 陣列 component（die_count_x × die_count_y 組）
! APLOT

! ========= 選出 Z 0~-0.013 範圍內、對應 bump_full_array 的 volume
ALLSEL,ALL
CMSEL,S,bump_full_array,AREA
VSLA,S,0                                     ! 選出與這些 bump 面相連的 volume（VLKEY=0：面只要有一部分在選取集合內就算）
VSEL,R,LOC,Z,-0.013-loc_tol,0+loc_tol         ! 再用 Z 範圍篩選，避免選到其他 Z 位置、但同樣連到這些面的 volume
CM,bump_vol_array,VOLU                        ! 存成 component，供後續使用
VATT,25                                       ! 把目前選取的 bump volume 材料改成 MAT=25
ALLSEL,ALL

! ========= 選出 bump_full_array 座標範圍對應的元素，並換材料
ALLSEL,ALL
CMSEL,S,bump_vol_array,VOLU
ESLV,S                                        ! 選出附屬於這些 bump volume 的元素
EMODIF,ALL,MAT,25                             ! 把選取的元素材料改成 MAT=25
ALLSEL,ALL

! ========= 依座標選出 bump 所在的另一個面，建立 bg_area component
ALLSEL,ALL
ASEL,S,LOC,X,bg_x1-loc_tol,bg_x2+loc_tol
ASEL,R,LOC,Y,bg_y1-loc_tol,bg_y2+loc_tol
CMSEL,U,bump_full_array,AREA                  ! 排除 bump 面本身，只留下承載 bump 的那個背景大面
CM,bg_area,AREA                               ! 存成 component，方便後面 CMSEL/ASBA 取用
ALLSEL,ALL

boptn,nwarn,0    ! 布林運算「沒效果」時跳警告（預設值）
btol,1e-3        ! 布林運算點重合容差，放寬到 1e-3（預設 1e-5）以吸收浮點誤差，提高布林運算成功率

! ========= 布林第一種作法：Boolean Overlap（兩面重疊處切成共用面，供後續網格共節點用）
boptn,keep,0                                  ! Overlap 後保留輸入面（AOVLAP 預設即保留重疊前的原面，這裡維持預設）
CMSEL,S,bg_area,AREA
CMSEL,A,bump_full_array,AREA
AOVLAP,ALL                                    ! 兩面重疊部分切成共用面，供後續網格共用節點用
ALLSEL,ALL

! ========= 布林第二種作法：Boolean Subtract，KEEP,KEEP（背景面、bump 面兩個輸入面都保留，另外產生扣除後的新面）
! 注意：KEEP,KEEP 會讓原本重疊的背景面/bump 面跟新產生的扣除後面「並存」，屬於重疊但不共節點的幾何，
!       跟上面 Overlap 共節點的目的不同，若要共節點網格請用上面那段，不要同時啟用這段。
! ASBA,bg_area,bump_full_array,,KEEP,KEEP      ! NA1=bg_area、NA2=bump_full_array 都保留，額外產生扣除後的新面
! ALLSEL,ALL
